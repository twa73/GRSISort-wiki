72 bool TEventBuildingLoop::Iteration()
73 {
74  // Pull something off of the input queue.
75  std::shared_ptr<const TFragment> input_frag = nullptr;
76  fInputSize = fInputQueue->Pop(input_frag, 0);
77  if(fInputSize < 0) {
78  fInputSize = 0;
79  }
80 
81  if(input_frag != nullptr) {
82  ++fItemsPopped;
83  if(!fSkipInputSort) {
84  fOrdered.insert(input_frag);
85  if(fOrdered.size() < fSortingDepth) {
86  // Got a new event, but we want to have more to sort
87  return true;
88  }
89  // Got a new event, and we have enough to sort.
90  }
91  } else {
92  if(!fInputQueue->IsFinished()) {
93  // If the parent is live, wait for it
94  std::this_thread::sleep_for(std::chrono::milliseconds(1000));
95  return true;
96  }
97  if(fOrdered.empty()) {
98  // Parent is dead, and we have passed on all events
99  // check if last event needs to be pushed
100  if(!fNextEvent.empty()) {
101  fOutputQueue->Push(fNextEvent);
102  }
103  fOutputQueue->SetFinished();
104  return false;
105  }
106  // Parent is dead, but we still have items.
107  // Continue through the function to process them.
108  }
109 
110  // We have data, and we want to add it to the next fragment;
111  std::shared_ptr<const TFragment> next_fragment;
112  if(!fSkipInputSort) {
113  next_fragment = *fOrdered.begin();
114  fOrdered.erase(fOrdered.begin());
115  } else {
116  next_fragment = input_frag;
117  }
118 
119  if(CheckBuildCondition(next_fragment)) {
120  fNextEvent.push_back(next_fragment);
121  }
122 
123  return true;
124 }
