170 void TChannel::AddChannel(TChannel* chan, Option_t* opt)
171 {
172  /// Add a TChannel to fChannelMap. If the TChannel doesn't exist, create a new TChannel and add that the fChannelMap.
173  /// Options:
174  /// "overwrite" - The TChannel in the fChannelMap at the same address is overwritten.
175  /// If this option is not specified, an Error is returned if the TChannel already
176  /// exists in the fChannelMap.
177  /// "save" - The temporary channel is not deleted after being placed in the map.
178  if(chan == nullptr) {
179  return;
180  }
181  if(fChannelMap->count(chan->GetAddress()) == 1) { // if this channel exists
182  if(strcmp(opt, "overwrite") == 0) {
183  TChannel* oldchan = GetChannel(chan->GetAddress());
184  int oldNumber = oldchan->GetNumber();
185  oldchan->OverWriteChannel(chan);
186  // Need to also update the channel number map RD
187  if(oldNumber != oldchan->GetNumber()) {
188  // channel number has changed so we need to delete the old one and insert the new one
189  fChannelNumberMap->erase(oldNumber);
190  if((oldchan->GetNumber() != 0) && (fChannelNumberMap->count(oldchan->GetNumber()) == 0)) {
191  fChannelNumberMap->insert(std::make_pair(oldchan->GetNumber(), oldchan));
192  }
193  }
194  return;
195  }
196  std::cout<<"Trying to add a channel that already exists!"<<std::endl;
197  return;
198  }
199  if((chan->GetAddress() & 0x00ffffff) == 0x00ffffff) {
200  // this is the default tigress value for i am not there.
201  // we should not imclude it in the map.
202  delete chan;
203  } else {
204  // We need to update the channel maps to correspond to the new channel that has been added.
205  fChannelMap->insert(std::make_pair(chan->GetAddress(), chan));
206  if((chan->GetNumber() != 0) && (fChannelNumberMap->count(chan->GetNumber()) == 0)) {
207  fChannelNumberMap->insert(std::make_pair(chan->GetNumber(), chan));
208  }

983 {

987 
988  std::string line;
989  int linenumber = 0;
990  int newchannels = 0;
991 
992  // bool creatednewchannel = false;
993  bool brace_open = false;
994  // int detector = 0;
995  std::string name;
996 
997  // Parse the cal file. This is useful because if the cal file contains something that
998  // the parser does not recognize, it just skips it!
999  while(std::getline(infile, line)) {
1000  linenumber++;
1001  trim(line);
1002  size_t comment = line.find("//");
1003  if(comment != std::string::npos) {
1004  line = line.substr(0, comment);
1005  }
1006  if(line.length() == 0u) {
1007  continue;
1008  }
1009  size_t openbrace = line.find('{');
1010  size_t closebrace = line.find('}');
1011  size_t colon = line.find(':');
1012 
1013  if(openbrace == std::string::npos && closebrace == std::string::npos && colon == std::string::npos) {
1014  continue;
1015  }
1016 
1017  //*************************************//
1018  if(closebrace != std::string::npos) {
1019  brace_open = false;
1020  if(channel != nullptr) {
1021  TChannel* currentchan = GetChannel(channel->GetAddress(), false);
1022  if(currentchan == nullptr) {
1023  AddChannel(channel); // consider using a default option here
1024  newchannels++;
1025  } else {
1026  currentchan->UpdateChannel(channel);
1027  delete channel;
1028  newchannels++;
1029  }
1030  } else {
1031  delete channel;
1032  }
1033  channel = nullptr;
1034  name.clear();
1035  }
1036  //*************************************//
1037  if(openbrace != std::string::npos) {
1038  brace_open = true;
1039  name = line.substr(0, openbrace);
1040  trim(name);
1041  channel = new TChannel("");
1042  if(!name.empty()) channel->SetName(name.c_str());
1043  }
1044  //*************************************//
1045  if(brace_open) {
1046  size_t ntype = line.find(':');
1047  if(ntype != std::string::npos) {
1048  std::string type = line.substr(0, ntype);
1049  line = line.substr(ntype + 1, line.length());
1050  trim(line);
1051  std::istringstream ss(line);
1052  // transform type to upper case
1053  std::transform(type.begin(), type.end(), type.begin(), ::toupper);
1054  if(type.compare("NAME") == 0) {
1055  channel->SetName(line.c_str());
1056  } else if(type.compare("ADDRESS") == 0) {
1057  unsigned int tempadd = 0;
1058  ss >> tempadd;

1061  newss<<std::hex<<line;
1062  newss >> tempadd;
1063  }

1066  } else if(type.compare("INTEGRATION") == 0) {
1067  int tempint;
1068  ss >> tempint;

1070  } else if(type.compare("NUMBER") == 0) {

1080  ss >> tempstream;

1082  } else if(type.compare("DIGITIZER") == 0) {
1083  channel->SetDigitizerType(TPriorityValue<std::string>(line, pr));
1084  } else if(type.compare("ENGCHI2") == 0) {
