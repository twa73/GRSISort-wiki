161 TVector3 TTrific::GetPosition(Int_t detectorNumber)
162 {
163  //this is called on a TRIFIC hit, and will use the GetPosition() function below to calculate
164  //the x,y,z position of the hit at that grid number
165  //gives outputs in (x,y,z) in cartesian (beam) coordinates
166  // the coordinates (0,0,0) correspond to the centre of the target location, where (presumably) the beam is hitting the target
167 
168  //detectorNumber is indexed at 1.
169 
170  //TRIFIC only holds 24 detectors, so doesn't make sense to get the position for detectors 25+. Also doesn't make sense to get the position for grid numbers <1.
171  if (24 < detectorNumber || 1 > detectorNumber) return TVector3(-1,-1,-1*abs(detectorNumber));
172  //-1*abs(detNum) ensures we return a value of (-1,-1,-#), which indicates a problem
173 
174  TVector3 vec = TTrific::GetPosition();
175 
176  //check if TTrific::GetPosition() returned an error vector for bad reconstruction. If so, we don't need to do anything more and can return an error vector
177  if (TVector3(-100,-100,-100) == vec) return vec; //this vector should be well outside expected XY coordinates
178 
179  double zCart = fTargetToWindowCart + fInitialSpacingCart + fSpacingCart*detectorNumber; //cartesian distance to the grid of choice in the z-direction
180  //this ignores the extra (or lack of) Z due to the tilt of the grids
181  //this includes the target-to-window distance. This can be set via SetSharc() or SetTip() (or via other yet-defined functions)
182 
183  zCart += zCart*vec.Y()/(TMath::Sqrt(3)-vec.Y()); //this adds (or subtracts) the extra Z distance (Z') due to the offset in Y (which are tilted at 30 degs. from vertical)
184  //notes on geometry: 30-60-90 triangle, so Y=sqrt(3)*Z', and Y/(Z+Z') = tan(theta)
185 
186  TVector3 particleCart(zCart*vec.X(),zCart*vec.Y(),zCart);
187 
188  //return TVector3(0,0,detectorNumber);
189  return particleCart;
190 }

192 TVector3 TTrific::GetPosition()
193 {
194  //Will calculate the x,y,z vector for the TRIFIC event and then return a 3-D vector
195  //that represents the TRIFIC event
196 
197  //check if we've already calculated the position for this event. If so, just return it. If not, then reset the position variable.
198  if (fTrificBits.TestBit(ETrificBits::kPositionCalculated)) return fParticle;
199  else fParticle.SetXYZ(0,0,0);
200  //flag that we're going to (try to at least) calculate the position vector for this event
201  fTrificBits.SetBit(ETrificBits::kPositionCalculated,true);
202 
203  //Get the grids that are the X and Y grids in this setup
204  GetXYGrid();
205 
206  //if we don't have both an x and y grid hit in this event, position reconstruction won't be possible.
207  if(0 == fXFragments.size() || 0 == fYFragments.size()){
208  //return an error vector
209  fParticle.SetXYZ(-100,-100,-100);
210  return fParticle;
211  }
212 
213  double xMean = 0.0;
214  double xEngTotal = 0.0;
215 
216  std::vector<int> hitXDets; //vector to hold the x-segments that have a nonzero hit in them
217 
218  for (auto hit: fXFragments){
219  //TDetectorHit *hit = fXFragments.at(i);
220  //TDetectorHit *hit = i;
221  Int_t seg = hit->GetSegment();
222 
223  if (3 > hit->GetEnergy()) continue; //arbitrary threshold to avoid weird E<0 events
224 
225  xMean += fXmm[seg]*hit->GetEnergy();
226  xEngTotal += hit->GetEnergy();
227 
228  hitXDets.push_back(seg);
229  }
230 
231  //check if hitXDets.size() is zero. This happens when we have an x-grid hit or hits, but the energy for all hits is below our arbitrary "noise" threshold.
232  //without this, the function will seg-fault when it tries to check for the continuity
233  if (!hitXDets.size()){
234  //return an error vector
235  fParticle.SetXYZ(-100,-100,-100);
236  return fParticle;
237  }
238 
239  //to check for hit continuity in the grids, we need to sort the vector of segments and then check that every segment
240  //between the first and last segment with a nonzero energy in the array is present. If not, we have a discontinuous hit
241  //and will return an error vector for now. in the future we might instead flag this event
242  //Ex: we want hits that like this: [0,0,4,3,5,6,5,3] not [0,0,5,0,0,2,3,0]
243  //where the index of the example vector is the segment number and the value is the energy
244 
245  std::sort(hitXDets.begin(),hitXDets.end());
246 
247  for (unsigned int i = 1; i< hitXDets.size(); i++){ //note: this currently just rejects pileup events outright. The class will be modified later to deal with pileup properly
248  if(hitXDets[i] != hitXDets[i-1]+1){
249  fParticle.SetXYZ(-100,-100,-100);
250  return fParticle;
251  }
252  }
253 
254  //divide weighted mean by total energy
255  xMean /= xEngTotal;
256 
257  //now we do the same for the y-hits
258 
259  double yMean = 0.0;
260  double yEngTotal = 0.0;
261 
262  std::vector<int> hitYDets; //vector to hold the y-segments that have a nonzero hit in them
263 
264  for (auto hit: fYFragments){
265  //TDetectorHit *hit = i;
266  Int_t seg = hit->GetSegment();
267 
268  if (3 > hit->GetEnergy()) continue; //arbitrary threshold to avoid weird E<0 events
269 
270  yMean += fYmm[seg]*hit->GetEnergy();
271  yEngTotal += hit->GetEnergy();
272 
273  hitYDets.push_back(seg);
274  }
275 
276  //check if hitYDets.size() is zero. This happens when we have an y-grid hit or hits, but the energy for all hits is below our arbitrary "noise" threshold.
277  //without this, the function will seg-fault when it tries to check for the continuity
278  if (!hitYDets.size()){
279  //return an error vector
280  fParticle.SetXYZ(-100,-100,-100);
281  return fParticle;
282  }
283 
284  //check again for continuity
285 
286  std::sort(hitYDets.begin(),hitYDets.end());
287 
288  //this will search through the sorted vector to ensure continuity
289  for (unsigned int i = 1; i< hitYDets.size(); i++){
290  if(hitYDets[i] != hitYDets[i-1]+1){ //again, this rejects pileup events
291  fParticle.SetXYZ(-100,-100,-100);
292  return fParticle;
293  }
294  }
295 
296  //divide weighted mean by total energy
297  yMean /= yEngTotal;
298 
299  //convert them into cartesion coordinates
300  double yCart = yMean*TMath::Sin(fAngle); //shifts from grid coordinates to XYZ coordinates
301  double zYCart = fTargetToWindowCart + fInitialSpacingCart + fSpacingCart*fGridY+yMean*TMath::Cos(fAngle); //add the initial distance from target to grid, window to the grid, plus the number of gaps between the initial grid and the Y grid, plus the extra z-amount due to the hit location
302  double zXCart = fTargetToWindowCart + fInitialSpacingCart + fSpacingCart*fGridX; //adds the initial distance from target to grid, window to grid, plus gaps until the x grid. No angle dependence since the grids aren't shifted in angle in x-direction
303 
304  double tanX = xMean/zXCart; //determine tangent of the angle in the XY plane
305  double tanY = yCart/zYCart; //tan of angle in YZ plane
306 
307  //the particle trajectory below is working under the assumption that the beam is centered and hitting the target at exactly (0,0,0)
308  //while this may not be 100% true, the beam should be tuned enough that any deviations from that will be small
309 
310  fParticle.SetXYZ(tanX,tanY,1); //unnormalized "unit" vector of the particle's trajectory.
311  // "X" and "Y" coordinates are the tangent of the angles of the trajectory in the XY and YZ planes, respectively
312  //this is all done relative to the target centre
313 
314  //TVector3 particleCart = TVector3(xMean,yCart,1); //this makes the vector in cartesian as opposed to angles.
315 
316  //double ratioZX = 1./fNormalGridVec.Dot(fParticle.Unit()); //the Z->R corretion factor
317 
318  //fParticle.Print();
319 
320  return fParticle;
321 }

323 Int_t TTrific::GetRange()
324 {
325  //Gets the last grid with an event in it
326  //can't just use the size of fHits because there is no guarantee that every
327  //grid gets an event, plus the XY position grids may have multiplicity>1
328 
329  //check if we've already calculated the fRange for this event. If so, return it. If not, we need to reset the range.
330  if (fTrificBits.TestBit(ETrificBits::kRangeCalculated)) return fRange;
331  else fRange=0;
332 
333  //first we'll check the single grid fragment, since there are more of them and they extend further
334  for (auto hit: fSingFragments){
335  if(hit->GetDetector() > fRange) fRange = hit->GetDetector();
336  }
337  //check if the range is less than the furthest position grid. If so, we need to check that the range isn't at the x or y grid
338  //this was changed because there is no guarantee that the x and y grids will stay at positions 3 and 5, so this is generic
339 
340  //determine the x and y grid locations
341  GetXYGrid();
342 
343  //check if the range is less than the max of the grid numbers+1. The +1 is because the higher grid number will be at std::max(xGrid,yGrid) by default
344  if (fRange < std::max(fGridX,fGridY)+1){
345 
346  if (fRange < fGridX && fXFragments.size()) fRange = fGridX; //we need to check if the current range is less than the x grid AND that there is an x-grid hit in this event
347  if (fRange < fGridY && fYFragments.size()) fRange = fGridY; //we need to check if the current range is less than the y grid AND that there is a y-grid hit in this event
348  } //there may be a problem here if fXFragments.size() != 0 (or Y frags too) but all the fragments have E<arb cutoff. However, GetRange() isn't referenced in any other function currently,
349  //and the likelihood of that occuring is very small I think.
350 
351  //flag that we've calculated the range for this event
352  fTrificBits.SetBit(ETrificBits::kRangeCalculated,true);
353 
354  return fRange;
355 }
