549 int TMidasEvent::SwapBytes(bool force)
550 {
551  // Swaps bytes for endian-ness reasons
552  TMidas_BANK_HEADER* pbh;
553  TMidas_BANK* pbk;
554  TMidas_BANK32* pbk32;
555  void* pdata;
556  uint16_t type;
557 
558  pbh = reinterpret_cast<TMidas_BANK_HEADER*>(fData);
559 
560  uint32_t dssw = pbh->fDataSize;
561 
562  DWORD_SWAP(&dssw);
563 
564  // only swap if flags in high 16-bit
565  //
566  if(pbh->fFlags < 0x10000 && !force) {
567  return 0;
568  }
569 
570  if(pbh->fDataSize == 0x6d783f3c) { // string "<xml..." in wrong-endian format
571  return 1;
572  }
573 
574  if(pbh->fDataSize == 0x3c3f786d) { // string "<xml..."
575  return 1;
576  }
577 
578  if(dssw > fEventHeader.fDataSize + 100) { // swapped data size looks wrong. do not swap.
579  return 1;
580  }
581 
582  //
583  // swap bank header
584  //
585  DWORD_SWAP(&pbh->fDataSize);
586  DWORD_SWAP(&pbh->fFlags);
587  //
588  // check for 32-bit banks
589  //
590  bool b32 = IsBank32();
591 
592  pbk = reinterpret_cast<TMidas_BANK*>(pbh + 1);
593  pbk32 = reinterpret_cast<TMidas_BANK32*>(pbk);
594  //
595  // scan event
596  //
597  while(reinterpret_cast<char*>(pbk) < reinterpret_cast<char*>(pbh) + pbh->fDataSize + sizeof(TMidas_BANK_HEADER)) {
598  //
599  // swap bank header
600  //
601  if(b32) {
602  DWORD_SWAP(&pbk32->fType);
603  DWORD_SWAP(&pbk32->fDataSize);
604  pdata = pbk32 + 1;
605  type = static_cast<uint16_t>(pbk32->fType);
606  } else {
607  WORD_SWAP(&pbk->fType);
608  WORD_SWAP(&pbk->fDataSize);
609  pdata = pbk + 1;
610  type = pbk->fType;
611  }
612  //
613  // pbk points to next bank
614  //
615  if(b32) {
616  assert(pbk32->fDataSize < fEventHeader.fDataSize + 100);
617  pbk32 = reinterpret_cast<TMidas_BANK32*>(reinterpret_cast<char*>(pbk32 + 1) + (((pbk32->fDataSize) + 7) & ~7));
618  pbk = reinterpret_cast<TMidas_BANK*>(pbk32);
619  } else {
620  assert(pbk->fDataSize < fEventHeader.fDataSize + 100);
621  pbk = reinterpret_cast<TMidas_BANK*>(reinterpret_cast<char*>(pbk + 1) + (((pbk->fDataSize) + 7) & ~7));
622  pbk32 = reinterpret_cast<TMidas_BANK32*>(pbk);
623  }
624 
625  switch(type) {
626  case 4:
627  case 5:
