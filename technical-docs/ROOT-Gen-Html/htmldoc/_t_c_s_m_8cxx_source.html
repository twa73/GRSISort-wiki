23 void TCSM::AddFragment(const std::shared_ptr<const TFragment>& frag, TChannel* chan)
24 {
25  /// This function just stores the fragments and mnemonics in vectors, separated by detector number and type
26  /// (horizontal/vertical strip or pad).
27  /// The hits themselves are built in the BuildHits function because the way we build them depends on the number of
28  /// hits.
29 
30  // first index: detector number, second index: 0 = deltaE, 1 = E; third index: 0 = horizontal, 1 = vertical; fourth
31  // index: fragments
32  int type = -1;
33  if(chan->GetMnemonic()->ArraySubPositionString().compare(0, 1, "D") == 0) {
34  type = 0;
35  } else if(chan->GetMnemonic()->ArraySubPositionString().compare(0, 1, "E") == 0) {
36  type = 1;
37  }
38  int orientation = -1;
39  if(chan->GetMnemonic()->CollectedChargeString().compare(0, 1, "N") == 0) {
40  // N = Horizontal Strips. aka "front"
41  orientation = 0;
42  } else if(chan->GetMnemonic()->CollectedChargeString().compare(0, 1, "P") == 0) {
43  // P = Vertical Strips. aka "back"
44  orientation = 1;
45  }
46 
47  if(type < 0 || orientation < 0) {
48  return;
49  }
50 
51  // if this is the first time we got this detector number we make a new vector (of a vector) of fragments
52  if(fFragments.find(chan->GetMnemonic()->ArrayPosition()) == fFragments.end()) {
53  fFragments[chan->GetMnemonic()->ArrayPosition()].resize(
54  2, std::vector<std::vector<std::pair<TFragment, TGRSIMnemonic>>>(2));
55  }
56 
57  fFragments[chan->GetMnemonic()->ArrayPosition()][type][orientation].push_back(
58  std::make_pair(*frag, *static_cast<const TGRSIMnemonic*>(chan->GetMnemonic())));
59 }

61 void TCSM::BuildHits()
62 {
63  /// This function takes the fragments that were stored in the successive AddFragment calls and builds hits out of
64  /// them
65  std::vector<std::vector<TDetectorHit*>> hits(2);
66 
67  // first index: detector number, second index: 0 = deltaE, 1 = E; third index: 0 = horizontal, 1 = vertical
68  // loop over all found detectors
69  for(auto& fFragment : fFragments) {
70  // loop over all types (only detectors 1/2 should have both D and E, detectors 3/4 should only have D)
71  for(size_t i = 0; i < fFragment.second.size(); ++i) {
72  BuildVH(fFragment.second.at(i), hits[i]);
73  }
74  }
75  BuilddEE(hits, fHits);
76 }

78 TVector3 TCSM::GetPosition(int detector, char pos, int horizontalstrip, int verticalstrip, double X, double Y, double Z)
79 {
80  // horizontal strips collect N charge!
81  // vertical strips collect P charge!
82  TVector3 Pos;
83  double SideX = 68;
84  double SideZ = -4.8834;
85  double dEX = 54.9721;
86  double dEZ = 42.948977;
87  double EX = 58.062412;
88  double EZ = 48.09198;
89  double detTheta = 31. * (TMath::Pi() / 180.);
90  double x = 0.0, y = 0.0, z = 0.0;
91 
92  if(detector == 3 && pos == 'D') {
93  // Right Side
94  verticalstrip = 15 - verticalstrip;
95  x = SideX;
96  z = SideZ + (50. / 32.) * (2 * verticalstrip + 1);
97  } else if(detector == 4 && pos == 'D') {
98  // Left Side
99  x = -SideX;
100  z = SideZ + (50. / 32.) * (2 * verticalstrip + 1);
101  } else if(detector == 1 && pos == 'D') {
102  // Right dE
103  verticalstrip = 15 - verticalstrip;
104  x = dEX - (50. / 32.) * cos(detTheta) * (2 * verticalstrip + 1);
105  z = dEZ + (50. / 32.) * sin(detTheta) * (2 * verticalstrip + 1);
106  } else if(detector == 2 && pos == 'D') {
107  // Left dE
108  x = -dEX + (50. / 32.) * cos(detTheta) * (2 * verticalstrip + 1);
109  z = dEZ + (50. / 32.) * sin(detTheta) * (2 * verticalstrip + 1);
110  } else if(detector == 1 && pos == 'E') {
111  // Right E
112  x = EX - (50. / 32.) * cos(detTheta) * (2 * verticalstrip + 1);
113  z = EZ + (50. / 32.) * sin(detTheta) * (2 * verticalstrip + 1);
114  } else if(detector == 2 && pos == 'E') {
115  // Left E
116  verticalstrip = 15 - verticalstrip;
117  x = -EX + (50. / 32.) * cos(detTheta) * (2 * verticalstrip + 1);
118  z = EZ + (50. / 32.) * sin(detTheta) * (2 * verticalstrip + 1);
119  } else {
120  printf("***Error, unrecognized detector and position combo!***\n");
121  }
122 
123  y = (50. / 32.) * (2 * horizontalstrip + 1) - (50 / 16.) * 8;
124  Pos.SetX(x + X);
125  Pos.SetY(y + Y);
126  Pos.SetZ(z + Z);
127 
128  return Pos;
129 }

131 void TCSM::BuildVH(std::vector<std::vector<std::pair<TFragment, TGRSIMnemonic>>>& strips, std::vector<TDetectorHit*>& hitVector)
132 {
133  /// Build hits from horizontal (index = 0) and vertical (index = 1) strips into the hitVector
134  if(strips[0].empty() && strips[1].empty()) {
135  return;
136  }
137  if(strips[0].size() == 1 && strips[1].empty()) {
138  RecoverHit('H', strips[0][0], hitVector);
139  } else if(strips[0].empty() && strips[1].size() == 1) {
140  RecoverHit('V', strips[1][0], hitVector);
141  } else if(strips[0].size() == 1 && strips[1].size() == 1) {
142  hitVector.push_back(MakeHit(strips[0][0], strips[1][0]));
143  } else if(strips[1].size() == 1 && strips[0].size() == 2) {
144  int he1 = strips[0][0].first.GetEnergy();
145  int he2 = strips[0][1].first.GetEnergy();
146  int ve1 = strips[1][0].first.GetEnergy();
147  if(AlmostEqual(ve1, he1 + he2) && SUMHITS) {
148  hitVector.push_back(MakeHit(strips[0], strips[1]));
149  } else if(AlmostEqual(ve1, he1)) {
150  hitVector.push_back(MakeHit(strips[0][0], strips[1][0]));
151  RecoverHit('H', strips[0][1], hitVector);
152  } else if(AlmostEqual(ve1, he2)) {
153  hitVector.push_back(MakeHit(strips[0][1], strips[1][0]));
154  RecoverHit('H', strips[0][0], hitVector);
155  }
156  } else if(strips[1].size() == 2 && strips[0].size() == 1) {
157  int he1 = strips[0][0].first.GetEnergy();
158  int ve1 = strips[1][0].first.GetEnergy();
159  int ve2 = strips[1][1].first.GetEnergy();
160  if(AlmostEqual(ve1 + ve2, he1) && SUMHITS) {
161  hitVector.push_back(MakeHit(strips[0], strips[1]));
162  } else if(AlmostEqual(ve1, he1)) {
163  hitVector.push_back(MakeHit(strips[0][0], strips[1][0]));
164  RecoverHit('V', strips[1][1], hitVector);
165  } else if(AlmostEqual(ve2, he1)) {
166  hitVector.push_back(MakeHit(strips[0][0], strips[1][1]));
167  RecoverHit('V', strips[1][0], hitVector);
168  }
169  } else if(strips[1].size() == 2 && strips[0].size() == 2) {
170  int he1 = strips[0][0].first.GetEnergy();
171  int he2 = strips[0][1].first.GetEnergy();
172  int ve1 = strips[1][0].first.GetEnergy();
173  int ve2 = strips[1][1].first.GetEnergy();
174  if((AlmostEqual(ve1, he1) && AlmostEqual(ve2, he2)) || (AlmostEqual(ve1, he2) && AlmostEqual(ve2, he1))) {
175  // I can build both 1,1 and 2,2 or 1,2 and 2,1
176  if(std::abs(ve1 - he1) + std::abs(ve2 - he2) <= std::abs(ve1 - he2) + std::abs(ve2 - he1)) {
177  // 1,1 and 2,2 mimimizes difference
178  hitVector.push_back(MakeHit(strips[0][0], strips[1][0]));
179  hitVector.push_back(MakeHit(strips[0][1], strips[1][1]));
180  } else if(std::abs(ve1 - he1) + std::abs(ve2 - he2) > std::abs(ve1 - he2) + std::abs(ve2 - he1)) {
181  // 1,2 and 2,1 mimimizes difference
182  hitVector.push_back(MakeHit(strips[0][0], strips[1][1]));
183  hitVector.push_back(MakeHit(strips[0][1], strips[1][0]));
184  }
185  } else if(AlmostEqual(ve1, he1)) {
186  hitVector.push_back(MakeHit(strips[0][0], strips[1][0]));
187  } else if(AlmostEqual(ve2, he1)) {
188  hitVector.push_back(MakeHit(strips[0][1], strips[1][0]));
189  } else if(AlmostEqual(ve1, he2)) {
190  hitVector.push_back(MakeHit(strips[0][0], strips[1][1]));
191  } else if(AlmostEqual(ve2, he2)) {
192  hitVector.push_back(MakeHit(strips[0][1], strips[1][1]));
193  }
194  }

267  std::cerr<<"\tSomething is wrong, Horizontal detector positions don't match in vector loop."<<std::endl;

270  EnergyH += hhV[i].first.GetEnergy();
271  }

477  EUsed.at(eiter) = true;
478  break;
479  }
480  }
481  }
482  }
483 
484  // This loop adds uncorrelated events in the telescope together. This may be bad, but let's see.
485  for(size_t i = 0; i < DHitVec.size(); i++) {
486  if(!DUsed.at(i)) {
487  for(size_t j = 0; j < EHitVec.size(); j++) {

492  EUsed.at(j) = true;
493  break;
494  }
495  }
496  }
497  }
498  }

500  // Send through the stragglers. This is very permissive, but we trust BuildVH to take care of the riff-raff
501  for(size_t i = 0; i < DHitVec.size(); i++) {
502  if(!DUsed.at(i)) {
503  BuiltHits.push_back(DHitVec.at(i));
504  }
505  }
506  for(size_t j = 0; j < EHitVec.size(); j++) {
507  if(!EUsed.at(j)) {
508  BuiltHits.push_back(EHitVec.at(j));
509  }
510  }
511 
512  if(printbit) {
513  for(auto& BuiltHit : BuiltHits) {
514  std::cout<<DGREEN;
515  BuiltHit->Print();
516  std::cout<<RESET_COLOR<<std::endl;
517  }
518  }
