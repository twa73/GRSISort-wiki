133 // How the make addback currently works
134 //
135 // Proper setting of a "bit" to say addback has been made hasnt been implemented yet.
136 //
137 // First we itterate through all pixel hits
138 // If the hit has not yet been assigned to a "cluster" a new cluster is created containing it
139 // The hit is compared all following hits against fAddbackCriterion
140 // If they are acceptable neighbours the second hit is also added to the cluster
141 //
142 // fAddbackCriterion checks if hits are flat surface contact (no corners) neighbours
143 // and compares time to the global TGRSIRunInfo::AddBackWindow()
144 // some limited energy constraints are also added
145 //
146 // Clusters are then ordered by energy and summed
147 // All base information is taken from the highest energy hit
148 
149 // Things to implement:
150 // -maybe dont use global TGRSIRunInfo::AddBackWindow() time
151 // -better energy restrictions could be placed on what is a pair to suppress crosstalk issues
152 // -instead of blindly summing entire clusters a decision could be made based on geometry of the pixels
153 // eg if an event claims to go through a row of 3 the middle one must have sufficient energy for an electron to have traversed
154 // this would obviously be less in the inner rings and basically 3 sectors in a row in the outer rings is completely absurd
155 // -the cluster energy sorting algorithm should probably be made more efficient
156 //
157 // On thinking about it we should probably have specific rules for clusters of 2 and 3 and if a cluster >3 either discard all events or use all as singles
158 //
159 // Possibly allow straight lines 3s only on the inner rings and then corner 3/4s further out straight lines
160 //
161 // Need to determine where the line is between throwing out bad reconstruction or saying it is just 2 very close singles

163 Int_t TSiLi::GetAddbackMultiplicity() {
164  // Automatically builds the addback hits using the addback_criterion (if the size of the addback_hits vector is zero) and return the number of addback hits.
165  short basehits=fSiLiHits.size();
166 
167  if(basehits == 0) {
168  return 0;
169  }
170 
171  //if the addback has been reset, clear the addback hits
172  if(!(fSiLiBits.TestBit(kAddbackSet))) {
173  fAddbackHits.clear();
174  }
175 
176  if(fAddbackHits.size() == 0) {
177 
178  // Create a matrix of "pairs"
179  std::vector < std::vector<bool> > pairs;
180  for(int i=0;i<basehits;i++)pairs.push_back(std::vector<bool>(basehits,false));
181 
182  std::vector<unsigned> clusters_id(basehits,0);
183  std::vector < std::vector<unsigned> > Clusters;
184 
185 
186  //In this loop we check all hits for pairing
187  for(int i=0;i<basehits;i++){
188  if(fSiLiHits[i].GetEnergy()>10){//We skip over any hits which have been calibrated to zero and some electronic noise
189 
190  if(clusters_id[i]<1){// If not yet paired start a new cluster
191  std::vector<unsigned> newCluster(1,i);
192  Clusters.push_back(newCluster);
193  clusters_id[i]=Clusters.size();
194  }
195  unsigned clus_id=clusters_id[i];
196 
197  for(int j=i+1;j<basehits;j++){
198  if(fSiLiHits[j].GetEnergy()>10){
199  //if a pair
200  if(fAddbackCriterion(GetSiLiHit(i),GetSiLiHit(j))){
201  Clusters[clus_id-1].push_back(j);
202  clusters_id[j]=clus_id;
203  }
204  }
205  }
206  }
207  }
208 
209  //Clusters are lists of hits that have been identified as coincident neighbours
210  //Will be length 1 if no neighbours
211 
212  for(unsigned i=0;i<Clusters.size();i++){
213 
214  TSiLi::SortCluster(Clusters[i]);//Energy sort the clusters also deletes any invalid hit numbers//Not an efficient function
215  if(Clusters[i].size()>0){
216  uint s=fAddbackHits.size();
217  //We have to add it and THEN do the SumHit because the push_back copies the charge but not the energy, which is the bit we sum
218  //This is desired behaviour of TGRSIDetectorHit for speed of sorts, but messy for the addback, which should only be done "on the fly" not stored to TSiLi on disk
219  fAddbackHits.push_back(TSiLiHit());
220  for(unsigned j=0;j<Clusters[i].size();j++){
221  fAddbackHits[s].SumHit(GetSiLiHit(Clusters[i][j]));
222  }
223  }
224  }
225 
226  fSiLiBits.SetBit(kAddbackSet, true);
227  }
228 
229  return fAddbackHits.size();
230 }

266 void TSiLi::SortCluster(std::vector<unsigned>& cluster){
267  unsigned cs=cluster.size();
268  if(cs>1){
269  std::vector<double> energy;
270  double Emax=0;
271  int Imax=-1;
272 
273  //Go through them all the referenced hits, get the energies and which one is the max
274  for(unsigned i=0;i<cs;i++){
275  if(cluster[i]<fSiLiHits.size()){
276  double e=fSiLiHits[cluster[i]].GetEnergy();
277  energy.push_back(e);
278  if(e>Emax){Emax=e;Imax=cluster[i];}
279  }else{energy.push_back(0);}
280  }
281 
282  std::vector<unsigned> ordered;
283 
284  //Note if the cluster contained no valid hits Imax=-1 still and the loop will be skipped
285  //cluster will effectively be deleted by the copy of blank order
286 
287  while(Imax>=0){
288  ordered.push_back(Imax);//At start of loop because we want to do the previously determind max here
289  if(cs==ordered.size())break;
290  Imax=-1;//Will stay -1 unless there is a next highest
291  double Emaxloop=0;
292  for(unsigned i=0;i<cs;i++){
293  if(energy[i]>Emaxloop&&energy[i]<Emax){Imax=cluster[i];Emaxloop=energy[i];};
294  }
295  Emax=Emaxloop;
296  }
297 
298  cluster=ordered;
299  }
300  if(!(cluster[0]<fSiLiHits.size()))cluster.clear();
