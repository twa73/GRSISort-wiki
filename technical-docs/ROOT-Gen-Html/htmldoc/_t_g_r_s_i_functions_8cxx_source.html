277 Double_t TGRSIFunctions::Bateman(std::vector<Double_t>& dim, std::vector<Double_t>& par, UInt_t nChain, Double_t)
278 {
279  //****NOT TESTED****The Bateman equation is the general closed form for a decay chain of nuclei. This functions
280  // returns
281  // the total activity from a given chain of nuclei.
282  // Requires the following parameters:
283  // - dim[0]: channels being fit
284  // - par[0+3*i]: Initial Activity s^(-1)
285  // - par[1+3*i]: Decay Rate in seconds^(-1)
286  // - par[2+3*i]: Activity at time t s^(-1)
287  // NOTE: The lowest paramters correspond to the most 'senior' nuclei
288 
289  if(par.size() < (nChain * 3)) {
290  std::cout<<"not enough parameters passed to function"<<std::endl;
291  return 0;
292  }
293 
294  Double_t totalActivity = 0.0;
295 
296  // LOOP OVER ALL NUCLEI
297 
298  for(UInt_t n = 0; n < nChain; n++) {
299  // Calculate this equation for the nth nucleus.
300  Double_t firstterm = 1.0;
301  // Compute the first multiplication
302  for(UInt_t j = 0; j < n - 1; j++) {
303  firstterm *= par[1 + 3 * j];
304  }
305  Double_t secondterm = 0.0;
306  for(UInt_t i = 0; i < n; i++) {
307  Double_t sum = 0.0;
308  for(UInt_t j = i; j < n; j++) {
309  Double_t denom = 1.0;
310  for(UInt_t p = i; p < n; p++) {
311  if(p != j) {
312  denom *= par[1 + 3 * p] - par[1 + 3 * j];
313  }
314  }
315  sum += par[0 + 3 * i] / par[1 + 3 * i] * TMath::Exp(-par[1 + 3 * j] * dim[0]) / denom;
316  }
317  secondterm += sum;
318  }
319  par[2 + 3 * n] = par[1 + 3 * n] * firstterm * secondterm;
320  totalActivity += par[2 + 3 * n];
321  }
322  return totalActivity;
323 }
