15 bool TGRSIFunctions::CheckParameterErrors(const TFitResultPtr& fitres, std::string opt)
16 {
17  /// This function compares the parameter error with the square root of the corresponding
18  /// diagonal entry of the covariance matrix. If the difference between the two is larger
19  /// than 0.1 (arbitrarily chosen cutoff), the check fails.
20  /// Implemented options are "q" to be quiet (prints nothing), or "v" to be verbose
21  /// (prints message not just for failed parameters but also ones that pass the test).
22 
23  // if we do not have a fit result, we have to assume everything is okay for now
24  if(fitres.Get() == nullptr) return true;
25  std::transform(opt.begin(), opt.end(), opt.begin(), [](unsigned char c){ return std::tolower(c); });
26  bool quiet = (opt.find('q') != std::string::npos);
27  bool verbose = (opt.find('v') != std::string::npos);
28  if(quiet && verbose) {
29  std::cout<<"Don't know how to be quiet and verbose at once ("<<opt<<"), going to be verbose!"<<std::endl;
30  quiet = false;
31  }
32  bool result = true;
33  // loop over all parameters and compare the parameter error with the square root
34  // of the diagonal element of the covariance matrix
35  auto covarianceMatrix = fitres->GetCovarianceMatrix();
36  // if we fail to get a covariance matrix we assume there was a problem in the fit
37  if(covarianceMatrix.GetNrows() != static_cast<Int_t>(fitres->NPar()) || covarianceMatrix.GetNcols() != static_cast<Int_t>(fitres->NPar())) return false;
38  for(unsigned int i = 0; i < fitres->NPar(); ++i) {
39  if(std::fabs(fitres->ParError(i) - TMath::Sqrt(covarianceMatrix[i][i])) > 0.1) {
40  if(!quiet) {
41  std::cout<<RED<<"Parameter "<<i<<" = "<<fitres->GetParameterName(i)<<" is at or close to the limit, error is "<<fitres->ParError(i)<<", and square root of diagonal is "<<TMath::Sqrt(covarianceMatrix[i][i])<<RESET_COLOR<<std::endl;
42  }
43  result = false;
44  // we don't break here even though we could, because we want to print out
45  // all parameters with issues
46  } else if(verbose) {
47  std::cout<<GREEN<<"Parameter "<<i<<" = "<<fitres->GetParameterName(i)<<" is not close to the limit, error is "<<fitres->ParError(i)<<", and square root of diagonal is "<<TMath::Sqrt(covarianceMatrix[i][i])<<RESET_COLOR<<std::endl;
48  }
49  }
50  return result;
51 }

57  /*
58  p[0]-p[4] are t0, tRC, tF, TS, TGamma

66  }

71  return s;

317 Double_t TGRSIFunctions::Bateman(std::vector<Double_t>& dim, std::vector<Double_t>& par, UInt_t nChain, Double_t)
318 {
319  //****NOT TESTED****The Bateman equation is the general closed form for a decay chain of nuclei. This functions
320  // returns
321  // the total activity from a given chain of nuclei.
322  // Requires the following parameters:
323  // - dim[0]: channels being fit
324  // - par[0+3*i]: Initial Activity s^(-1)
325  // - par[1+3*i]: Decay Rate in seconds^(-1)
326  // - par[2+3*i]: Activity at time t s^(-1)
327  // NOTE: The lowest paramters correspond to the most 'senior' nuclei
328 
329  if(par.size() < (nChain * 3)) {
330  std::cout<<"not enough parameters passed to function"<<std::endl;
331  return 0;
332  }
333 
334  Double_t totalActivity = 0.0;
335 
336  // LOOP OVER ALL NUCLEI
337 
338  for(UInt_t n = 0; n < nChain; n++) {
339  // Calculate this equation for the nth nucleus.
340  Double_t firstterm = 1.0;
341  // Compute the first multiplication
342  for(UInt_t j = 0; j < n - 1; j++) {
343  firstterm *= par[1 + 3 * j];
344  }
345  Double_t secondterm = 0.0;
346  for(UInt_t i = 0; i < n; i++) {
347  Double_t sum = 0.0;
348  for(UInt_t j = i; j < n; j++) {
349  Double_t denom = 1.0;
350  for(UInt_t p = i; p < n; p++) {
351  if(p != j) {
352  denom *= par[1 + 3 * p] - par[1 + 3 * j];
353  }
354  }
355  sum += par[0 + 3 * i] / par[1 + 3 * i] * TMath::Exp(-par[1 + 3 * j] * dim[0]) / denom;
356  }
357  secondterm += sum;
358  }
359  par[2 + 3 * n] = par[1 + 3 * n] * firstterm * secondterm;
360  totalActivity += par[2 + 3 * n];
361  }
362  return totalActivity;
363 }
