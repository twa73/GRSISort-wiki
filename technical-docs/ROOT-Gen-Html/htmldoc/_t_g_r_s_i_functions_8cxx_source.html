264 //****NOT TESTED****The Bateman equation is the general closed form for a decay chain of nuclei. This functions returns
265 //the total activity from a given chain of nuclei.
266 //Requires the following parameters:
267 // - dim[0]: channels being fit
268 // - par[0+3*i]: Initial Activity s^(-1)
269 // - par[1+3*i]: Decay Rate in seconds^(-1)
270 // - par[2+3*i]: Activity at time t s^(-1)
271 // NOTE: The lowest paramters correspond to the most 'senior' nuclei
272

274  if(sizeof(par)/sizeof(par[0]) < (nChain*3)){
275  std::cout << "not enough parameters passed to function" << std::endl;
276  return 0;
277  }
278 
279  Double_t totalActivity = 0.0;
280 
281 //LOOP OVER ALL NUCLEI
282 
283  for(UInt_t n=0; n<nChain;n++){
284  //Calculate this equation for the nth nucleus.
285  Double_t firstterm = 1.0;
286  //Compute the first multiplication
287  for(UInt_t j=0; j<n-1; j++){
288  firstterm*=par[1+3*j];
289  }
290  Double_t secondterm = 0.0;
291  for(UInt_t i=0; i<n; i++){
292  Double_t sum = 0.0;
293  for(UInt_t j=i; j<n; j++){
294  Double_t denom = 1.0;
295  for(UInt_t p=i;p<n;p++){
296  if(p!=j){ denom*=par[1+3*p]-par[1+3*j]; }
297  }
298  sum+=par[0+3*i]/par[1+3*i]*TMath::Exp(-par[1+3*j]*dim[0])/denom;
299  }
300  secondterm += sum;
301  }
302  par[2+3*n] = par[1+3*n]*firstterm*secondterm;
303  totalActivity += par[2+3*n];
