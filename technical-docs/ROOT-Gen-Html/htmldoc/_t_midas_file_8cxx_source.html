100 /// - ssh://username\@hostname/path/file.mid.gz and file.mid.bz2 - same for compressed files

834  printf(BG_WHITE DRED "problem parsing odb data, arrays are different sizes, channels not set." RESET_COLOR "\n");
835  }

838  // "/Experiment/Edit on start/PPG Cycle" is a link to the PPG cycle used (always "/PPG/Current"???)

876  return;
