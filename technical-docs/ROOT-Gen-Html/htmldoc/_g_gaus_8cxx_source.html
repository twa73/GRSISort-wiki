361  res[points] = (hist->GetBinContent(i) - Eval(hist->GetBinCenter(i))) + GetParameter("Height") / 2;
362  bin[points] = hist->GetBinCenter(i);
363  points++;
364  }
365  new GCanvas();
366  auto* residuals = new TGraph(points, bin, res);
367  residuals->Draw("*AC");
