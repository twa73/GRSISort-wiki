203 TH1* scaled_back_subtract(TH1* gate,TH1* back ,double backfrack,TH1* pass,double uncertainfrac){
204  double backcount=back->Integral();
205  double forecount=gate->Integral();
206 
207  if(!back->GetSumw2N())back->Sumw2();
208  if(!gate->GetSumw2N())gate->Sumw2();
209 
210  double subtraction=-(backfrack*forecount)/backcount;
211  pass->Add(gate,back,1.0,subtraction);
212 
213  if(uncertainfrac>0){static_cast< TH1ErrorAdj* > (pass)->AdjustError(back,subtraction*uncertainfrac);}
214  return pass;
