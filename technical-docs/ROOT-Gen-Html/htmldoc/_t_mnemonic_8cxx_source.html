57 void TMnemonic::Parse(std::string* name)
58 {
59  if((name == nullptr) || name->length() < 9) {
60  if((name->length() < 1) && (name->compare(0, 2, "RF") == 0)) {
61  SetRFMNEMONIC(name);
62  }
63  return;
64  }
65  std::string buf;
66  fSystemString.assign(*name, 0, 2);
67  fSubSystemString.assign(*name, 2, 1);
68  EnumerateMnemonic(fSubSystemString, fSubSystem);
69  buf.clear();
70  buf.assign(*name, 3, 2);
71  fArrayPosition = static_cast<uint16_t>(atoi(buf.c_str()));
72 // TIP is a Bad Mnemonic and uses 3 characters for array position this may be changed in the future - S. Gillespie
73  if(fSystemString.compare("TP") == 0){
74  fArraySubPositionString.assign(*name, 5, 2);
75  }
76  else {
77  fArraySubPositionString.assign(*name, 5, 1);
