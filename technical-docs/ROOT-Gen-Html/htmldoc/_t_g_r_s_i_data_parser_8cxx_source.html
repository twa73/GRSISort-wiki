201 
202  break;

504 int TGRSIDataParser::GriffinDataToFragment(uint32_t* data, int size, EBank bank, unsigned int midasSerialNumber,
505  time_t midasTime)
506 {
507  /// Converts a Griffin flavoured MIDAS file into a TFragment and returns the number of words processed (or the
508  /// negative index of the word it failed on)
509  std::shared_ptr<TFragment> eventFrag = std::make_shared<TFragment>();
510  // no need to delete eventFrag, it's a shared_ptr and gets deleted when it goes out of scope
511  fFragmentHasWaveform = false;
512  fState = EDataParserState::kGood;
513  int failedWord = -1; // Variable stores which word we failed on (if we fail). This is somewhat duplicate information
514  // to fState, but makes things easier to track.
515  bool multipleErrors = false; // Variable to store if multiple errors occured parsing one fragment
516 
517  if(fOptions == nullptr) {
518  fOptions = TGRSIOptions::Get();
519  }
520 
521  int x = 0;
522  if(!SetGRIFHeader(data[x++], eventFrag, bank)) {
523  printf(DYELLOW "data[0] = 0x%08x" RESET_COLOR "\n", data[0]);
524  // we failed to get a good header, so we don't know which detector type this fragment would've belonged to
525  TParsingDiagnostics::Get()->BadFragment(-1);
526  // this is the first word, so no need to check is the state/failed word has been set before
527  fState = EDataParserState::kBadHeader;
528  failedWord = 0;
529  }
530 
531  eventFrag->SetDaqTimeStamp(midasTime);
532  eventFrag->SetDaqId(midasSerialNumber);
533 
534  // Changed on 11 Aug 2015 by RD to include PPG events. If the event has ModuleType 4 and address 0xffff, it is a PPG
535  // event.
536  if((eventFrag->GetModuleType() == 1 || eventFrag->GetModuleType() == 4) && eventFrag->GetAddress() == 0xffff) {
537  return GriffinDataToPPGEvent(data, size, midasSerialNumber, midasTime);
538  }
539  // If the event has detector type 15 (0xf) it's a scaler event.
540  if(eventFrag->GetDetectorType() == 0xf) {
541  // a scaler event (trigger or deadtime) has 8 words (including header and trailer), make sure we have at least
542  // that much left
543  if(size < 8) {
544  if(fState == EDataParserState::kGood) {
545  fState = EDataParserState::kMissingWords;
546  failedWord = x;
547  } else {
548  multipleErrors = true;
549  }
550  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
551  }
552  return GriffinDataToScalerEvent(data, eventFrag->GetAddress());
553  }
554 
555  // The Network packet number is for debugging and is not always written to the midas file.
556  if(SetGRIFNetworkPacket(data[x], eventFrag)) {
557  ++x;
558  }
559 
560  // The Primary Filter Pattern is in an unstable state right now and is not
561  // always written to the midas file
562  if(SetGRIFPrimaryFilterPattern(data[x], eventFrag, bank)) {
563  ++x;
564  }
565 
566  // We can get multiple filter ids (one fragment can pass multiple filter conditions)
567  // so we have to loop until we don't find one
568  while(SetGRIFPrimaryFilterId(data[x], eventFrag)) {
569  ++x;
570  }
571 
572  // The channel trigger ID is in an unstable state right now and is not
573  // always written to the midas file
574  if(!SetGRIFChannelTriggerId(data[x++], eventFrag)) {
575  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
576  if(fState == EDataParserState::kGood) {
577  fState = EDataParserState::kBadTriggerId;
578  failedWord = x - 1; // -1 compensates the incrementation in the if-statement
579  } else {
580  multipleErrors = true;
581  }
582  }
583 
584  // The Network packet number is for debugging and is not always written to
585  // the midas file.
586  if(SetGRIFNetworkPacket(data[x], eventFrag)) {
587  x++;
588  }
589 
590  if(!SetGRIFTimeStampLow(data[x++], eventFrag)) {
591  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
592  if(fState == EDataParserState::kGood) {
593  fState = EDataParserState::kBadLowTS;
594  failedWord = x - 1; // -1 compensates the incrementation in the if-statement
595  } else {
596  multipleErrors = true;
597  }
598  }
599 
600  if(!SetGRIFDeadTime(data[x++], eventFrag)) {
601  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
602  if(fState == EDataParserState::kGood) {
603  fState = EDataParserState::kBadHighTS;
604  failedWord = x - 1; // -1 compensates the incrementation in the if-statement
605  } else {
606  multipleErrors = true;
607  }
608  }
609 
610  std::vector<Int_t> tmpCharge;
611  std::vector<Short_t> tmpIntLength;
612  std::vector<Int_t> tmpCfd;
613 
614  for(; x < size; x++) {
615  uint32_t dword = data[x];
616  uint32_t packet = dword >> 28;
617  uint32_t value = dword & 0x0fffffff;
618 
619  switch(packet) {
620  case 0x8: // The 8 packet type is for event headers
621  // if this happens, we have "accidentally" found another event.
622  // currently the GRIF-C only sets the primary/secondary port of the address for the first header (of the corrupt
623  // event)
624  // so we want to ignore this corrupt event and the next event which has a wrong address
625  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
626  if(fState == EDataParserState::kGood) {
627  fState = EDataParserState::kSecondHeader;
628  failedWord = x + 1; //+1 to ensure we don't read this header as start of a good event
629  } else {
630  multipleErrors = true;
631  }
632  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
633  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
634  break;
635  case 0xc: // The c packet type is for waveforms
636  if(!fNoWaveforms) {
637  SetGRIFWaveForm(value, eventFrag);
638  }
639  break;
640  case 0xd:
641  SetGRIFNetworkPacket(dword, eventFrag); // The network packet placement is not yet stable.
642  break;
643  case 0xe:
644  // changed on 21 Apr 2015 by JKS, when signal processing code from Chris changed the trailer.
645  // change should be backward-compatible
646  if((value & 0x3fff) == (eventFrag->GetChannelId() & 0x3fff)) {
647  if(!fOptions->SuppressErrors() && (eventFrag->GetModuleType() == 2) && (bank < EBank::kGRF3)) {
648  // check whether the nios finished and if so whether it finished with an error
649  if(((value >> 14) & 0x1) == 0x1) {
650  if(((value >> 16) & 0xff) != 0) {
651  printf(BLUE "0x%04x: NIOS code finished with error 0x%02x" RESET_COLOR "\n",
652  eventFrag->GetAddress(), (value >> 16) & 0xff);
653  }
654  }
655  }
656 
657  if((eventFrag->GetModuleType() == 1) || (bank > EBank::kGRF2)) { // 4Gs have this only for banks newer than GRF2
658  eventFrag->SetAcceptedChannelId((value >> 14) & 0x3fff);
659  } else {
660  eventFrag->SetAcceptedChannelId(0);
661  }
662 
663  // check the number of words the header said we should have with the number of words we've read
664  // the number of words is only set for bank >= GRF3
665  // if the fragment has a waveform, we can't compare the number of words
666  // the headers number of words includes the header (word 0) itself, so we need to compare to x plus one
667  if(fOptions->WordOffset() >= 0 && eventFrag->GetNumberOfWords() > 0 && !eventFrag->HasWave() && eventFrag->GetNumberOfWords() != x + fOptions->WordOffset()) {
668  if(fState == EDataParserState::kGood) {
669  fState = EDataParserState::kWrongNofWords;
670  failedWord = x;
671  } else {
672  multipleErrors = true;
673  }
674  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
675  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
676  }
677 
678  // the way we insert the fragment(s) depends on the module type and bank:
679  // 1 - for module type 1 & bank GRF4, we can't insert the fragments yet, we need to put them in a separate queue
680  // 2 - for module type 2 (4G, all banks) and module type 1 & bank GRF3 we set the single charge, cfd, and
681  // IntLength, and insert the fragment
682  // 3 - for module type 1 & banks GRF1/GRF2 we loop over the charge, cfd, and IntLengths, and insert the
683  // (multiple) fragment(s)
684  // the last two cases can be treated the same since the second case will just have a single length charge,
685  // cfd, and IntLengths
686 
687  // so we only need to check for the first case
688  if(eventFrag->GetModuleType() == 1 && bank == EBank::kGRF4) {
689  if(tmpCfd.size() != 1) {
690  if(fRecordDiag) {
691  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
692  }
693  if(fState == EDataParserState::kGood) {
694  fState = EDataParserState::kNotSingleCfd;
695  failedWord = x;
696  } else {
697  multipleErrors = true;
698  }
699  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
700  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
701  }
702  eventFrag->SetCfd(tmpCfd[0]);
703  if(fRecordDiag) {
704  TParsingDiagnostics::Get()->GoodFragment(eventFrag);
705  }
706  fFragmentMap.Add(eventFrag, tmpCharge, tmpIntLength);
707  return x;
708  }
709  if(tmpCharge.size() != tmpIntLength.size() || tmpCharge.size() != tmpCfd.size()) {
710  if(fRecordDiag) {
711  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
712  }
713  if(fState == EDataParserState::kGood) {
714  fState = EDataParserState::kSizeMismatch;
715  failedWord = x;
716  } else {
717  multipleErrors = true;
718  }
719  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
720  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
721  }
722  for(size_t h = 0; h < tmpCharge.size(); ++h) {
723  eventFrag->SetCharge(tmpCharge[h]);
724  eventFrag->SetKValue(tmpIntLength[h]);
725  eventFrag->SetCfd(tmpCfd[h]);
726  if(fRecordDiag) {
727  TParsingDiagnostics::Get()->GoodFragment(eventFrag);
728  }
729  if(fState == EDataParserState::kGood) {
730  if(fOptions->ReconstructTimeStamp()) {
731  fLastTimeStampMap[eventFrag->GetAddress()] = eventFrag->GetTimeStamp();
732  }
733  Push(fGoodOutputQueues, std::make_shared<TFragment>(*eventFrag));
734  } else {
735  if(fOptions->ReconstructTimeStamp() && fState == EDataParserState::kBadHighTS && !multipleErrors) {
736  // std::cout<<"reconstructing timestamp from 0x"<<std::hex<<eventFrag->GetTimeStamp()<<" using
737  // 0x"<<fLastTimeStampMap[eventFrag->GetAddress()];
738  // reconstruct the high bits of the timestamp from the high bits of the last time stamp of the
739  // same address after converting the saved timestamp back to 10 ns units
740  if((eventFrag->GetTimeStamp() & 0x0fffffff) <
741  (fLastTimeStampMap[eventFrag->GetAddress()] & 0x0fffffff)) {
742  // we had a wrap-around of the low time stamp, so we need to set the high bits to the old
743  // high bits plus one
744  eventFrag->AppendTimeStamp(((fLastTimeStampMap[eventFrag->GetAddress()] >> 28) + 1)<<28);
745  } else {
746  eventFrag->AppendTimeStamp(fLastTimeStampMap[eventFrag->GetAddress()] & 0x3fff0000000);
747  }
748  // std::cout<<" => 0x"<<eventFrag->GetTimeStamp()<<std::dec<<std::endl;
749  Push(fGoodOutputQueues, std::make_shared<TFragment>(*eventFrag));
750  } else {
751  // std::cout<<"Can't reconstruct time stamp, "<<fOptions->ReconstructTimeStamp()<<",
752  // state "<<fState<<" = "<<EDataParserState::kBadHighTS<<", "<<multipleErrors<<std::endl;
753  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
754  }
755  }
756  }
757  return x;
758  } else {
759  if(fRecordDiag) {
760  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
761  }
762  if(fState == EDataParserState::kGood) {
763  fState = EDataParserState::kBadFooter;
764  failedWord = x;
765  } else {
766  multipleErrors = true;
767  }
768  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
769  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
770  }
771  break;
772  case 0xf:
773  switch(bank) {
774  case EBank::kGRF1: // format from before May 2015 experiments
775  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
776  if(fState == EDataParserState::kGood) {
777  fState = EDataParserState::kFault;
778  failedWord = x;
779  } else {
780  multipleErrors = true;
781  }
782  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
783  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
784  break;
785  case EBank::kGRF2: // from May 2015 to the end of 2015 0xf denoted a psd-word from a 4G
786  if(x + 1 < size) {

788  ++x;
789  dword = data[x];

791  } else {

794  fState = EDataParserState::kMissingPsd;
795  failedWord = x;
796  } else {
797  multipleErrors = true;
798  }
799  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
800  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
801  }
802  break;
803  case EBank::kGRF3: // from 2016 on we're back to reserving 0xf for faults
804  case EBank::kGRF4:
805  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
806  if(fState == EDataParserState::kGood) {
807  fState = EDataParserState::kFault;
808  failedWord = x;
809  } else {
810  multipleErrors = true;
811  }
812  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
813  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
814  break;
815  default: printf("This bank not yet defined.\n"); break;
816  }
817  break;
818 
819  default:
820  // these are charge/cfd words which are different depending on module type, and bank number/detector type
821  switch(eventFrag->GetModuleType()) {
822  case 1:
823  switch(bank) { // the GRIF-16 data format depends on the bank number
824  case EBank::kGRF1: // bank's 1&2 have n*2 words with (5 high bits IntLength, 26 Charge)(5 low bits IntLength, 26
825  // Cfd)
826  case EBank::kGRF2:
827  // read this pair of charge/cfd words, check if the next word is also a charge/cfd word
828  if(((data[x] & 0x80000000) == 0x0) && x + 1 < size && (data[x + 1] & 0x80000000) == 0x0) {
829  Short_t tmp = (data[x] & 0x7c000000) >> 21; // 21 = 26 minus space for 5 low bits
830  tmpCharge.push_back((data[x] & 0x03ffffff) | (((data[x] & 0x02000000) == 0x02000000)
831  ? 0xfc000000
832  : 0x0)); // extend the sign bit of 26bit charge word
833  ++x;
834  tmpIntLength.push_back(tmp | ((data[x] & 0x7c000000) >> 26));
835  tmpCfd.push_back(data[x] & 0x03ffffff);
836  } else {
837  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
838  if(fState == EDataParserState::kGood) {
839  fState = EDataParserState::kMissingCfd;
840  failedWord = x;
841  } else {
842  multipleErrors = true;
843  }
844  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
845  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
846  }
847  break;
848  case EBank::kGRF3: // bank 3 has 2 words with (5 high bits IntLength, 26 Charge)(9 low bits IntLength, 22 Cfd)
849  if(x + 1 < size &&
850  (data[x + 1] & 0x80000000) == 0x0) { // check if the next word is also a charge/cfd word
851  Short_t tmp = (data[x] & 0x7c000000) >> 17; // 17 = 26 minus space for 9 low bits
852  tmpCharge.push_back((data[x] & 0x03ffffff) | (((data[x] & 0x02000000) == 0x02000000)
853  ? 0xfc000000
854  : 0x0)); // extend the sign bit of 26bit charge word
855  ++x;
856  tmpIntLength.push_back(tmp | ((data[x] & 0x7fc00000) >> 22));
857  tmpCfd.push_back(data[x] & 0x003fffff);
858  break;
859  } else {
860  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
861  if(fState == EDataParserState::kGood) {
862  fState = EDataParserState::kMissingCfd;
863  failedWord = x;
864  } else {
865  multipleErrors = true;
866  }
867  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
868  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
869  }
870  break;
871  case EBank::kGRF4: // bank 4 can have more than one integration (up to four), but these have to be combined with
872  // other fragments/hits!
873  // we always have 2 words with (5 high bits IntLength, 26 Charge)(9 low bits IntLength, 22 Cfd)
874  if((data[x] & 0x80000000) == 0x0 && x + 1 < size &&
875  (data[x + 1] & 0x80000000) == 0x0) { // check if the next word is also a charge/cfd word
876  Short_t tmp = ((data[x] & 0x7c000000) >> 17) |
877  (((data[x] & 0x40000000) == 0x40000000) ? 0xc000 : 0x0); // 17 = 26 minus space for 9
878  // low bits; signed, so we
879  // extend the sign bit from 14
880  // (31) to 16 bits
881  if(false && (data[x] & 0x02000000) == 0x02000000) { // overflow bit was set - disabled VB
882  tmpCharge.push_back(std::numeric_limits<int>::max());
883  } else {
884  tmpCharge.push_back((data[x] & 0x01ffffff) |
885  (((data[x] & 0x01000000) == 0x01000000)
886  ? 0xfe000000
887  : 0x0)); // extend the sign bit of 25bit charge word
888  }
889  ++x;
890  tmpIntLength.push_back(tmp | ((data[x] & 0x7fc00000) >> 22));
891  tmpCfd.push_back(data[x] & 0x003fffff);
892  // check if we have two more words (X & XI) with (8 num hits, 2 reserved, 14 IntLength2)(31 Charge2);
893  // x has already been incremented once!
894  if(x + 2 < size && (data[x + 1] & 0x80000000) == 0x0 && (data[x + 2] & 0x80000000) == 0x0) {
895  ++x;
896  tmpIntLength.push_back((data[x] & 0x3fff) | (((data[x] & 0x2000) == 0x2000) ? 0xc000 : 0x0));
897  // eventFrag->SetNumberOfPileups((data[x] >> 16) & 0xff);
898  ++x;
899  if((data[x] & 0x02000000) == 0x02000000) { // overflow bit was set
900  tmpCharge.push_back(std::numeric_limits<int>::max());
901  } else {
902  tmpCharge.push_back((data[x] & 0x01ffffff) |
903  (((data[x] & 0x01000000) == 0x01000000)
904  ? 0xfe000000
905  : 0x0)); // extend the sign bit of 25bit charge word
906  }
907  // check if we have two more words (XI & XIII) with (14 IntLength4, 2 reserved, 14 IntLength3)(31
908  // Charge3); x has already been incremented thrice!
909  if(x + 2 < size && (data[x + 1] & 0x80000000) == 0x0 && (data[x + 2] & 0x80000000) == 0x0) {
910  ++x;
911  tmpIntLength.push_back((data[x] & 0x3fff) | (((data[x] & 0x2000) == 0x2000) ? 0xc000 : 0x0));
912  tmpIntLength.push_back((data[x] >> 16) |
913  (((data[x] & 0x20000000) == 0x20000000) ? 0xc000 : 0x0));
914  ++x;
915  if((data[x] & 0x02000000) == 0x02000000) { // overflow bit was set
916  tmpCharge.push_back(std::numeric_limits<int>::max());
917  } else {
918  tmpCharge.push_back((data[x] & 0x01ffffff) |
919  (((data[x] & 0x01000000) == 0x01000000)
920  ? 0xfe000000
921  : 0x0)); // extend the sign bit of 25bit charge word
922  }
923  // check if we have one final word with (31 Charge4), otherwise remove the last integration
924  // length (IntLength4)
925  if(x + 1 < size && (data[x + 1] & 0x80000000) == 0x0) {
926  ++x;
927  if((data[x] & 0x02000000) == 0x02000000) { // overflow bit was set
928  tmpCharge.push_back(std::numeric_limits<int>::max());
929  } else {
930  tmpCharge.push_back((data[x] & 0x01ffffff) |
931  (((data[x] & 0x01000000) == 0x01000000)
932  ? 0xfe000000
933  : 0x0)); // extend the sign bit of 25bit charge word
934  }
935  } else {
936  tmpIntLength.pop_back();
937  }
938  } else if((data[x + 1] & 0x80000000) == 0x0) { // 5 words
939  ++x;
940  }
941  } else if((data[x + 1] & 0x80000000) == 0x0) { // 3 words
942  ++x;
943  }
944  } else {
945  // these types of corrupt events quite often end without a trailer which leads to the header of the
946  // next event missing the primary/secondary part of the address
947  // so we look for the next trailer and stop there
948  while(x < size && (data[x] & 0xf0000000) != 0xe0000000) {
949  ++x;
950  }
951  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
952  if(fState == EDataParserState::kGood) {
953  fState = EDataParserState::kMissingCharge;
954  failedWord = x;
955  } else {
956  multipleErrors = true;
957  }
958  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
959  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
960  }
961  break;
962  default:
963  if(!fOptions->SuppressErrors()) {
964  printf(DRED "Error, bank type %d not implemented yet" RESET_COLOR "\n", static_cast<std::underlying_type<EBank>::type>(bank));
965  }
966  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
967  if(fState == EDataParserState::kGood) {
968  fState = EDataParserState::kBadBank;
969  failedWord = x;
970  } else {
971  multipleErrors = true;
972  }
973  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
974  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
975  break;
976  }
977  break;
978  case 2:
979  // the 4G data format depends on the detector type, but the first two words are always the same
980  if(x + 1 < size && (data[x + 1] & 0x80000000) == 0x0) { // check if the next word is also a charge/cfd word
981  Short_t tmp = (data[x] & 0x7c000000) >> 21; // 21 = 26 minus space for 5 low bits
982  tmpCharge.push_back((data[x] & 0x03ffffff) | (((data[x] & 0x02000000) == 0x02000000)
983  ? 0xfc000000
984  : 0x0)); // extend the sign bit of 26bit charge word
985  ++x;
986  tmpIntLength.push_back(tmp | ((data[x] & 0x7c000000) >> 26));
987  tmpCfd.push_back(data[x] & 0x03ffffff);
988  } else {
989  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
990  if(fState == EDataParserState::kGood) {
991  fState = EDataParserState::kMissingCfd;
992  failedWord = x;
993  } else {
994  multipleErrors = true;
995  }
996  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
997  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
998  }

1000  // GRF1 never had descant)

1002  eventFrag->GetDetectorType() == 11)) {
1003  ++x;
1004  if(x + 1 < size && (data[x + 1] & 0x80000000) == 0x0) {
1005  SetGRIFCc(value, eventFrag);
1006  ++x;
1007  dword = data[x];
1008  SetGRIFPsd(dword, eventFrag);
1009  } else {
1010  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
1011  if(fState == EDataParserState::kGood) {
1012  fState = EDataParserState::kMissingPsd;
1013  failedWord = x;
1014  } else {
1015  multipleErrors = true;
1016  }
1017  Push(*fBadOutputQueue, std::make_shared<TBadFragment>(*eventFrag, data, size, failedWord, multipleErrors));
1018  throw TGRSIDataParserException(fState, failedWord, multipleErrors);
1019  }
1020  }
1021  break;
1022  default:
1023  if(!fOptions->SuppressErrors()) {
1024  printf(DRED "Error, module type %d not implemented yet" RESET_COLOR "\n", eventFrag->GetModuleType());
1025  }
1026  TParsingDiagnostics::Get()->BadFragment(eventFrag->GetDetectorType());
1027  if(fState == EDataParserState::kGood) {
1028  fState = EDataParserState::kBadModuleType;
