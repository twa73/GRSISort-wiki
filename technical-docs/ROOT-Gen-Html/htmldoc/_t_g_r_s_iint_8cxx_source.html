91 void TGRSIint::ApplyOptions()
92 {
93  /// Applies options from TGRSIOptions. This include things such as batch sorting,
94  /// reading material, and logo. Also includes the setup of what to do with mid
95  /// and root files that are input.
96  TGRSIOptions* opt = TGRSIOptions::Get();
97 
98  if(opt->Batch()) {
99  MakeBatch();
100  }
101 
102  bool missing_raw_file = !all_files_exist(opt->InputFiles());
103 
104  if(!false) { // this will be change to something like, if(!ClassicRoot)
105  LoadGROOTGraphics();
106  }
107 
108  if(opt->ReadingMaterial()) {
109  std::thread fnews = std::thread(ReadTheNews);
110  fnews.detach();
111  }
112 
113  ////////////////////////////////////////////////////////
114  ////////////////////////////////////////////////////////
115  ////// Start of redoing section ///////
116  ////////////////////////////////////////////////////////
117  ////////////////////////////////////////////////////////
118 
119  // load parser library if provided
120  if(!opt->ParserLibrary().empty()) {
121  try {
122  TParserLibrary::Get()->Load();
123  } catch(std::runtime_error& e) {
124  // if we failed to load the library, try to continue w/o it
125  std::cerr<<DRED<<e.what()<<RESET_COLOR<<std::endl;
126  }
127  }
128 
129  TRunInfo::ClearVersion();
130  TRunInfo::SetVersion(GRSI_RELEASE);

167 void TGRSIint::LoopUntilDone()
168 {
169  /// Outputs the thread status until all of the threads are complete.
170  int iter = 0;
171  while(StoppableThread::AnyThreadRunning()) {
172  std::this_thread::sleep_for(std::chrono::seconds(1));
173 
174  // We need to process events in case a different thread is asking for a file to be opened.
175  // However, if there is no stdin, ProcessEvents() will call Terminate().
176  // This prevents the terminate from taking effect while in this context.
177  fAllowedToTerminate = false;
178  gSystem->ProcessEvents();
179  fAllowedToTerminate = true;
180 
181  ++iter;
182  if(TGRSIOptions::Get()->StatusInterval() > 0 && iter % TGRSIOptions::Get()->StatusInterval() == 0) {
183  std::cout<<"\r"<<StoppableThread::AllThreadStatus()<<std::endl;
184  }
185  std::cout<<"\r"<<StoppableThread::AllThreadProgress()<<std::flush;
186  }
187  std::cout<<std::endl;
188 }

244 Long_t TGRSIint::ProcessLine(const char* line, Bool_t sync, Int_t* error)
245 {
246  /// This takes over the native root command line. There are two main reasons for this
247  /// 1. To keep the command line thread-safe.
248  /// 2. To block TCanvas from opening, and to instead use our GCanvas.
249 
250  // If you print while fIsTabComplete is true, you will break tab complete.
251  // Any diagnostic print statements should be done after this if statement.
252  if(fIsTabComplete) {
253  long res = TRint::ProcessLine(line, sync, error);
254  return res;
255  }
256 
257  const char* canvas = strstr(line, "TCanvas");
258  if(canvas != nullptr) {
259  const_cast<char*>(canvas)[0] = 'G';
260  }
261 
262  if(std::this_thread::get_id() != main_thread_id) {
263  return DelayedProcessLine(line);
264  }
265 
266  return TRint::ProcessLine(line, sync, error);
267 }

307 TFile* TGRSIint::OpenRootFile(const std::string& filename, Option_t* opt)
308 {
309  /// Opens root files provided on the command line. Also tells you where these files
310  /// are stored (ie _file0). If these files are analysis or fragment trees, they are
311  /// automatically chained into chains called gFragment and gAnalysis. Once this is
312  /// complete, the TChannels, GValues and RunInfo are also read in.
313  TString sopt(opt);
314  sopt.ToLower();
315 
316  TFile* file = nullptr;
317  if(sopt.Contains("recreate") || sopt.Contains("new")) {
318  // We are being asked to make a new file.
319  file = new TFile(filename.c_str(), "RECREATE");
320  if(file != nullptr && file->IsOpen()) {
321  // Give access to the file inside the interpreter.
322  const char* command = Form("TFile* _file%i = (TFile*)%luL;", fRootFilesOpened, (unsigned long)file);
323  TRint::ProcessLine(command);
324  fRootFilesOpened++;
325  } else {
326  std::cout<<"Could not create "<<filename<<std::endl;
327  }
328  } else {
329  // Open an already existing file.
330  file = new TFile(filename.c_str(), opt);
331  if(file != nullptr && file->IsOpen()) {
332  // Give access to the file inside the interpreter.
333  const char* command = Form("TFile* _file%i = (TFile*)%luL;", fRootFilesOpened, (unsigned long)file);
334  TRint::ProcessLine(command);
335  std::cout<<"\tfile "<<BLUE<<file->GetName()<<RESET_COLOR<<" opened as "<<BLUE<<"_file"
336  <<fRootFilesOpened<<RESET_COLOR<<std::endl;
337 
338  // If FragmentTree exists, add the file to the chain.
339  if(file->FindObjectAny("FragmentTree") != nullptr) {
340  if(gFragment == nullptr) {
341  // TODO: Once we have a notifier set up
342  gFragment = new TChain("FragmentChain");
343  // gFragment->SetNotify(GrutNotifier::Get());
344  }
345  printf("file %s added to gFragment.\n", file->GetName());
346 #if MAJOR_ROOT_VERSION < 6
347  gFragment->AddFile(file->GetName(), TChain::kBigNumber, "FragmentTree");
348 #else
349  gFragment->AddFile(file->GetName(), TTree::kMaxEntries, "FragmentTree");
350 #endif
351  }
352 
353  // If AnalysisTree exists, add the file to the chain.
354  if(file->FindObjectAny("AnalysisTree") != nullptr) {
355  if(gAnalysis == nullptr) {
356  gAnalysis = new TChain("AnalysisChain");
357  // TODO: Once we have a notifier set up
358  // gAnalysis->SetNotify(GrutNotifier::Get());
359  }
360  printf("file %s added to gAnalysis.\n", file->GetName());
361 #if MAJOR_ROOT_VERSION < 6
362  gAnalysis->AddFile(file->GetName(), TChain::kBigNumber, "AnalysisTree");
363 #else
364  gAnalysis->AddFile(file->GetName(), TTree::kMaxEntries, "AnalysisTree");
365 #endif
366  }
367 
368  if(file->FindObjectAny("Channel") != nullptr) {
369  file->Get("Channel"); // this calls TChannel::Streamer
370  }
371  if(file->FindObjectAny("Values") != nullptr) {
372  file->Get("Values");
373  }
374  fRootFilesOpened++;
375  } else {
376  std::cout<<"Could not open "<<filename<<std::endl;
377  }
378  }
379 
380  AddFileToGUI(file);
381  TRunInfo::ReadInfoFromFile();
382 
383  return file;
384 }

409 void TGRSIint::SetupPipeline()
410 {
411  /// Finds all of the files input as well as flags provided and makes all
412  /// of the decisions about what to sort and what order to open everything up
413  /// in. This also creates the output files. Starts the threads and gets the
414  /// sorting going. This is really the brains of the command line sorting routine.
415  TGRSIOptions* opt = TGRSIOptions::Get();
416 
417  // Determining which parts of the pipeline need to be set up.
418 
419  bool missing_raw_file = false;
420  for(auto& filename : opt->InputFiles()) {
421  if(!file_exists(filename.c_str())) {
422  missing_raw_file = true;
423  std::cerr<<"File not found: "<<filename<<std::endl;
424  }
425  }
426 
427  // Which input files do we have
428  bool has_raw_file = !opt->InputFiles().empty() && opt->SortRaw() && !missing_raw_file;
429  bool has_input_fragment_tree = gFragment != nullptr; // && opt->SortRoot();
430  bool has_input_analysis_tree = gAnalysis != nullptr; // && opt->SortRoot();
431 
432  // Which output files are could possibly be made
433  bool able_to_write_fragment_histograms =
434  ((has_raw_file || has_input_fragment_tree) && opt->FragmentHistogramLib().length() > 0);
435  bool able_to_write_fragment_tree = (has_raw_file && !missing_raw_file);
436  bool able_to_write_analysis_histograms = ((has_raw_file || has_input_fragment_tree || has_input_analysis_tree) &&
437  opt->AnalysisHistogramLib().length() > 0);
438  bool able_to_write_analysis_tree = (able_to_write_fragment_tree || has_input_fragment_tree);
439 
440  // Which output files will we make
441  bool write_fragment_histograms = (able_to_write_fragment_histograms && opt->MakeHistos());
442  bool write_fragment_tree = able_to_write_fragment_tree && opt->WriteFragmentTree();
443  bool write_analysis_histograms =
444  (able_to_write_analysis_histograms && opt->MakeHistos()); //TODO: make it so we aren't always trying to generate both frag and analysis histograms
445  bool write_analysis_tree = (able_to_write_analysis_tree && opt->MakeAnalysisTree());
446 
447  // Which steps need to be performed to get from the inputs to the outputs
448  bool self_stopping = opt->CloseAfterSort();
449 
450  bool read_from_raw = (has_raw_file && (write_fragment_histograms || write_fragment_tree ||
451  write_analysis_histograms || write_analysis_tree));
452 
453  bool read_from_fragment_tree =
454  (has_input_fragment_tree && (write_fragment_histograms || write_analysis_histograms || write_analysis_tree));
455 
456  bool generate_analysis_data =
457  ((read_from_raw || read_from_fragment_tree) && (write_analysis_histograms || write_analysis_tree));
458 
459  bool read_from_analysis_tree =
460  (has_input_analysis_tree && (write_analysis_histograms || write_analysis_tree) && !generate_analysis_data);
461 
462  // Extract the run number and sub run number from whatever we were given
463  int run_number = 0;
464  int sub_run_number = 0;
465  if(read_from_raw) {
466  run_number = fRawFiles[0]->GetRunNumber();
467  sub_run_number = fRawFiles[0]->GetSubRunNumber();
468  } else if(read_from_fragment_tree) {
469  auto run_title = gFragment->GetListOfFiles()->At(0)->GetTitle();
470  run_number = GetRunNumber(run_title);
471  sub_run_number = GetSubRunNumber(run_title);
472  } else if(read_from_analysis_tree) {
473  auto run_title = gAnalysis->GetListOfFiles()->At(0)->GetTitle();
474  run_number = GetRunNumber(run_title);
475  sub_run_number = GetSubRunNumber(run_title);
476  }
477 
478  // Choose output file names for the 4 possible output files
479  std::string output_fragment_tree_filename = opt->OutputFragmentFile();
480  if(output_fragment_tree_filename.length() == 0) {
481  if(sub_run_number == -1) {
482  output_fragment_tree_filename = Form("fragment%05i.root", run_number);
483  } else {
484  output_fragment_tree_filename = Form("fragment%05i_%03i.root", run_number, sub_run_number);
485  }
486  }
487 
488  std::string output_fragment_hist_filename = opt->OutputFragmentHistogramFile();
489  if(output_fragment_hist_filename.length() == 0) {
490  if(sub_run_number == -1) {
491  output_fragment_hist_filename = Form("hist_fragment%05i.root", run_number);
492  } else {
493  output_fragment_hist_filename = Form("hist_fragment%05i_%03i.root", run_number, sub_run_number);
494  }
495  }
496 
497  std::string output_analysis_tree_filename = opt->OutputAnalysisFile();
498  if(output_analysis_tree_filename.length() == 0) {
499  if(sub_run_number == -1) {
500  output_analysis_tree_filename = Form("analysis%05i.root", run_number);
501  } else {
502  output_analysis_tree_filename = Form("analysis%05i_%03i.root", run_number, sub_run_number);
503  }
504  }
505 
506  std::string output_analysis_hist_filename = opt->OutputAnalysisHistogramFile();
507  if(output_analysis_hist_filename.length() == 0) {
508  if(sub_run_number == -1) {
509  output_analysis_hist_filename = Form("hist_analysis%05i.root", run_number);
510  } else {
511  output_analysis_hist_filename = Form("hist_analysis%05i_%03i.root", run_number, sub_run_number);
512  }
513  }
514 
515  if(read_from_analysis_tree) {
516  std::cerr<<"Reading from analysis tree not currently supported"<<std::endl;
517  }
518 
519  ////////////////////////////////////////////////////
520  //////////// Setting up the loops ////////////////
521  ////////////////////////////////////////////////////
522 
523  if(!write_fragment_histograms && !write_fragment_tree && !write_analysis_histograms && !write_analysis_tree) {
524  // We still might want to read the calibration, values, or run info files
525  for(const auto& cal_filename : opt->CalInputFiles()) {
526  TChannel::ReadCalFile(cal_filename.c_str());
527  }
528  for(const auto& val_filename : opt->ValInputFiles()) {
529  GValue::ReadValFile(val_filename.c_str());
530  }
531  for(const auto& info_filename : opt->ExternalRunInfo()) {
532  TRunInfo::Get()->ReadInfoFile(info_filename.c_str());
533  }
534  return;
535  }
536  // Set the width of the status and each column
537  StoppableThread::ColumnWidth(TGRSIOptions::Get()->ColumnWidth());
538  StoppableThread::StatusWidth(TGRSIOptions::Get()->StatusWidth());
539 
540  // Different queues that can show up
541  std::vector<std::shared_ptr<ThreadsafeQueue<std::shared_ptr<const TFragment>>>> fragmentQueues;
542  std::vector<std::shared_ptr<ThreadsafeQueue<std::shared_ptr<TEpicsFrag>>>> scalerQueues;
543  std::vector<std::shared_ptr<ThreadsafeQueue<std::shared_ptr<TUnpackedEvent>>>> analysisQueues;
544 
545  // The different loops that can run
546  TDataLoop* dataLoop = nullptr;
547  TUnpackingLoop* unpackLoop = nullptr;
548  TFragmentChainLoop* fragmentChainLoop = nullptr;
549  TEventBuildingLoop* eventBuildingLoop = nullptr;
550  TDetBuildingLoop* detBuildingLoop = nullptr;
551 
552  // If needed, read from the raw file
553  if(read_from_raw) {
554  if(fRawFiles.size() > 1) {
555  std::cerr<<"I'm going to ignore all but first .mid"<<std::endl;
556  }
557 
558  dataLoop = TDataLoop::Get("1_input_loop", fRawFiles[0]);
559  dataLoop->SetSelfStopping(self_stopping);
560 
561  unpackLoop = TUnpackingLoop::Get("2_unpack_loop");
562  unpackLoop->InputQueue() = dataLoop->OutputQueue();
563  }
564 
565  // If needed, read from the fragment tree
566  if(read_from_fragment_tree) {
567  fragmentChainLoop = TFragmentChainLoop::Get("1_chain_loop", gFragment);
568  fragmentChainLoop->SetSelfStopping(self_stopping);
569  }
570 
571  // if I am passed any calibrations, lets load those, this
572  // will overwrite any with the same address previously read in.
573  for(const auto& cal_filename : opt->CalInputFiles()) {
574  TChannel::ReadCalFile(cal_filename.c_str());
575  }
576  // Set the run number and sub-run number
577  if(!fRawFiles.empty()) {
578  TRunInfo::Get()->SetRunInfo(fRawFiles[0]->GetRunNumber(), fRawFiles[0]->GetSubRunNumber());
579  } else {
580  TRunInfo::Get()->SetRunInfo(0, -1);
581  }
582 
583  for(const auto& val_filename : opt->ValInputFiles()) {
584  GValue::ReadValFile(val_filename.c_str());
585  }
586  for(const auto& info_filename : opt->ExternalRunInfo()) {
587  TRunInfo::Get()->ReadInfoFile(info_filename.c_str());
588  }
589 
590  TEventBuildingLoop::EBuildMode event_build_mode = TEventBuildingLoop::EBuildMode::kDefault;
591  if(TRunInfo::Get()->GetDetectorInformation() != nullptr) {
592  // call DetectorInformation::Set again, in case the calibration files we've loaded now changed things
593  TRunInfo::Get()->GetDetectorInformation()->Set();
594  event_build_mode = TRunInfo::Get()->GetDetectorInformation()->BuildMode();
595  } else {
596  std::cout<<"no detector information, can't set build mode"<<std::endl;
597  }
598 
599  // If requested, write the fragment histograms
600  if(write_fragment_histograms) {
601  TFragHistLoop* loop = TFragHistLoop::Get("3_frag_hist_loop");
602  loop->SetOutputFilename(output_fragment_hist_filename);
