427 bool TGRSIint::FileAutoDetect(std::string filename, long filesize) {
428  //first search for extensions.
429  std::string ext = filename.substr(filename.find_last_of('.')+1);
430  //printf("\text = %s\n",ext.c_str());
431  if(ext.compare("root")==0 && filesize > 0) {
432  //printf("\tFound root file: %s\n",filename.c_str());
433  //fInputRootFile->push_back(filename);
434  TGRSIOptions::AddInputRootFile(filename);
435  return true;
436  } else if((ext.compare("mid")==0 || ext.compare("bz2")==0) && filesize > 0) {
437  //printf("\tFound midas file: %s\n",filename.c_str());
438  //fInputMidasFile->push_back(filename);
439  TGRSIOptions::AddInputMidasFile(filename);
440  fAutoSort = true;
441  return true;
442  } else if(ext.compare("cal")==0 && filesize > 0) {
443  //printf("\tFound custom calibration file: %s\n",filename.c_str());
444  //fInputCalFile->push_back(filename);
445  TGRSIOptions::AddInputCalFile(filename);
446  return true;
447  } else if(ext.compare("info")==0 && filesize > 0) {
448  if(TGRSIRunInfo::ReadInfoFile(filename.c_str()))
449  return true;
450  else {
451  printf("Problem reading run-info file %s\n",filename.c_str());
452  return false;
453  }
454  } else if(ext.compare("xml")==0 && filesize > 0) {
455  //fInputOdbFile->push_back(filename);
456  TGRSIOptions::AddInputOdbFile(filename);
457  //printf("\tFound xml odb file: %s\n",filename.c_str());
458  return true;
459  } else if(ext.compare("odb")==0 && filesize > 0) {
460  //printf("\tFound c-like odb file: %s\n",filename.c_str());
461  printf("c-like odb structures can't be read yet.\n");
462  return false;
463  } else if((ext.compare("c")==0) || (ext.compare("C")==0) || (ext.compare("c+")==0) || (ext.compare("C+")==0)) {
464  //scripts are the only files that don't have to exist, they may also be found in the macro-paths
465  TGRSIOptions::AddMacroFile(filename);
466  return true;
467  } else {
468  printf("\tDiscarding unknown file: %s\n",filename.c_str());
469  return false;
470  }
471  return false;
472 }
