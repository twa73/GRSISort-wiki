286 TVector3 TS3::GetPosition(int ring, int sector, bool downstream, double offset) {
287  TVector3 position;
288  double dist = 0;
289  if(offset == 0)
290  dist = fTargetDistance;
291  else
292  dist = offset;
293  double ring_width=(fOuterDiameter-fInnerDiameter)*0.5/fRingNumber; // 24 rings radial width!
294  double inner_radius=fInnerDiameter/2.0;
295  double correctedsector = 6+sector; //moe is currently checking.
296  double phi = 2.*TMath::Pi()/fSectorNumber * (correctedsector + 0.5); //the phi angle....
297  double radius = inner_radius + ring_width * (ring + 0.5) ;
298  if(downstream)
299  position.SetMagThetaPhi(sqrt(radius*radius + dist*dist),atan((radius/dist)),phi+fOffsetPhi);
300  else
301  position.SetMagThetaPhi(sqrt(radius*radius + dist*dist),TMath::Pi() - atan((radius/dist)),phi+fOffsetPhi);
302 
303  return position;
