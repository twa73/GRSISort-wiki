336 TVector3 TS3::GetPosition(int ring, int sector, double offsetphi, double offsetZ, bool sectorsdownstream, bool smear)
337 {
338 
339  double ring_width = (fOuterDiameter - fInnerDiameter) * 0.5 / fRingNumber; // 24 rings radial width!
340  double inner_radius = fInnerDiameter / 2.0;
341  double phi_width = 2. * TMath::Pi() / fSectorNumber;
342  double radius = inner_radius + ring_width * (ring + 0.5);
343  double phi = phi_width * sector; // the phi angle....
344  phi += fOffsetPhiCon;
345  // The above calculates the position on the S3
346 
347  // This sets orientation of the detector face relative to the beam
348  if(sectorsdownstream) {
349  phi = -phi;
350  }
351 
352  //This rotates the detector around the beam-axis into the correct lab position
353  phi += offsetphi;
354 
355  //This produces a uniform distribution over the area of a pixel
356  if(smear) {
357  double sep = ring_width * 0.025;
358  double r1 = radius - ring_width * 0.5 + sep, r2 = radius + ring_width * 0.5 - sep;
359  radius = sqrt(gRandom->Uniform(r1 * r1, r2 * r2));
360  double sepphi = sep / radius;
361  phi = gRandom->Uniform(phi - phi_width * 0.5 + sepphi, phi + phi_width * 0.5 - sepphi);
362  }
363 
364  return TVector3(cos(phi) * radius, sin(phi) * radius, offsetZ);
365 }
