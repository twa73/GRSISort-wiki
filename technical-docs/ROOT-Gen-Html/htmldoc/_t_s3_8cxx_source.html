281 TVector3 TS3::GetPosition(int ring, int sector, double offsetphi,double offsetZ, bool sectorsdownstream,bool smear) {
282 
283  double ring_width=(fOuterDiameter-fInnerDiameter)*0.5/fRingNumber; // 24 rings radial width!
284  double inner_radius=fInnerDiameter/2.0;
285  double phi_width=2.*TMath::Pi()/fSectorNumber;
286 
287  double radius = inner_radius + ring_width * (ring + 0.5) ;
288 
289  double phi = phi_width*-sector; //the phi angle....
290  phi+=fOffsetPhiCon;
291  if(!sectorsdownstream)phi=-phi;
292  phi+=offsetphi;
293 
294  if(smear){
295  double sep=ring_width*0.025;
296  double r1=radius-ring_width*0.5+sep,r2=radius+ring_width*0.5-sep;
297  radius=sqrt(s3_rand.Uniform(r1*r1,r2*r2));
298  double sepphi=sep/radius;
299  phi=s3_rand.Uniform(phi-phi_width*0.5+sepphi,phi+phi_width*0.5-sepphi);
300  }
301 
302  return TVector3(cos(phi)*radius,sin(phi)*radius,offsetZ);
303 }
