99 void TParsingDiagnostics::GoodFragment(const std::shared_ptr<const TFragment>& frag)
100 {
101  /// increment the counter of good fragments for this detector type and check if any trigger ids have been lost
102  fNumberOfGoodFragments[frag->GetDetectorType()]++;
103 
104  UInt_t channelAddress = frag->GetAddress();
105  UInt_t channelId = frag->GetChannelId();
106  long timeStamp = frag->GetTimeStampNs();
107 
108  // check if this is a new minimum/maximum of the channel id
109  if(fMinChannelId.find(channelAddress) == fMinChannelId.end()) { // check if this channel has been found before
110  fMinChannelId[channelAddress] = channelId;
111  fMaxChannelId[channelAddress] = channelId;
112  } else {
113  if(channelId < fMinChannelId[channelAddress]) {
114  fMinChannelId[channelAddress] = channelId;
115  }
116  if(channelId > fMaxChannelId[channelAddress]) {
117  fMaxChannelId[channelAddress] = channelId;
118  }
119  }
120 
121  // count the number of hits for this channel
122  if(fNumberOfHits.find(channelAddress) == fNumberOfHits.end()) {
123  fNumberOfHits[channelAddress] = 0;
124  }
125  ++fNumberOfHits[channelAddress];
126 
127  // check if this is a new minimum/maximum network packet id
128  if(frag->GetNetworkPacketNumber() > 0) {
129  ++fNumberOfNetworkPackets;
130  if(frag->GetNetworkPacketNumber() < fMinNetworkPacketNumber) {
131  fMinNetworkPacketNumber = frag->GetNetworkPacketNumber();
132  }
133  if(frag->GetNetworkPacketNumber() > fMaxNetworkPacketNumber) {
134  fMaxNetworkPacketNumber = frag->GetNetworkPacketNumber();
135  }
136  }
137 
138  // increment the dead time and set per channel min/max timestamps
139  if(fDeadTime.find(channelAddress) == fDeadTime.end()) {
140  fDeadTime[channelAddress] = frag->GetDeadTime();
141  fMinTimeStamp[channelAddress] = timeStamp;
142  fMaxTimeStamp[channelAddress] = timeStamp;
143  } else {
144  fDeadTime[channelAddress] += frag->GetDeadTime();
145  if(timeStamp < fMinTimeStamp[channelAddress]) {
146  fMinTimeStamp[channelAddress] = timeStamp;
147  }
148  if(timeStamp > fMaxTimeStamp[channelAddress]) {
149  fMaxTimeStamp[channelAddress] = timeStamp;
150  }
151  }
152 
153  if(fMinDaqTimeStamp == 0 || frag->GetDaqTimeStamp() < fMinDaqTimeStamp) {
154  fMinDaqTimeStamp = frag->GetDaqTimeStamp();
155  }
156  if(fMaxDaqTimeStamp == 0 || frag->GetDaqTimeStamp() > fMaxDaqTimeStamp) {
157  fMaxDaqTimeStamp = frag->GetDaqTimeStamp();
158  }
159 }

185  fIdHist->SetAxisRange(minChannel, maxChannel + 1);
186  }
187
