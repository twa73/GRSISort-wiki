139 void TParsingDiagnostics::GoodFragment(const std::shared_ptr<const TFragment>& frag)
140 {
141  /// increment the counter of good fragments for this detector type and check if any trigger ids have been lost
142  fNumberOfGoodFragments[frag->GetDetectorType()]++;
143 
144  UInt_t channelAddress = frag->GetAddress();

179 void TParsingDiagnostics::Draw(Option_t* opt)
180 {
181  UInt_t minChannel = fChannelAddressData.begin()->first;
182  UInt_t maxChannel = fChannelAddressData.begin()->first;
183 
184  for(auto it : fChannelAddressData) {
185  if(it.first < minChannel) minChannel = it.first;
186  if(it.first > maxChannel) maxChannel = it.first;
187  }
188 
189  // check that the histogram (if it already exists) has the right number of bins
190  if(fIdHist != nullptr && fIdHist->GetNbinsX() != static_cast<Int_t>(maxChannel - minChannel + 1)) {
191  delete fIdHist;
192  fIdHist = nullptr;
193  }
194  if(fIdHist == nullptr) {
195  fIdHist = new TH1F("IdHist", "Event survival;channel number;survival rate [%]", maxChannel - minChannel + 1,
196  minChannel, maxChannel + 1);
197  } else {
198  // the histogram already had the right number of bins, but to be save we set the range
199  fIdHist->SetAxisRange(minChannel, maxChannel + 1);
200  }
201 
202  for(auto it : fChannelAddressData) {
203  if(it.second.MinChannelId() != 0 || it.second.MinChannelId() != it.second.MaxChannelId()) {
204  fIdHist->SetBinContent(fIdHist->GetXaxis()->FindBin(it.first),
205  (100. * it.second.NumberOfHits()) / (it.second.MaxChannelId() - it.second.MinChannelId() + 1.));
206  }
207  }
208 
209  fIdHist->Draw(opt);
210 }
