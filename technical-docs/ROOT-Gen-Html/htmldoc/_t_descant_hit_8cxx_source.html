160 bool TDescantHit::AnalyzeWaveform()
161 {
162  bool error = false;
163 
164  if(fWaveform.empty()) {
165  return false; // Error!
166  }
167  std::vector<Int_t> baselineCorrections(8, 0);
168  std::vector<Short_t> smoothedWaveform;
169 
170  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as
171  // integers)
172  unsigned int interpolationSteps = 256;
173  int delay = 8;
174  double attenuation = 24. / 64.;
175  int halfSmoothingWindow = 0; // 2*halfsmoothingwindow + 1 = number of samples in moving window.
176 
177  // baseline algorithm: correct each adc with average of first two samples in that adc
178  for(size_t i = 0; i < 8 && i < fWaveform.size(); ++i) {
179  baselineCorrections[i] = fWaveform[i];
180  }
181  for(size_t i = 8; i < 16 && i < fWaveform.size(); ++i) {
182  baselineCorrections[i - 8] =
183  ((baselineCorrections[i - 8] + fWaveform[i]) + ((baselineCorrections[i - 8] + fWaveform[i]) > 0 ? 1 : -1)) >>
184  1; // Average
185  }
186  for(size_t i = 0; i < fWaveform.size(); ++i) {
187  fWaveform[i] -= baselineCorrections[i % 8];
188  }
189 
190  SetCfd(CalculateCfd(attenuation, delay, halfSmoothingWindow, interpolationSteps));
191 
192  // PSD
193  // time to zero-crossing algorithm: time when sum reaches n% of the total sum minus the cfd time
194  double fraction = 0.90;
195 
196  SetPsd(CalculatePsd(fraction, interpolationSteps));
197 
198  SetCharge(CalculatePartialSum().back());
199 
200  if(fPsd < 0) {
201  error = true;
202  }
203 
204  return !(error);
205 }

216 Int_t TDescantHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfSmoothingWindow,
217  unsigned int interpolationSteps, std::vector<Short_t>& monitor)
218 {
219 
220  Short_t monitormax = 0;
221  bool armed = false;
222 
223  Int_t cfd = 0;
224  if(fWaveform.empty()) {
225  return INT_MAX; // Error!
226  }
227  std::vector<Short_t> smoothedWaveform;
228 
229  if(fWaveform.size() > delay + 1) {
230  if(halfSmoothingWindow > 0) {
231  smoothedWaveform = TDescantHit::CalculateSmoothedWaveform(halfSmoothingWindow);
232  } else {
233  smoothedWaveform = fWaveform;
234  }
235 
236  monitor.clear();
237  monitor.resize(smoothedWaveform.size() - delay);
238  monitor[0] = attenuation * smoothedWaveform[delay] - smoothedWaveform[0];
239  if(monitor[0] > monitormax) {
240  armed = true;
241  monitormax = monitor[0];
242  }
243 
244  for(size_t i = delay + 1; i < smoothedWaveform.size(); ++i) {
245  monitor[i - delay] = attenuation * smoothedWaveform[i] - smoothedWaveform[i - delay];
246  if(monitor[i - delay] > monitormax) {
247  armed = true;
248  monitormax = monitor[i - delay];
249  } else {
250  if(armed && monitor[i - delay] < 0) {
251  armed = false;
252  if(monitor[i - delay - 1] - monitor[i - delay] != 0) {
253  // Linear interpolation.
254  cfd = (i - delay - 1) * interpolationSteps +
255  (monitor[i - delay - 1] * interpolationSteps) / (monitor[i - delay - 1] - monitor[i - delay]);
256  } else {
257  // Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
258  cfd = 0;
259  }
260  }
261  }
262  }
263  } else {
264  monitor.resize(0);
265  }
266 
267  if(TGRSIOptions::Get()->Debug()) {
268  fCfdMonitor = monitor;
269  }
270 
271  // correct for remainder between the 100MHz timestamp and the 125MHz start of the waveform
272  // we save this in the upper bits, otherwise we can't correct the waveform themselves
273  cfd = (cfd & 0x3fffff) | (static_cast<Int_t>(fCfd) & 0x7c00000);
274 
275  return cfd;
276 }

279 {

351 Int_t TDescantHit::CalculatePsdAndPartialSums(double fraction, unsigned int interpolationSteps,
352  std::vector<Int_t>& partialSums)
353 {
354 
355  Int_t psd = 0;
356 
357  partialSums = CalculatePartialSum();
358  if(partialSums.empty()) {
359  return -1;
360  }
361  int totalSum = partialSums.back();
362 
363  fPsd = -1;
364  if(partialSums[0] < fraction * totalSum) {
365  for(size_t i = 1; i < partialSums.size(); ++i) {
366  if(partialSums[i] >= fraction * totalSum) {
367  psd = i * interpolationSteps - ((partialSums[i] - fraction * totalSum) * interpolationSteps) / fWaveform[i];
368  break;
369  }
370  }
371  }
372 
373  return psd;
