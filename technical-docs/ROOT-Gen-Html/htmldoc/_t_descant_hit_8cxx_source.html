67 bool TDescantHit::AnalyzeWaveform() {
68  bool error = false;
69 
70  std::vector<Short_t> *waveform = GetWaveform();
71  if(waveform->empty()) {
72  return false; //Error!
73  }
74  std::vector<Int_t> baselineCorrections(8, 0);
75  std::vector<Short_t> smoothedWaveform;
76 
77  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as integers)
78  unsigned int interpolationSteps = 256;
79  int delay = 8;
80  double attenuation = 24./64.;
81  int halfSmoothingWindow = 0; //2*halfsmoothingwindow + 1 = number of samples in moving window.
82 
83  // baseline algorithm: correct each adc with average of first two samples in that adc
84  for(size_t i = 0; i < 8 && i < waveform->size(); ++i) {
85  baselineCorrections[i] = (*waveform)[i];
86  }
87  for(size_t i = 8; i < 16 && i < waveform->size(); ++i) {
88  baselineCorrections[i-8] = ((baselineCorrections[i-8] + (*waveform)[i]) + ((baselineCorrections[i-8] + (*waveform)[i]) > 0 ? 1 : -1)) >> 1; //Average
89  }
90  for(size_t i = 0; i < waveform->size(); ++i) {
91  (*waveform)[i] -= baselineCorrections[i%8];
92  }
93 
94  this->SetCfd(CalculateCfd(attenuation, delay, halfSmoothingWindow, interpolationSteps));
95 
96  // PSD
97  // time to zero-crossing algorithm: time when sum reaches n% of the total sum minus the cfd time
98  double fraction = 0.90;
99 
100  SetPsd(CalculatePsd(fraction, interpolationSteps));
101 
102  SetCharge(CalculatePartialSum().back());
103 
104 
105  if(fPsd < 0) {
106  error = true;
107  }
108 
109  return !(error);
110 
111 }

121 Int_t TDescantHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfSmoothingWindow, unsigned int interpolationSteps, std::vector<Short_t> &monitor) {
122 
123  Short_t monitormax = 0;
124  bool armed = false;
125 
126  Int_t cfd = 0;
127  std::vector<Short_t> *waveform = GetWaveform();
128  if(waveform->empty()) {
129  return INT_MAX; //Error!
130  }
131  std::vector<Short_t> smoothedWaveform;
132 
133  if(waveform->size() > delay+1) {
134  if(halfSmoothingWindow > 0) {
135  smoothedWaveform = TDescantHit::CalculateSmoothedWaveform(halfSmoothingWindow);
136  } else {
137  smoothedWaveform = *waveform;
138  }
139 
140  monitor.clear();
141  monitor.resize(smoothedWaveform.size()-delay);
142  monitor[0] = attenuation*smoothedWaveform[delay]-smoothedWaveform[0];
143  if(monitor[0] > monitormax) {
144  armed = true;
145  monitormax = monitor[0];
146  }
147 
148  for(size_t i = delay + 1; i < smoothedWaveform.size(); ++i) {
149  monitor[i-delay] = attenuation*smoothedWaveform[i]-smoothedWaveform[i-delay];
150  if(monitor[i-delay] > monitormax) {
151  armed=true;
152  monitormax = monitor[i-delay];
153  } else {
154  if(armed == true && monitor[i-delay] < 0) {
155  armed = false;
156  if(monitor[i-delay-1] - monitor[i-delay] != 0) {
157  //Linear interpolation.
158  cfd = (i-delay-1)*interpolationSteps + (monitor[i-delay-1]*interpolationSteps)/(monitor[i-delay-1]-monitor[i-delay]);
159  } else {
160  //Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
161  cfd = 0;
162  }
163  }
164  }
165  }
166  } else {
167  monitor.resize(0);
168  }
169 
170  return cfd;
171 }

254  psd = i*interpolationSteps - ((partialSums[i]-fraction*totalSum)*interpolationSteps)/(*waveform)[i];

256  }
257  }
258  }
