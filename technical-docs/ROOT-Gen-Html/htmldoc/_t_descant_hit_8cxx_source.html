150 bool TDescantHit::AnalyzeWaveform() {
151  bool error = false;
152 
153  if(fWaveform.empty()) {
154  return false; //Error!
155  }
156  std::vector<Int_t> baselineCorrections(8, 0);
157  std::vector<Short_t> smoothedWaveform;
158 
159  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as integers)
160  unsigned int interpolationSteps = 256;
161  int delay = 8;
162  double attenuation = 24./64.;
163  int halfSmoothingWindow = 0; //2*halfsmoothingwindow + 1 = number of samples in moving window.
164 
165  // baseline algorithm: correct each adc with average of first two samples in that adc
166  for(size_t i = 0; i < 8 && i < fWaveform.size(); ++i) {
167  baselineCorrections[i] = fWaveform[i];
168  }
169  for(size_t i = 8; i < 16 && i < fWaveform.size(); ++i) {
170  baselineCorrections[i-8] = ((baselineCorrections[i-8] + fWaveform[i]) + ((baselineCorrections[i-8] + fWaveform[i]) > 0 ? 1 : -1)) >> 1; //Average
171  }
172  for(size_t i = 0; i < fWaveform.size(); ++i) {
173  fWaveform[i] -= baselineCorrections[i%8];
174  }
175 
176  SetCfd(CalculateCfd(attenuation, delay, halfSmoothingWindow, interpolationSteps));
177 
178  // PSD
179  // time to zero-crossing algorithm: time when sum reaches n% of the total sum minus the cfd time
180  double fraction = 0.90;
181 
182  SetPsd(CalculatePsd(fraction, interpolationSteps));
183 
184  SetCharge(CalculatePartialSum().back());
185 
186  if(fPsd < 0) {
187  error = true;
188  }
189 
190  return !(error);
191 
192 }

202 Int_t TDescantHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfSmoothingWindow, unsigned int interpolationSteps, std::vector<Short_t> &monitor) {
203 
204  Short_t monitormax = 0;
205  bool armed = false;
206 
207  Int_t cfd = 0;
208  if(fWaveform.empty()) {
209  return INT_MAX; //Error!
210  }
211  std::vector<Short_t> smoothedWaveform;
212 
213  if(fWaveform.size() > delay+1) {
214  if(halfSmoothingWindow > 0) {
215  smoothedWaveform = TDescantHit::CalculateSmoothedWaveform(halfSmoothingWindow);
216  } else {
217  smoothedWaveform = fWaveform;
218  }
219 
220  monitor.clear();
221  monitor.resize(smoothedWaveform.size()-delay);
222  monitor[0] = attenuation*smoothedWaveform[delay]-smoothedWaveform[0];
223  if(monitor[0] > monitormax) {
224  armed = true;
225  monitormax = monitor[0];
226  }
227 
228  for(size_t i = delay + 1; i < smoothedWaveform.size(); ++i) {
229  monitor[i-delay] = attenuation*smoothedWaveform[i]-smoothedWaveform[i-delay];
230  if(monitor[i-delay] > monitormax) {
231  armed=true;
232  monitormax = monitor[i-delay];
233  } else {
234  if(armed == true && monitor[i-delay] < 0) {
235  armed = false;
236  if(monitor[i-delay-1] - monitor[i-delay] != 0) {
237  //Linear interpolation.
238  cfd = (i-delay-1)*interpolationSteps + (monitor[i-delay-1]*interpolationSteps)/(monitor[i-delay-1]-monitor[i-delay]);
239  } else {
240  //Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
241  cfd = 0;
242  }
243  }
244  }
245  }
246  } else {
247  monitor.resize(0);
248  }
249 
250  if(TGRSIOptions::Get()->Debug()) {
251  fCfdMonitor = monitor;
252  }
253 
254  // correct for remainder between the 100MHz timestamp and the 125MHz start of the waveform
255  // we save this in the upper bits, otherwise we can't correct the waveform themselves
256  cfd = (cfd & 0x3fffff) | (fCfd & 0x7c00000);
257 
258  return cfd;
259 }

328 Int_t TDescantHit::CalculatePsdAndPartialSums(double fraction, unsigned int interpolationSteps, std::vector<Int_t>& partialSums) {
329 
330  Int_t psd = 0;
331 
332  partialSums = CalculatePartialSum();
333  if(partialSums.empty()) {
334  return -1;
335  }
336  int totalSum = partialSums.back();
337 
338  fPsd = -1;
339  if(partialSums[0] < fraction*totalSum) {
340  for(size_t i = 1; i < partialSums.size(); ++i) {
341  if(partialSums[i] >= fraction*totalSum) {
342  psd = i*interpolationSteps - ((partialSums[i]-fraction*totalSum)*interpolationSteps)/fWaveform[i];

346  }
