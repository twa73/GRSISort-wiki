550 minimalist_th3slice& minimalist_th3slice::operator= (TH3* input){
551 
552  raw_3d=input;
553  while(raw_3d->GetXaxis()->GetNbins()>1000)raw_3d->RebinX();
554  while(raw_3d->GetYaxis()->GetNbins()>1000)raw_3d->RebinY();
555  while(raw_3d->GetZaxis()->GetNbins()>1000)raw_3d->RebinZ();
556 
557  bin=0;
