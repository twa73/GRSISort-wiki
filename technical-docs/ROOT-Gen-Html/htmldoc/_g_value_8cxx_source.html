228 int GValue::ParseInputData(const std::string& input, EPriority priority, Option_t* opt)
229 {
230  std::istringstream infile(input);
231  GValue* value = nullptr;
232  std::string line;
233  int linenumber = 0;
234  int newvalues = 0;
235 
236  bool brace_open = false;
237  std::string name;
238 
239  while( !std::getline(infile, line).fail() ) {
240  linenumber++;
241  trim(&line);
242  size_t comment = line.find("//");
243  if(comment != std::string::npos) {
244  line = line.substr(0, comment);
245  }
246  if(line.length() == 0) {
247  continue;
248  }
249  size_t openbrace = line.find('{');
250  size_t closebrace = line.find('}');
251  size_t colon = line.find(':');
252 
253  //=============================================//
254  if(openbrace == std::string::npos && closebrace == std::string::npos && colon == std::string::npos) {
255  continue;
256  }
257  //=============================================//
258  if(openbrace != std::string::npos) {
259  brace_open = true;
260  name = line.substr(0, openbrace);
261  trim(&name);
262  value = new GValue(name.c_str());
263  }
264  //=============================================//
265  if(brace_open) {
266  if(colon != std::string::npos) {
267  std::string type;
268  if(openbrace == std::string::npos || openbrace > colon) {
269  type = line.substr(0, colon);
270  } else {
271  type = line.substr(openbrace + 1, colon - (openbrace + 1));
272  }
273  line = line.substr(colon + 1, line.length());
274  trim(&line); //strip beginning whitespace (not needed for value itself, but for the readability of info)
275  trim(&type);
276  int j = 0;
277  while(type[j] != 0) {
278  char c = *(type.c_str() + j);
279  c = toupper(c);
280  type[j++] = c;
281  }
282  if(type.compare("NAME") == 0) {
283  value->SetName(line.c_str());
284  } else if(type.compare("VALUE") == 0) {
285  value->SetValue(std::atof(line.c_str()));
286  value->fPriority = priority;
287  } else if(type.compare("INFO") == 0) {
288  value->SetInfo(line.c_str());
289  }
290  }
291  }
292  //=============================================//
293  if(closebrace != std::string::npos) {
294  brace_open = false;
295  if(value != nullptr) {
296  // Check whether value is in vector. If it isn't add it.
297  GValue* cur_value = FindValue(value->GetName());
298  if(cur_value == nullptr) {
299  AddValue(value);
300  newvalues++;
301  } else {
302  value->AppendValue(cur_value);
303  delete value;
304  newvalues++;
305  }
306  }
307  value = nullptr;
308  name.clear();
309  }
310  }
311  if(strcmp(opt, "debug") == 0) {
312  std::cout<<"parsed "<<linenumber<<" lines"<<std::endl;
313  }
314  return newvalues;
315 }
