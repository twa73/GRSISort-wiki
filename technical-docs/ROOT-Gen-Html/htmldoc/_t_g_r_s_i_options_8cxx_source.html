413 kFileType TGRSIOptions::DetermineFileType(const std::string& filename) const
414 {
415  size_t dot_pos = filename.find_last_of('.');
416  size_t slash_pos = filename.find_last_of('/');
417  // if we didn't find a . (or if it was before the last /) we don't have any extension
418  // => so it's a TDR file
419  if(dot_pos == std::string::npos || (dot_pos < slash_pos && slash_pos != std::string::npos)) {
420  return kFileType::TDR_FILE;
421  }
422  std::string ext = filename.substr(dot_pos + 1);
423 
424  //check if this is a zipped file and if so get the extension before the zip-extension
425  bool isZipped = (ext == "gz") || (ext == "bz2") || (ext == "zip");
426  if(isZipped) {
427  std::string remaining = filename.substr(0, dot_pos);
428  ext = remaining.substr(remaining.find_last_of('.') + 1);
429  }
430 
431  if(ext == "mid") {
432  return kFileType::MIDAS_FILE;
433  }
434  if(ext == "lst") {
435  return kFileType::LST_FILE;
436  }
437  if(ext == "evt") {
438  return kFileType::NSCL_EVT;
439  }
440  if(ext == "cal") {
441  return kFileType::CALIBRATED;
442  }
443  if(ext == "root") {
444  return kFileType::ROOT_DATA;
445  }
446  if((ext == "c") || (ext == "C") || (ext == "c+") || (ext == "C+") || (ext == "c++") || (ext == "C++")) {
447  return kFileType::ROOT_MACRO;
448  }
449  if(ext == "dat" || ext == "cvt") {
450  if(filename.find("GlobalRaw") != std::string::npos) {
451  return kFileType::GRETINA_MODE3;
452  }
453  return kFileType::GRETINA_MODE2;
454  }
455  if(ext == "hist") {
456  return kFileType::GUI_HIST_FILE;
457  }
458  if(ext == "so") {
459  return kFileType::COMPILED_SHARED_LIBRARY;
460  }
461  if(ext == "info") {
462  return kFileType::CONFIG_FILE;
463  }
464  if(ext == "val") {
465  return kFileType::GVALUE;
466  }
467  if(ext == "win") {
468  return kFileType::PRESETWINDOW;
469  }
470  if(ext == "cuts") {
471  return kFileType::CUTS_FILE;
472  }
473  if(ext == "xml") {
474  return kFileType::XML_FILE;
475  }
476 
477  // strip possible parenthese with arguments for the script from the extension
478  size_t opening_pos = ext.find_first_of('(');
479  if(opening_pos != std::string::npos) {
480  ext = ext.substr(0,opening_pos);
481  if((ext == "c") || (ext == "C") || (ext == "c+") || (ext == "C+") || (ext == "c++") || (ext == "C++")) {
482  return kFileType::ROOT_MACRO;
483  }
484  }
485  return kFileType::UNKNOWN_FILETYPE;
486 }

488 bool TGRSIOptions::FileAutoDetect(const std::string& filename)
489 {
490  /// Detects the type of file provided on the command line. This uses the extension to determine
491  /// the type of the file. Once the type is determined, the file is sent to the appropriate list
492  /// in TGRSIOptions. This is also smart enough to dynamically link histogramming libraries.
493  switch(DetermineFileType(filename)) {
494  case kFileType::NSCL_EVT:
495  case kFileType::GRETINA_MODE2:
496  case kFileType::GRETINA_MODE3:
497  case kFileType::MIDAS_FILE: fInputFiles.push_back(filename); return true;
498 
499  case kFileType::LST_FILE: fInputFiles.push_back(filename); return true;
500 
501  case kFileType::TDR_FILE: fInputFiles.push_back(filename); return true;
502 
503  case kFileType::ROOT_DATA: fInputRootFiles.push_back(filename); return true;
504 
505  case kFileType::ROOT_MACRO: fMacroFiles.push_back(filename); return true;
506 
507  case kFileType::CALIBRATED: fInputCalFiles.push_back(filename); return true;
508 
509  case kFileType::COMPILED_SHARED_LIBRARY: {
510 
511  bool used = false;
512  DynamicLibrary lib(filename);
513  if(lib.GetSymbol("MakeFragmentHistograms") != nullptr) {
514  fFragmentHistogramLib = filename;
515  used = true;
516  }
517  if(lib.GetSymbol("MakeAnalysisHistograms") != nullptr) {
518  fAnalysisHistogramLib = filename;
519  used = true;
520  }
521  if(lib.GetSymbol("CreateParser") != nullptr && lib.GetSymbol("DestroyParser") != nullptr &&
522  lib.GetSymbol("CreateFile") != nullptr && lib.GetSymbol("DestroyFile") != nullptr) {
523  fParserLibrary = filename;
524  used = true;
525  }
526  if(!used) {
527  std::cerr<<filename<<" did not contain MakeFragmentHistograms() or MakeAnalysisHistograms()"<<std::endl;
528  }
529  return true;
530  }
531 
532  case kFileType::GVALUE: fInputValFiles.push_back(filename); return true;
533 
534  case kFileType::PRESETWINDOW: fInputWinFiles.push_back(filename); return true;
535 
536  case kFileType::CUTS_FILE: fInputCutFiles.push_back(filename); return true;
537 
538  case kFileType::CONFIG_FILE: return false;
539 
540  case kFileType::XML_FILE: fInputOdbFiles.push_back(filename); return true;
541 
542  case kFileType::UNKNOWN_FILETYPE:
543  default: printf("\tDiscarding unknown file: %s\n", filename.c_str()); return false;
544  }
545 }
