359 kFileType TGRSIOptions::DetermineFileType(const std::string& filename) const{
360  size_t dot_pos = filename.find_last_of('.');
361  std::string ext = filename.substr(dot_pos+1);
362 
363  bool isZipped = (ext=="gz") || (ext=="bz2") || (ext=="zip");
364  if(isZipped){
365  std::string remaining = filename.substr(0,dot_pos);
366  ext = remaining.substr(remaining.find_last_of('.')+1);
367  }
368 
369  if(ext=="mid"){
370  return kFileType::MIDAS_FILE;
371  } else if(ext=="evt"){
372  return kFileType::NSCL_EVT;
373  } else if (ext == "cal") {
374  return kFileType::CALIBRATED;
375  } else if (ext == "root") {
376  return kFileType::ROOT_DATA;
377  } else if ((ext == "c") || (ext == "C")
378  || (ext == "c+") || (ext == "C+")
379  || (ext == "c++") || (ext == "C++")) {
380  return kFileType::ROOT_MACRO;
381  } else if (ext == "dat" || ext == "cvt") {
382  if(filename.find("GlobalRaw")!=std::string::npos)
383  return kFileType::GRETINA_MODE3;
384  return kFileType::GRETINA_MODE2;
385  } else if (ext == "hist") {
386  return kFileType::GUI_HIST_FILE;
387  } else if (ext == "so") {
388  return kFileType::COMPILED_SHARED_LIBRARY;
389  } else if (ext == "info") {
390  return kFileType::CONFIG_FILE;
391  } else if (ext == "val"){
392  return kFileType::GVALUE;
393  } else if (ext == "win"){
394  return kFileType::PRESETWINDOW;
395  } else if (ext == "cuts") {
396  return kFileType::CUTS_FILE;
397  } else if (ext == "xml") {
398  return kFileType::XML_FILE;
399  } else {
400  return kFileType::UNKNOWN_FILETYPE;
401  }
402 }

404 bool TGRSIOptions::FileAutoDetect(const std::string& filename) {
405  switch(DetermineFileType(filename)){
406  case kFileType::NSCL_EVT:
407  case kFileType::GRETINA_MODE2:
408  case kFileType::GRETINA_MODE3:
409  case kFileType::MIDAS_FILE:
410  fInputMidasFiles.push_back(filename);
411  return true;
412 
413  case kFileType::ROOT_DATA:
414  fInputRootFiles.push_back(filename);
415  return true;
416 
417  case kFileType::ROOT_MACRO:
418  fMacroFiles.push_back(filename);
419  return true;
420 
421  case kFileType::CALIBRATED:
422  fInputCalFiles.push_back(filename);
423  return true;
424 
425  case kFileType::COMPILED_SHARED_LIBRARY: {
426 
427  bool used = false;
428  DynamicLibrary lib(filename);
429  if(lib.GetSymbol("MakeFragmentHistograms")) {
430  fragment_histogram_lib = filename;
431  used = true;
432  }
433  if(lib.GetSymbol("MakeAnalysisHistograms")) {
434  analysis_histogram_lib = filename;
435  used = true;
436  }
437  if(!used) {
438  std::cerr << filename << " did not contain MakeFragmentHistograms() or MakeAnalysisHistograms()"
439  << std::endl;
440  }
441  return true;
442  }
443 
444  case kFileType::GVALUE:
445  fInputValFiles.push_back(filename);
446  return true;
447 
448  case kFileType::PRESETWINDOW:
449  fInputWinFiles.push_back(filename);
450  return true;
451 
452  case kFileType::CUTS_FILE:
453  fInputCutsFiles.push_back(filename);
454  return true;
455 
456  case kFileType::CONFIG_FILE:
457  return false;
458 
459  case kFileType::XML_FILE:
460  fInputOdbFiles.push_back(filename);
461  return true;
462 
463  case kFileType::UNKNOWN_FILETYPE:
464  default:
465  printf("\tDiscarding unknown file: %s\n",filename.c_str());
466  return false;
467  }
468 }
