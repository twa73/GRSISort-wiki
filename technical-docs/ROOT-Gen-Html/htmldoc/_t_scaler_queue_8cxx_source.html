49 {
50  /// Clear the entire Queue, Queue counters, and timer.
51  bool locked = false;
52  if(!fDeadtimeScalerQueue.empty()) {
53  while(!TDeadtimeScalerQueue::All.try_lock()) {
54  // do nothing
55  }
56  locked = true;
57  }
58 
59  if(fScalersInQueue != 0) {
60  printf(RED "\n\tWarning, discarding %i Scalers!" RESET_COLOR "\n", fScalersInQueue);
61  while(!fDeadtimeScalerQueue.empty()) {
62  fDeadtimeScalerQueue.pop();
63  }
64  fScalersInQueue = 0;
65  }
66 
67  fScalersIn = 0;
68  fScalersOut = 0;
69 
70  fTotalScalersIn = 0;
71  fTotalScalersOut = 0;
72 
73  fStopwatch->Reset();
74 
75  if(locked) {
76  TDeadtimeScalerQueue::All.unlock();
77  }
78  return;

169 void TDeadtimeScalerQueue::StatusUpdate()
170 {
171  // Updates the status of the scaler Queue
172  float time = 0;
173  float scaler_rate_in = 0;
174  float scaler_rate_out = 0;
175 
176  while(fStatusUpdateOn) {

178  scaler_rate_in = fScalersIn / time;
179  scaler_rate_out = fScalersOut / time;
180  while(!TDeadtimeScalerQueue::All.try_lock()) {
181  // do nothing

183 
184  printf(BLUE "\n\tscalers rate in = %.2f/sec, nqueue = %d\n" RESET_COLOR, scaler_rate_in, Size());
185  printf(DGREEN "\tscalers rate out = %.2f/sec\n" RESET_COLOR, scaler_rate_out);
186  TDeadtimeScalerQueue::All.unlock();
187  ResetRateCounter();
188  fStopwatch->Start();
189  }
