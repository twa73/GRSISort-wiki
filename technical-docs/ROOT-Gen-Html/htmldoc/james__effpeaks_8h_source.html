76 //The 7 parameter radware gamma effciency function for TF1 use
77 inline Double_t radeff(Double_t *x, Double_t *par){
78 
79  if(x[0]<1)return 0;
80 
81  double xx=log(x[0]/100.);
82  double yy=log(x[0]/1000.);
83 
84  double AA=(par[0]+xx*par[1]);
85  double BB=(par[2]+yy*par[3]+yy*yy*par[4]);
86 
87  double a=0,b=0,c=0,d=0;
88  double G=abs(par[5])+2;//dont let it be less than ^2
89 
90  if(AA>0)a=1/pow(AA,G);
91  if(BB>0)b=1/pow(BB,G);
92  c=a+b;
93  if(c>0)d=1./pow(c,1./G);
94 
95  if((AA<log(1000)&&abs(AA)>0)||d<log(1000))return 0;
96 
97  double ret=(exp(d)/1000.0)-1;//Because function only works in stupid range
98 // if(ret<0)return 0;
99  return ret;
100 
101 }

111 //A 4 parameter simple gamma efficiency function for TF1 use
112 inline Double_t stefeff(Double_t *x, Double_t *par){
113 
114  if(x[0]<1)return 0;
115 
116  double xx=x[0];
117  double yy=log10(x[0]);
118 
119  double a=par[0]*yy+par[1]*yy*yy+par[2]/(xx*xx);
120 
121  double d=pow(10,a);
122 
123  if(d<0)return 0;
124 
125  return d*par[3];
126 }

146 //A 3 parameter high energy gamma efficiency function from gosia gremlin
147 inline Double_t GosiaHigh(Double_t *x, Double_t *par){
148  if(x[0]<1)return 0;
149  double xx=log(x[0]/50.);
150  double ret=0;
151  for(int i=0;i<4;i++){
152  ret+=par[i]*pow(xx,i);
153  }
154  return exp(ret);
155 }

157 //A 5 parameter gamma efficiency function from gosia gremlin
158 inline Double_t GosiaAdhoc(Double_t *x, Double_t *par){
159  if(x[0]<1)return 0;
160  double xx=log(x[0]/50.);
161  return GosiaHigh(x,par)*exp(-abs(par[4])/pow(xx,abs(par[5])+1));
162 }
