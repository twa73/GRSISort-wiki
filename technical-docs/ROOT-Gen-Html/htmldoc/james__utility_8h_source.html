70 //
71 // Sort a second vector based on the first vector
72 // first bool=true ascending
73 // second bool=true sort the first vector aswell
74 //
75 template<typename T>
76 void vector_dd_sort(vector< double >& fValues,vector< T >& fValuesT, bool fUp=true,bool both=false){
77  double fN=fValues.size();
78  vector<pair<double,T>> fPair;
79  for(int i=0;i<fN;i++)fPair.push_back(pair<double,T>(fValues[i],fValuesT[i]));
80 
81  if(fUp)std::sort(fPair.begin(), fPair.end(), PairCompareUp<T>);
82  else std::sort(fPair.begin(), fPair.end(), PairCompareDown<T>);
83 
84  for(int i=0;i<fN;i++){
85  if(both)fValues[i]=fPair[i].first;
86  fValuesT[i]=fPair[i].second;
87  }
88 }
