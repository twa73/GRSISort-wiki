56  // This function Loops through all libraries in GRSISort, and brute-force loads them into proof

59 
60  const char* pPath = getenv("GRSISYS");
61 
62  // First set the include path on each slave
63  Exec(Form(R"(gInterpreter->AddIncludePath("%s/include"))", pPath));
64  std::cout<<"Loading library '"<<pPath<<"/lib/libGRSI.so'"<<std::endl;
65  Exec(Form(R"(gSystem->Load("%s/lib/libGRSI.so");)", pPath));
66  // if we have a data parser/detector library load it
67  std::string library = TGRSIOptions::Get()->ParserLibrary();
68  if(library.empty()) {
69  std::cout<<DYELLOW<<"Not loading any parser library, this might not work!"<<RESET_COLOR<<std::endl;

72  Exec(Form(R"(gSystem->Load("%s");)", library.c_str()));
