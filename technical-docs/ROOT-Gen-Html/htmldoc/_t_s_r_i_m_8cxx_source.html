211 double TSRIM::GetAdjustedEnergy(double energy,double thickness,double stepsize) {
212  if (fEnergyLoss == 0) {
213  printf("energy loss file has not yet been read in.\n");
214  return 0.0;

216 
217  double energy_temp = energy;
218  double xstep = stepsize, xtot = 0.0; // MAKE XSTEP SMALLER FOR BETTER RESULTS. 1UM SHOULD BE FINE ... UNLESS YOU ARE AT THE BRAGG PEAK ??
219  energy_temp -= fmod(thickness,xstep) * sEnergyLoss->Eval(energy_temp); // get rid of fractional distance
220  xtot += fmod(thickness,xstep);
221 
222  if(thickness >= xstep) {
223 
224  while(xtot<thickness){
225 // printf("energy_temp = %.02f\t DEDX = %.02f\n",energy_temp,fEnergyLoss->Eval(energy_temp,0,"S"));
226  energy_temp -= xstep * sEnergyLoss->Eval(energy_temp); // update energy recursively so that it decreases each step
227  xtot += xstep;
228  if(energy_temp<=0.0)
229  return 0.0; // if no energy is remaining then final energy is zero
230  }
231  }
232 
233  return energy_temp;
