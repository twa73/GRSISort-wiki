397 {
398  // Increment cell defined by namex,namey by a weight w
399  //
400  // if x or/and y is less than the low-edge of the corresponding axis first bin,
401  // the Underflow cell is incremented.
402  // if x or/and y is greater than the upper edge of corresponding axis last bin,
403  // the Overflow cell is incremented.
404  //
405  // If the storage of the sum of squares of weights has been triggered,
406  // via the function Sumw2, then the sum of the squares of weights is incremented
407  // by w^2 in the cell corresponding to x,y.
408  //
409

413  biny = fYaxis.FindBin(namey);
414  if(binx < 0 || biny < 0) {
415  return -1;
416  }
417  if(biny >= binx) {
418  bin = binx;
419  binx = biny;
420  biny = bin;
421  }
422  bin = biny * (2 * fXaxis.GetNbins() - biny + 3) / 2 + binx;
423  AddBinContent(bin, w);
424  if(fSumw2.fN != 0) {
425  fSumw2.fArray[bin] += w * w;
426  }
427  if(binx == 0 || binx > fXaxis.GetNbins()) {

429  }

431  return -1;
432  }
433  Double_t x = fXaxis.GetBinCenter(binx);
434  Double_t y = fYaxis.GetBinCenter(biny);

438  fTsumwx2 += w * x * x;

474 void GHSym::FillRandom(const char* fname, Int_t ntimes)
475 {
476  //*-*-*-*-*-*-*Fill histogram following distribution in function fname*-*-*-*
477  //*-* =======================================================
478  //*-*
479  //*-* The distribution contained in the function fname (TF2) is integrated
480  //*-* over the channel contents.
481  //*-* It is normalized to 1.
482  //*-* Getting one random number implies:
483  //*-* - Generating a random number between 0 and 1 (say r1)
484  //*-* - Look in which bin in the normalized integral r1 corresponds to
485  //*-* - Fill histogram channel
486  //*-* ntimes random numbers are generated
487  //*-*
488  //*-* One can also call TF2::GetRandom2 to get a random variate from a function.
489  //*-*
490  //*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-**-*-*-*-*-*-*-*
491 
492  Int_t bin, binx, biny, ibin;
493  Double_t r1, x, y;
494  //*-*- Search for fname in the list of ROOT defined functions
495  TObject* fobj = gROOT->GetFunction(fname);
496  if(fobj == nullptr) {
497  Error("FillRandom", "Unknown function: %s", fname);
498  return;
499  }
500  TF2* f1 = static_cast<TF2*>(fobj);
501  if(f1 == nullptr) {
502  Error("FillRandom", "Function: %s is not a TF2", fname);
503  return;
504  }
505 
506  //*-*- Allocate temporary space to store the integral and compute integral
507  Int_t nbinsx = GetNbinsX();
508  Int_t nbinsy = GetNbinsY();
509  Int_t nbins = nbinsx * nbinsy;
510 
511  auto* integral = new Double_t[nbins + 1];
512  ibin = 0;
513  integral[ibin] = 0;
514  for(biny = 1; biny <= nbinsy; ++biny) {
515  for(binx = 1; binx <= nbinsx; ++binx) {
516  ++ibin;
517  Double_t fint = f1->Integral(fXaxis.GetBinLowEdge(binx), fXaxis.GetBinUpEdge(binx), fYaxis.GetBinLowEdge(biny),
518  fYaxis.GetBinUpEdge(biny));
519  integral[ibin] = integral[ibin - 1] + fint;
520  }
521  }
522 
523  //*-*- Normalize integral to 1
524  if(integral[nbins] == 0) {
525  delete[] integral;
526  Error("FillRandom", "Integral = zero");
527  return;
528  }
529  for(bin = 1; bin <= nbins; ++bin) {
530  integral[bin] /= integral[nbins];
531  }
532 
533  //*-*--------------Start main loop ntimes
534  for(int loop = 0; loop < ntimes; ++loop) {
535  r1 = gRandom->Rndm(loop);
536  ibin = TMath::BinarySearch(nbins, &integral[0], r1);
537  biny = ibin / nbinsx;
538  binx = 1 + ibin - nbinsx * biny;
539  ++biny;
540  x = fXaxis.GetBinCenter(binx);
541  y = fYaxis.GetBinCenter(biny);
542  Fill(x, y);
543  }
544  delete[] integral;
545 }

650 void GHSym::FitSlices(TF1* f1, Int_t firstbin, Int_t lastbin, Int_t cut, Option_t* option, TObjArray* arr)
651 {
652  // Project slices along X in case of a 2-D histogram, then fit each slice
653  // with function f1 and make a histogram for each fit parameter
654  // Only bins along Y between firstbin and lastbin are considered.
655  // By default (firstbin == 0, lastbin == -1), all bins in y including
656  // over- and underflows are taken into account.
657  // If f1=0, a gaussian is assumed
658  // Before invoking this function, one can set a subrange to be fitted along X
659  // via f1->SetRange(xmin,xmax)
660  // The argument option (default="QNR") can be used to change the fit options.
661  // "Q" means Quiet mode
662  // "N" means do not show the result of the fit
663  // "R" means fit the function in the specified function range
664  // "G2" merge 2 consecutive bins along X
665  // "G3" merge 3 consecutive bins along X
666  // "G4" merge 4 consecutive bins along X
667  // "G5" merge 5 consecutive bins along X
668  // "S" sliding merge: merge n consecutive bins along X accordingly to what Gn is given.
669  // It makes sense when used together with a Gn option
670  //
671  // The generated histograms are returned by adding them to arr, if arr is not NULL.
672  // arr's SetOwner() is called, to signal that it is the user's respponsability to
673  // delete the histograms, possibly by deleting the arrary.
674  // TObjArray aSlices;
675  // h2->FitSlicesX(func, 0, -1, 0, "QNR", &aSlices);
676  // will already delete the histograms once aSlice goes out of scope. aSlices will
677  // contain the histogram for the i-th parameter of the fit function at aSlices[i];
678  // aSlices[n] (n being the number of parameters) contains the chi2 distribution of
679  // the fits.
680  //
681  // If arr is NULL, the generated histograms are added to the list of objects
682  // in the current directory. It is the user's responsability to delete
683  // these histograms.
684  //
685  // Example: Assume a 2-d histogram h2
686  // Root > h2->FitSlicesX(); produces 4 TH1D histograms
687  // with h2_0 containing parameter 0(Constant) for a Gaus fit
688  // of each bin in Y projected along X
689  // with h2_1 containing parameter 1(Mean) for a gaus fit
690  // with h2_2 containing parameter 2(RMS) for a gaus fit
691  // with h2_chi2 containing the chisquare/number of degrees of freedom for a gaus fit
692  //
693  // Root > h2->FitSlicesX(0,15,22,10);
694  // same as above, but only for bins 15 to 22 along Y
695  // and only for bins in Y for which the corresponding projection
696  // along X has more than cut bins filled.
697  //
698  // NOTE: To access the generated histograms in the current directory, do eg:
699  // TH1D *h2_1 = (TH1D*)gDirectory->Get("h2_1");
700 
701  Int_t nbins = fYaxis.GetNbins();
702  if(firstbin < 0) {
703  firstbin = 0;
704  }
705  if(lastbin < 0 || lastbin > nbins + 1) {
706  lastbin = nbins + 1;
707  }
708  if(lastbin < firstbin) {
709  firstbin = 0;
710  lastbin = nbins + 1;
711  }
712  TString opt = option;
713  opt.ToLower();
714  Int_t ngroup = 1;
715  if(opt.Contains("g2")) {
716  ngroup = 2;
717  opt.ReplaceAll("g2", "");
718  }
719  if(opt.Contains("g3")) {
720  ngroup = 3;
721  opt.ReplaceAll("g3", "");
722  }
723  if(opt.Contains("g4")) {
724  ngroup = 4;
725  opt.ReplaceAll("g4", "");
726  }
727  if(opt.Contains("g5")) {
728  ngroup = 5;
729  opt.ReplaceAll("g5", "");
730  }
731 
732  // implement option S sliding merge for each bin using in conjunction with a given Gn
733  Int_t nstep = ngroup;
734  if(opt.Contains("s")) {
735  nstep = 1;
736  }
737 
738  // default is to fit with a gaussian
739  if(f1 == nullptr) {
740  f1 = static_cast<TF1*>(gROOT->GetFunction("gaus"));
741  if(f1 == nullptr) {
742  f1 = new TF1("gaus", "gaus", fXaxis.GetXmin(), fXaxis.GetXmax());
743  } else {
744  f1->SetRange(fXaxis.GetXmin(), fXaxis.GetXmax());
745  }
746  }
747  Int_t npar = f1->GetNpar();
748  if(npar <= 0) {
749  return;
750  }
751  auto* parsave = new Double_t[npar];
752  f1->GetParameters(parsave);
753 
754  if(arr != nullptr) {
755  arr->SetOwner();
756  arr->Expand(npar + 1);
757  }
758 
759  // Create one histogram for each function parameter
760  auto** hlist = new TH1D*[npar];
761  auto* name = new char[2000];
762  auto* title = new char[2000];
763  const TArrayD* bins = fYaxis.GetXbins();
764  for(Int_t ipar = 0; ipar < npar; ++ipar) {
765  snprintf(name, 2000, "%s_%d", GetName(), ipar);
766  snprintf(title, 2000, "Fitted value of par[%d]=%s", ipar, f1->GetParName(ipar));
767  delete gDirectory->FindObject(name);
768  if(bins->fN == 0) {
769  hlist[ipar] = new TH1D(name, title, nbins, fYaxis.GetXmin(), fYaxis.GetXmax());
770  } else {
771  hlist[ipar] = new TH1D(name, title, nbins, bins->fArray);
772  }
773  hlist[ipar]->GetXaxis()->SetTitle(fYaxis.GetTitle());
774  if(arr != nullptr) {
775  (*arr)[ipar] = hlist[ipar];
776  }
777  }
778  snprintf(name, 2000, "%s_chi2", GetName());
779  delete gDirectory->FindObject(name);
780  TH1D* hchi2 = nullptr;
781  if(bins->fN == 0) {
782  hchi2 = new TH1D(name, "chisquare", nbins, fYaxis.GetXmin(), fYaxis.GetXmax());
783  } else {
784  hchi2 = new TH1D(name, "chisquare", nbins, bins->fArray);
785  }
786  hchi2->GetXaxis()->SetTitle(fYaxis.GetTitle());
787  if(arr != nullptr) {
788  (*arr)[npar] = hchi2;
789  }
790 
791  // Loop on all bins in Y, generate a projection along X
792  Int_t bin;
793  Long64_t nentries;
794  // in case of sliding merge nstep=1, i.e. do slices starting for every bin
795  // now do not slices case with overflow (makes more sense)
796  for(bin = firstbin; bin + ngroup - 1 <= lastbin; bin += nstep) {
797  TH1D* hp = Projection("_temp", bin, bin + ngroup - 1, "e");
798  if(hp == nullptr) {
799  continue;
800  }
801  nentries = Long64_t(hp->GetEntries());
802  if(nentries == 0 || nentries < cut) {
803  delete hp;
804  continue;
805  }
806  f1->SetParameters(parsave);
807  hp->Fit(f1, opt.Data());
808  Int_t npfits = f1->GetNumberFitPoints();
809  if(npfits > npar && npfits >= cut) {
810  Int_t binOn = bin + ngroup / 2;
811  for(Int_t ipar = 0; ipar < npar; ++ipar) {
812  hlist[ipar]->Fill(fYaxis.GetBinCenter(binOn), f1->GetParameter(ipar));
813  hlist[ipar]->SetBinError(binOn, f1->GetParError(ipar));
814  }
815  hchi2->Fill(fYaxis.GetBinCenter(binOn), f1->GetChisquare() / (npfits - npar));
816  }
817  delete hp;
818  }
819  delete[] parsave;
820  delete[] name;
821  delete[] title;
822  delete[] hlist;
823 }

846 Double_t GHSym::GetBinWithContent2(Double_t c, Int_t& binx, Int_t& biny, Int_t firstxbin, Int_t lastxbin,
847  Int_t firstybin, Int_t lastybin, Double_t maxdiff) const
848 {
849  // compute first cell (binx,biny) in the range [firstxbin,lastxbin][firstybin,lastybin] for which
850  // diff = abs(cell_content-c) <= maxdiff
851  // In case several cells in the specified range with diff=0 are found
852  // the first cell found is returned in binx,biny.
853  // In case several cells in the specified range satisfy diff <=maxdiff
854  // the cell with the smallest difference is returned in binx,biny.
855  // In all cases the function returns the smallest difference.
856  //
857  // NOTE1: if firstxbin < 0, firstxbin is set to 1
858  // if(lastxbin < firstxbin then lastxbin is set to the number of bins in X
859  // ie if firstxbin=1 and lastxbin=0 (default) the search is on all bins in X except
860  // for X's under- and overflow bins.
861  // if firstybin < 0, firstybin is set to 1
862  // if(lastybin < firstybin then lastybin is set to the number of bins in Y
863  // ie if firstybin=1 and lastybin=0 (default) the search is on all bins in Y except
864  // for Y's under- and overflow bins.
865  // NOTE2: if maxdiff=0 (default), the first cell with content=c is returned.
866 
867  if(fDimension != 2) {
868  binx = -1;
869  biny = -1;
870  Error("GetBinWithContent2", "function is only valid for 2-D histograms");
871  return 0;
872  }
873  if(firstxbin < 0) {
874  firstxbin = 1;
875  }
876  if(lastxbin < firstxbin) {
877  lastxbin = fXaxis.GetNbins();
878  }
879  if(firstybin < 0) {
880  firstybin = 1;
881  }
882  if(lastybin < firstybin) {
883  lastybin = fYaxis.GetNbins();
884  }
885  Double_t diff, curmax = 1.e240;
886  for(Int_t j = firstybin; j <= lastybin; j++) {
887  for(Int_t i = firstxbin; i <= lastxbin; i++) {
888  diff = TMath::Abs(GetBinContent(i, j) - c);
889  if(diff <= 0) {
890  binx = i;
891  biny = j;
892  return diff;
893  }
894  if(diff < curmax && diff <= maxdiff) {
895  curmax = diff, binx = i;
896  biny = j;
897  }
898  }
899  }
900  return curmax;
901 }

1010 {

1012  // The array stats must be correctly dimensionned in the calling program.
1013  // stats[0] = sumw
1014  // stats[1] = sumw2
1015  // stats[2] = sumwx
1016  // stats[3] = sumwx2
1017  // stats[4] = sumwy
1018  // stats[5] = sumwy2
1019  // stats[6] = sumwxy
1020  //
1021  // If no axis-subranges are specified (via TAxis::SetRange), the array stats
1022  // is simply a copy of the statistics quantities computed at filling time.
1023  // If sub-ranges are specified, the function recomputes these quantities
1024  // from the bin contents in the current axis ranges.
1025  //
1026  // Note that the mean value/RMS is computed using the bins in the currently
1027  // defined ranges (see TAxis::SetRange). By default the ranges include
1028  // all bins from 1 to nbins included, excluding underflows and overflows.
1029  // To force the underflows and overflows in the computation, one must
1030  // call the static function TH1::StatOverflows(kTRUE) before filling

1032 
1033  if(fBuffer != nullptr) {
1034  const_cast<GHSym*>(this)->BufferEmpty();
1035  }

1037  Int_t bin, binx, biny;
1038  Double_t w, err;
1039  Double_t x, y;

1041  for(bin = 0; bin < 7; ++bin) {
1042  stats[bin] = 0;
1043  }

1046  Int_t lastBinX = fXaxis.GetLast();

1140 #if ROOT_VERSION_CODE < ROOT_VERSION(6, 20, 0)
1141 Double_t GHSym::Interpolate(Double_t x, Double_t y)
1142 #else
1143 Double_t GHSym::Interpolate(Double_t x, Double_t y) const
1144 #endif
1145 {
1146  // Given a point P(x,y), Interpolate approximates the value via bilinear
1147  // interpolation based on the four nearest bin centers
1148  // see Wikipedia, Bilinear Interpolation
1149  // Andy Mastbaum 10/8/2008
1150  // vaguely based on R.Raja 6-Sep-2008
1151 
1152  Double_t f = 0;
1153  Double_t x1 = 0;
1154  Double_t x2 = 0;
1155  Double_t y1 = 0;
1156  Double_t y2 = 0;
1157  Double_t dx, dy;
1158  Int_t bin_x = fXaxis.FindBin(x);
1159  Int_t bin_y = fYaxis.FindBin(y);
1160  if(bin_x < 1 || bin_x > GetNbinsX() || bin_y < 1 || bin_y > GetNbinsY()) {
1161  Error("Interpolate", "Cannot interpolate outside histogram domain.");
1162  return 0;
1163  }
1164  Int_t quadrant = 0; // CCW from UR 1,2,3,4
1165  // which quadrant of the bin (bin_P) are we in?
1166  dx = fXaxis.GetBinUpEdge(bin_x) - x;
1167  dy = fYaxis.GetBinUpEdge(bin_y) - y;
1168  if(dx <= fXaxis.GetBinWidth(bin_x) / 2 && dy <= fYaxis.GetBinWidth(bin_y) / 2) {
1169  quadrant = 1; // upper right
1170  }
1171  if(dx > fXaxis.GetBinWidth(bin_x) / 2 && dy <= fYaxis.GetBinWidth(bin_y) / 2) {
1172  quadrant = 2; // upper left
1173  }
1174  if(dx > fXaxis.GetBinWidth(bin_x) / 2 && dy > fYaxis.GetBinWidth(bin_y) / 2) {
1175  quadrant = 3; // lower left
1176  }
1177  if(dx <= fXaxis.GetBinWidth(bin_x) / 2 && dy > fYaxis.GetBinWidth(bin_y) / 2) {
1178  quadrant = 4; // lower right
1179  }
1180  switch(quadrant) {
1181  case 1:
1182  x1 = fXaxis.GetBinCenter(bin_x);
1183  y1 = fYaxis.GetBinCenter(bin_y);
1184  x2 = fXaxis.GetBinCenter(bin_x + 1);
1185  y2 = fYaxis.GetBinCenter(bin_y + 1);
1186  break;
1187  case 2:
1188  x1 = fXaxis.GetBinCenter(bin_x - 1);
1189  y1 = fYaxis.GetBinCenter(bin_y);
1190  x2 = fXaxis.GetBinCenter(bin_x);
1191  y2 = fYaxis.GetBinCenter(bin_y + 1);
1192  break;
1193  case 3:
1194  x1 = fXaxis.GetBinCenter(bin_x - 1);
1195  y1 = fYaxis.GetBinCenter(bin_y - 1);
1196  x2 = fXaxis.GetBinCenter(bin_x);
1197  y2 = fYaxis.GetBinCenter(bin_y);
1198  break;
1199  case 4:

1247 Double_t GHSym::KolmogorovTest(const TH1* h2, Option_t* option) const
1248 {
1249  // Statistical test of compatibility in shape between
1250  // THIS histogram and h2, using Kolmogorov test.
1251  // Default: Ignore under- and overflow bins in comparison
1252  //
1253  // option is a character string to specify options
1254  // "U" include Underflows in test
1255  // "O" include Overflows
1256  // "N" include comparison of normalizations
1257  // "D" Put out a line of "Debug" printout
1258  // "M" Return the Maximum Kolmogorov distance instead of prob
1259  //
1260  // The returned function value is the probability of test
1261  // (much less than one means NOT compatible)
1262  //
1263  // The KS test uses the distance between the pseudo-CDF's obtained
1264  // from the histogram. Since in 2D the order for generating the pseudo-CDF is
1265  // arbitrary, two pairs of pseudo-CDF are used, one starting from the x axis the
1266  // other from the y axis and the maximum distance is the average of the two maximum
1267  // distances obtained.
1268  //
1269  // Code adapted by Rene Brun from original HBOOK routine HDIFF
1270 
1271  TString opt = option;
1272  opt.ToUpper();
1273 
1274  Double_t prb = 0;
1275  TH1* h1 = const_cast<TH1*>(static_cast<const TH1*>(this));
1276  if(h2 == nullptr) {
1277  return 0;
1278  }
1279  TAxis* xaxis1 = h1->GetXaxis();
1280  TAxis* xaxis2 = const_cast<TAxis*>(h2->GetXaxis());
1281  TAxis* yaxis1 = h1->GetYaxis();
1282  TAxis* yaxis2 = const_cast<TAxis*>(h2->GetYaxis());
1283  Int_t ncx1 = xaxis1->GetNbins();
1284  Int_t ncx2 = xaxis2->GetNbins();
1285  Int_t ncy1 = yaxis1->GetNbins();
1286  Int_t ncy2 = yaxis2->GetNbins();
1287 
1288  // Check consistency of dimensions
1289  if(h1->GetDimension() != 2 || h2->GetDimension() != 2) {
1290  Error("KolmogorovTest", "Histograms must be 2-D\n");
1291  return 0;
1292  }
1293 
1294  // Check consistency in number of channels
1295  if(ncx1 != ncx2) {
1296  Error("KolmogorovTest", "Number of channels in X is different, %d and %d\n", ncx1, ncx2);
1297  return 0;
1298  }
1299  if(ncy1 != ncy2) {
1300  Error("KolmogorovTest", "Number of channels in Y is different, %d and %d\n", ncy1, ncy2);
1301  return 0;
1302  }
1303 
1304  // Check consistency in channel edges
1305  Bool_t afunc1 = kFALSE;
1306  Bool_t afunc2 = kFALSE;
1307  Double_t difprec = 1e-5;
1308  Double_t diff1 = TMath::Abs(xaxis1->GetXmin() - xaxis2->GetXmin());
1309  Double_t diff2 = TMath::Abs(xaxis1->GetXmax() - xaxis2->GetXmax());
1310  if(diff1 > difprec || diff2 > difprec) {
1311  Error("KolmogorovTest", "histograms with different binning along X");
1312  return 0;
1313  }
1314  diff1 = TMath::Abs(yaxis1->GetXmin() - yaxis2->GetXmin());
1315  diff2 = TMath::Abs(yaxis1->GetXmax() - yaxis2->GetXmax());
1316  if(diff1 > difprec || diff2 > difprec) {
1317  Error("KolmogorovTest", "histograms with different binning along Y");
1318  return 0;
1319  }
1320 
1321  // Should we include Uflows, Oflows?
1322  Int_t ibeg = 1;
1323  Int_t jbeg = 1;
1324  Int_t iend = ncx1;
1325  Int_t jend = ncy1;
1326  if(opt.Contains("U")) {
1327  ibeg = 0;
1328  jbeg = 0;
1329  }
1330  if(opt.Contains("O")) {
1331  iend = ncx1 + 1;
1332  jend = ncy1 + 1;
1333  }
1334 
1335  Int_t i, j;
1336  Double_t sum1 = 0;
1337  Double_t sum2 = 0;
1338  Double_t w1 = 0;
1339  Double_t w2 = 0;
1340  for(i = ibeg; i <= iend; ++i) {
1341  for(j = jbeg; j <= jend; ++j) {
1342  sum1 += h1->GetBinContent(i, j);
1343  sum2 += h2->GetBinContent(i, j);
1344  Double_t ew1 = h1->GetBinError(i, j);
1345  Double_t ew2 = h2->GetBinError(i, j);
1346  w1 += ew1 * ew1;
1347  w2 += ew2 * ew2;
1348  }
1349  }
1350 
1351  // Check that both scatterplots contain events
1352  if(sum1 == 0) {
1353  Error("KolmogorovTest", "Integral is zero for h1=%s\n", h1->GetName());
1354  return 0;
1355  }
1356  if(sum2 == 0) {
1357  Error("KolmogorovTest", "Integral is zero for h2=%s\n", h2->GetName());
1358  return 0;
1359  }
1360  // calculate the effective entries.
1361  // the case when errors are zero (w1 == 0 or w2 ==0) are equivalent to
1362  // compare to a function. In that case the rescaling is done only on sqrt(esum2) or sqrt(esum1)
1363  Double_t esum1 = 0, esum2 = 0;
1364  if(w1 > 0) {
1365  esum1 = sum1 * sum1 / w1;
1366  } else {
1367  afunc1 = kTRUE; // use later for calculating z
1368  }
1369 
1370  if(w2 > 0) {
1371  esum2 = sum2 * sum2 / w2;
1372  } else {
1373  afunc2 = kTRUE; // use later for calculating z
1374  }
1375 
1376  if(afunc2 && afunc1) {
1377  Error("KolmogorovTest", "Errors are zero for both histograms\n");
1378  return 0;
1379  }
1380 
1381  // Find first Kolmogorov distance
1382  Double_t s1 = 1 / sum1;
1383  Double_t s2 = 1 / sum2;
1384  Double_t dfmax1 = 0;
1385  Double_t rsum1 = 0, rsum2 = 0;
1386  for(i = ibeg; i <= iend; ++i) {
1387  for(j = jbeg; j <= jend; ++j) {
1388  rsum1 += s1 * h1->GetCellContent(i, j);
1389  rsum2 += s2 * h2->GetCellContent(i, j);
1390  dfmax1 = TMath::Max(dfmax1, TMath::Abs(rsum1 - rsum2));
1391  }
1392  }
1393 
1394  // Find second Kolmogorov distance
1395  Double_t dfmax2 = 0;
1396  rsum1 = 0, rsum2 = 0;
1397  for(j = jbeg; j <= jend; ++j) {
1398  for(i = ibeg; i <= iend; ++i) {
1399  rsum1 += s1 * h1->GetCellContent(i, j);
1400  rsum2 += s2 * h2->GetCellContent(i, j);
1401  dfmax2 = TMath::Max(dfmax2, TMath::Abs(rsum1 - rsum2));
1402  }
1403  }
1404 
1405  // Get Kolmogorov probability: use effective entries, esum1 or esum2, for normalizing it
1406  Double_t factnm;
1407  if(afunc1) {
1408  factnm = TMath::Sqrt(esum2);
1409  } else if(afunc2) {
1410  factnm = TMath::Sqrt(esum1);
1411  } else {
1412  factnm = TMath::Sqrt(esum1 * sum2 / (esum1 + esum2));
1413  }
1414 
1415  // take average of the two distances
1416  Double_t dfmax = 0.5 * (dfmax1 + dfmax2);
1417  Double_t z = dfmax * factnm;
1418 
1419  prb = TMath::KolmogorovProb(z);
1420 
1421  Double_t prb1 = 0;
1422  Double_t prb2 = 0;
1423  // option N to combine normalization makes sense if both afunc1 and afunc2 are false
1424  if(opt.Contains("N") && !(afunc1 || afunc2)) {
1425  // Combine probabilities for shape and normalization
1426  prb1 = prb;
1427  Double_t d12 = esum1 - esum2;
1428  Double_t chi2 = d12 * d12 / (esum1 + esum2);
1429  prb2 = TMath::Prob(chi2, 1);
1430  // see Eadie et al., section 11.6.2
1431  if(prb > 0 && prb2 > 0) {
1432  prb = prb * prb2 * (1 - TMath::Log(prb * prb2));
1433  } else {
1434  prb = 0;
1435  }
1436  }
1437  // debug printout
1438  if(opt.Contains("D")) {
1439  printf(" Kolmo Prob h1 = %s, sum1=%g\n", h1->GetName(), sum1);
1440  printf(" Kolmo Prob h2 = %s, sum2=%g\n", h2->GetName(), sum2);
1441  printf(" Kolmo Probabil = %f, Max Dist = %g\n", prb, dfmax);
1442  if(opt.Contains("N")) {
1443  printf(" Kolmo Probabil = %f for shape alone, =%f for normalisation alone\n", prb1, prb2);
1444  }
1445  }
1446  // This numerical error condition should never occur:
1447  if(TMath::Abs(rsum1 - 1) > 0.002) {
1448  Warning("KolmogorovTest", "Numerical problems with h1=%s\n", h1->GetName());
1449  }
1450  if(TMath::Abs(rsum2 - 1) > 0.002) {
1451  Warning("KolmogorovTest", "Numerical problems with h2=%s\n", h2->GetName());
1452  }
1453 
1454  if(opt.Contains("M")) {
1455  return dfmax; // return avergae of max distance
1456  }
1457 
1458  return prb;
1459 }

2623 void GHSym::Smooth(Int_t ntimes, Option_t* option)
2624 {
2625  // Smooth bin contents of this 2-d histogram using kernel algorithms
2626  // similar to the ones used in the raster graphics community.
2627  // Bin contents in the active range are replaced by their smooth values.
2628  // If Errors are defined via Sumw2, they are also scaled and computed.
2629  // However, note the resulting errors will be correlated between different-bins, so
2630  // the errors should not be used blindly to perform any calculation involving several bins,
2631  // like fitting the histogram. One would need to compute also the bin by bin correlation matrix.
2632  //
2633  // 3 kernels are proposed k5a, k5b and k3a.
2634  // k5a and k5b act on 5x5 cells (i-2,i-1,i,i+1,i+2, and same for j)
2635  // k5b is a bit more stronger in smoothing
2636  // k3a acts only on 3x3 cells (i-1,i,i+1, and same for j).
2637  // By default the kernel "k5a" is used. You can select the kernels "k5b" or "k3a"
2638  // via the option argument.
2639  // If TAxis::SetRange has been called on the x or/and y axis, only the bins
2640  // in the specified range are smoothed.
2641  // In the current implementation if the first argument is not used (default value=1).
2642  //
2643  // implementation by David McKee (dmckee@bama.ua.edu). Extended by Rene Brun
2644 
2645  Double_t k5a[5][5] = {{0, 0, 1, 0, 0}, {0, 2, 2, 2, 0}, {1, 2, 5, 2, 1}, {0, 2, 2, 2, 0}, {0, 0, 1, 0, 0}};
2646  Double_t k5b[5][5] = {{0, 1, 2, 1, 0}, {1, 2, 4, 2, 1}, {2, 4, 8, 4, 2}, {1, 2, 4, 2, 1}, {0, 1, 2, 1, 0}};
2647  Double_t k3a[3][3] = {{0, 1, 0}, {1, 2, 1}, {0, 1, 0}};
2648 
2649  if(ntimes > 1) {
2650  Warning("Smooth", "Currently only ntimes=1 is supported");
2651  }
2652  TString opt = option;
2653  opt.ToLower();
2654  Int_t ksize_x = 5;
2655  Int_t ksize_y = 5;
2656  Double_t* kernel = &k5a[0][0];
2657  if(opt.Contains("k5b")) {
2658  kernel = &k5b[0][0];
2659  }
2660  if(opt.Contains("k3a")) {
2661  kernel = &k3a[0][0];
2662  ksize_x = 3;
2663  ksize_y = 3;
2664  }
2665 
2666  // find i,j ranges
2667  Int_t ifirst = fXaxis.GetFirst();
2668  Int_t ilast = fXaxis.GetLast();
2669  Int_t jfirst = fYaxis.GetFirst();
2670  Int_t jlast = fYaxis.GetLast();
2671 
2672  // Determine the size of the bin buffer(s) needed
2673  Double_t nentries = fEntries;
2674  Int_t nx = GetNbinsX();
2675  Int_t ny = GetNbinsY();
2676  Int_t bufSize = (nx + 2) * (ny + 2);
2677  auto* buf = new Double_t[bufSize];
2678  Double_t* ebuf = nullptr;
2679  if(fSumw2.fN != 0) {
2680  ebuf = new Double_t[bufSize];
2681  }
2682 
2683  // Copy all the data to the temporary buffers
2684  Int_t i, j, bin;
2685  for(i = ifirst; i <= ilast; ++i) {
2686  for(j = jfirst; j <= jlast; ++j) {
2687  bin = GetBin(i, j);
2688  buf[bin] = GetBinContent(bin);
2689  if(ebuf != nullptr) {
2690  ebuf[bin] = GetBinError(bin);
2691  }
2692  }
2693  }
2694 
2695  // Kernel tail sizes (kernel sizes must be odd for this to work!)
2696  Int_t x_push = (ksize_x - 1) / 2;
2697  Int_t y_push = (ksize_y - 1) / 2;
2698 
2699  // main work loop
2700  for(i = ifirst; i <= ilast; ++i) {
2701  for(j = jfirst; j <= jlast; ++j) {
2702  Double_t content = 0.0;
2703  Double_t error = 0.0;
2704  Double_t norm = 0.0;
2705 
2706  for(Int_t n = 0; n < ksize_x; ++n) {
2707  for(Int_t m = 0; m < ksize_y; ++m) {
2708  Int_t xb = i + (n - x_push);
2709  Int_t yb = j + (m - y_push);
2710  if((xb >= 1) && (xb <= nx) && (yb >= 1) && (yb <= ny)) {
2711  bin = GetBin(xb, yb);
2712  Double_t k = kernel[n * ksize_y + m];
2713  if(k != 0.0) {
2714  norm += k;
2715  content += k * buf[bin];
2716  if(ebuf != nullptr) {
2717  error += k * k * ebuf[bin] * ebuf[bin];
2718  }
2719  }
2720  }
2721  }
2722  }
2723 
2724  if(norm != 0.0) {
2725  SetBinContent(i, j, content / norm);
2726  if(ebuf != nullptr) {
2727  error /= (norm * norm);
2728  SetBinError(i, j, sqrt(error));
2729  }
2730  }
2731  }
2732  }
2733  fEntries = nentries;
2734 
2735  delete[] buf;
2736  delete[] ebuf;
2737 }
