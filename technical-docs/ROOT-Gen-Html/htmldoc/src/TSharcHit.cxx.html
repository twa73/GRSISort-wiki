TVector3 TSharcHit::GetPosition() const
{
	return GetPosition(GetDefaultDistance());
}

Double_t TSharcHit::GetTheta(double Xoff, double Yoff, double Zoff)
{
	TVector3 posOff;
	posOff.SetXYZ(Xoff, Yoff, Zoff);
	return (GetPosition() + posOff).Theta();
}
