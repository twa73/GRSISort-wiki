884 int TPulseAnalyzer::GetCsIShape()
885 {
886 
887  int dim=4;
888 
889  long double sum,tau,tau_i,tau_j;
890  int p,q,d;
891 
892  memset(lineq_matrix,0,sizeof(lineq_matrix));
893  memset(lineq_vector,0,sizeof(lineq_vector));
894  memset(lineq_solution,0,sizeof(lineq_solution));
895 
896  /* q is the low limit of the signal section */
897  q=(int)wpar->temax;
898  if(q >= N || q<=0){
899  shpar->chisq=-1;
900  return -1.;
901  }
902 
903  /* p is the high limit of the baseline section */
904  p=(int)wpar->temin;
905  if(p >= N || p<=0){
906  shpar->chisq=-1;
907  return -1.;
908  }
909  lineq_dim=dim;
910 
911  //initialize amplitudes to 0
912  for(int i=0;i<NSHAPE;i++)
913  shpar->am[i]=0.;
914 
915  //printf("N:\t%i\n",N);
916  //printf("p:\t%i\n",p);
917  //printf("lineqdim:\t%i\n",lineq_dim);
918 
919  d=N;
920 
921  //initialize chi square 0 and ndf = n-k to -k where k=dim
922  shpar->chisq=0.;
923  shpar->ndf=-lineq_dim;
924 
925  /**************************************************************************
926  linearized chi square fit is Mu = v where M is a data matrix
927  u, v are vectors; u is the parameter vector (solution)
928  note that in this formulation, chisq_min = y_i^2-sum(u_iv_i)
929  **************************************************************************/
930 
931  //create matrix for linearized fit
932  for(int i=1;i<lineq_dim;i++)
933  {
934  tau=GetCsITau(i);
935  tau_i=tau;
936  sum=-((double)q)/tau+log(1.-exp(-((double)(d-q))/tau));
937  sum-=log(1.-exp(-1./tau));
938  lineq_matrix[i][0]=exp(sum);
939  lineq_matrix[0][i]=exp(sum);
940 
941  tau/=2.;
942  sum=-((double)q)/tau+log(1.-exp(-((double)(d-q))/tau));
943  sum-=log(1.-exp(-1./tau));
944  lineq_matrix[i][i]=exp(sum);
945 
946  for(int j=i+1;j<lineq_dim;j++)
947  {
948  tau_j=GetCsITau(j);
949  tau=(tau_i*tau_j)/(tau_i+tau_j);
950  sum=-((double)q)/tau+log(1.-exp(-((double)(d-q))/tau));
951  sum-=log(1.-exp(-1./tau));
952  lineq_matrix[i][j]=exp(sum);
953  lineq_matrix[j][i]=exp(sum);
954  }
955 
956  }
957 
958  lineq_vector[0]=0;
959  lineq_matrix[0][0]=0;
960 
961  for(int j=q;j<N;j++)
962  {
963  lineq_vector[0]+=wavebuffer[j];
964  lineq_matrix[0][0]+=1;

966  shpar->ndf+=1;
967  }
968 
969  if(lineq_dim >= N){
970  shpar->chisq=-1;
971  return -1.;
972  }
973 
974  for(int i=1;i<lineq_dim;i++)
975  {
976  tau=GetCsITau(i);
977  lineq_vector[i]=0;
978  for(int j=q;j<N;j++)
979  lineq_vector[i]+=wavebuffer[j]*exp(-(double(j))/tau);
980  }
981 
982 
983  for(int j=0;j<p;j++)
984  {
985  lineq_vector[0]+=wavebuffer[j];
986  lineq_matrix[0][0]+=1;
987  shpar->chisq+=wavebuffer[j]*wavebuffer[j];
988  shpar->ndf+=1;
989  }
990 
991  //solve the matrix equation Mu = v -> u = M^(-1)v where M^(-1) is the inverse
992  //of M. note this has no solution if M is not invertable!
993 
994  //error if the matrix cannot be inverted
995  if(solve_lin_eq()==0)
996  {
997  //printf("Matrix could not be inverted\n");
998  shpar->chisq=BADCHISQ_MAT;
999  shpar->ndf=1;
1000 
1001  return BADCHISQ_MAT;
1002  }
1003  else //else try and find t0 and calculate amplitudes
1004  {
1005  //see the function comments for find_t0 for details
1006 
1007  shpar->t[0]=GetCsIt0();
1008 
1009  //if t0 is less than 0, return a T0FAIL
1010  if(shpar->t[0]<=0)
1011  {
1012  //printf("t0 less than/equal to 0\n");
1013  shpar->chisq=BADCHISQ_T0;
1014  shpar->ndf=1;
1015  return BADCHISQ_T0;
1016  }
1017 
1018  //calculate amplitudes
1019  shpar->am[0]=lineq_solution[0];
1020 
1021  for(int i=1;i<lineq_dim;i++)
1022  {
1023  tau=GetCsITau(i);
1024  shpar->am[i]=lineq_solution[i]*exp(-shpar->t[0]/tau);
1025  }
1026  //done calculating amplitudes
1027 
1028  for(int i=0;i<lineq_dim;i++)
1029  shpar->chisq-=lineq_solution[i]*lineq_vector[i];
1030 
1031  if(shpar->chisq<0)
1032  {
1033  shpar->chisq=BADCHISQ_NEG;
1034  shpar->ndf=1;
1035  return BADCHISQ_NEG;
1036  }
1037 
1038  for(int i=2;i<lineq_dim;i++)
1039  shpar->am[i]*=-1;
1040 
1041  }
1042 
1043 
1044  shpar->type=dim-2;
1045 
1046  return shpar->chisq;

1151 double TPulseAnalyzer::GetCsIt0()
1152 {
1153 
1154  //printf("Calculating t0\n");
1155 
1156  /*************************************************************************
1157  This function calculates t0 given the matrix solution from the get_shape.
1158  In this case, the fit function is written as follows:
1159 
1160  f(t) = C + (Af+As)*exp(t0/tRC)*exp(-t/tRC) - Af*exp(t0/tF')*exp(-t/tF')
1161  - As*exp(t0/tS')*exp(-t/tS')
1162 
1163  This can be re-written as:
1164 
1165  f(t) = C' + alpha*exp(-t/tRC) + beta*exp(-t/tF') + gamma*exp(-t/tS')
1166 
1167  Where:
1168  C = C'
1169  alpha = (Af+As)*exp(t0/tRC)
1170  beta = -Af*exp(t0/tF')
1171  gamma = -As*exp(t0/tS')
1172 
1173  Ignoring the constant, we have:
1174 
1175  f'(t0) = alpha*exp(-t0/tRC) + beta*exp(-t0/tF') + gamma*exp(-t0/tS') = 0
1176 
1177  For t<t0, f'(t)< 0, and for t>t0, f'(t)>0. This function finds the
1178  intersection of f'(t) and 0 by linear interpolation from these endpoints.
1179  *************************************************************************/
1180 
1181  double fa,fb,fc; //value of the fit function at points a,b,c
1182  double ta,tb,tc=0; //corresponding time (x-)axis values
1183  double slope; //linear interpolation slope
1184  double delta; //checks how close is the interpolated f(t0) is to 0
1185  double tau;
1186  int i;
1187 
1188  ta=wpar->baseline_range;
1189  //ta=wpar->temin;
1190  fa=0.;
1191 
1192  //t0 must be between the baseline and the max
1193  //calculates fit value (no constant) at the end of the baseline range
1194  //this is the t<t0 point
1195  for(i=1;i<lineq_dim;i++)
1196  {
1197  tau=GetCsITau(i);
1198  //printf("Tau @ 1117:\t%f\n",tau);
1199  //printf("i %d tau %f\n",i,tau);
1200  //getc(stdin);
1201  fa+=lineq_solution[i]*exp(-ta/tau);
1202  }
1203 
1204  tb=wpar->tmax;
1205  //tb=wpar->temax;
1206  fb=0.;
1207 
1208  //calculates fit value (no constant) at tmax
1209  //this is the t>t0 point
1210  for(i=1;i<lineq_dim;i++)
1211  {
1212  tau=GetCsITau(i);
1213  //printf("Tau @ 1131:\t%f\n",tau);
1214  fb+=lineq_solution[i]*exp(-tb/tau);
1215  }
1216 
1217  delta=1;
1218 
1219  if( (fa<0) && (fb>0) )
1220  {
1221  //keep the interpolation going until you get below epsilon
1222  /* |f(t0) - 0| = |f(t0)|< epsilon */
1223  while(delta>EPS)
1224  {
1225  slope=-fa/(fb-fa); //interpolation slope for dependent variable t
1226 
1227  //"reasonable" interpolation slopes
1228  if(slope>0.99)
1229  slope=0.99;
1230 
1231  if(slope<0.01)
1232  slope=0.01;
1233  //its pretty harmless computationally
1234 
1235  //tc is the estimate for t0
1236  tc=ta+slope*(tb-ta);
1237  fc=0.;
1238  for(i=1;i<lineq_dim;i++)
1239  {
1240  tau=GetCsITau(i);
1241  fc+=lineq_solution[i]*exp(-tc/tau);
1242  }
1243 
1244  //really should have this, just to be safe
1245  if(fc==0)
1246  {
1247  return tc;
1248  }
1249 
1250  else if(fc>0)
1251  {
1252  tb=tc;
1253  fb=fc;
1254  }
1255  else
1256  {
1257  ta=tc;
1258  fa=fc;
1259  }
1260  delta=fabs(fc);
1261  }
1262  }
1263  else
1264  {
1265  return -1;
1266  }
1267 
1268  //printf("tc:\t%f\n",tc);
1269 
1270  //set wpar->t0 here
1271  wpar->t0=tc;
1272
