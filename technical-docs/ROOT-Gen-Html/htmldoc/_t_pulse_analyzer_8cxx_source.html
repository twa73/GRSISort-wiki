101 // solve the determinant of the currently stored copy_matrix for dimentions m
102 long double TPulseAnalyzer::determinant(int m)
103 {
104  long double s;
105  if(m == 1) {
106  return copy_matrix[0][0];
107  }
108  if(copy_matrix[m - 1][m - 1] == 0.) {
109  int j = m - 1;
110  while(copy_matrix[m - 1][j] == 0 && j >= 0) {
111  j--;
112  }
113  if(j < 0) {
114  return 0.;
115  }
116  for(int i = 0; i < m; i++) {
117  s = copy_matrix[i][m - 1];
118  copy_matrix[i][m - 1] = copy_matrix[i][j];
119  copy_matrix[i][j] = s;
120  }
121  }
122  for(int j = m - 2; j >= 0; j--) {
123  for(int i = 0; i < m; i++) {
124  copy_matrix[i][j] -= copy_matrix[i][m - 1] / copy_matrix[m - 1][m - 1] * copy_matrix[m - 1][j];
125  }
126  }
127  return copy_matrix[m - 1][m - 1] * determinant(m - 1);
128 }

1016  sum = -(static_cast<double>(q)) / tau + log(1. - exp(-(static_cast<double>(d - q)) / tau));
1017  sum -= log(1. - exp(-1. / tau));

1019  lineq_matrix[j][i] = exp(sum);

1048  lineq_matrix[0][0] += 1;

1225

1227  - As*exp(t0/tS')*exp(-t/tS')

1230

1232

1234 C = C'

1240

1242 
1243 For t<t0, f'(t)< 0, and for t>t0, f'(t)>0. This function finds the
1244 intersection of f'(t) and 0 by linear interpolation from these endpoints.
