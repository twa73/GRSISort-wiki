60 ///
61 /// Remote files can be accessed using these special file names:
62 /// - pipein://command - read data produced by given command, see examples below
63 /// - ssh://username\@hostname/path/file.mid - read remote file through an ssh pipe
64 /// - ssh://username\@hostname/path/file.mid.gz and file.mid.bz2 - same for compressed files
65 /// - dccp://path/file.mid (also file.mid.gz and file.mid.bz2) - read data from dcache, requires dccp in the PATH
66 ///
67 /// Examples:
68 /// - ./event_dump.exe /ladd/data9/t2km11/data/run02696.mid.gz - read normal compressed file
69 /// - ./event_dump.exe ssh://ladd09//ladd/data9/t2km11/data/run02696.mid.gz - read compressed file through ssh to ladd09
70 /// (note double "/")
71 /// - ./event_dump.exe pipein://"cat /ladd/data9/t2km11/data/run02696.mid.gz | gzip -dc" - read data piped from a
72 /// command or script (note quotes)
73 /// - ./event_dump.exe pipein://"gzip -dc /ladd/data9/t2km11/data/run02696.mid.gz" - another way to read compressed
74 /// files
75 /// - ./event_dump.exe dccp:///pnfs/triumf.ca/data/t2km11/aug2008/run02837.mid.gz - read file directly from a dcache
76 /// pool (note triple "/")
77 ///
78 /// \param[in] filename The file to open.
79 /// \returns "true" for succes, "false" for error, use GetLastError() to see why
80 bool TLstFile::Open(const char* filename)
81 {
82  fFilename = filename;
83  int32_t headerSize = 0; // Count the number of bytes in the header
84 
85  //int32_t* boardHeaders = new int32_t[nbBoards]
86 
87  try {
88  std::ifstream in(GetFilename(), std::ifstream::in | std::ifstream::binary);
89  in.seekg(0, std::ifstream::end);
90  if(in.tellg() < 0) {
91  std::cout<<R"(Failed to open ")"<<GetFilename()<<"/"<<fFilename<<R"("!)"<<std::endl;
92  return false;
93  }
94  fFileSize = in.tellg();
95 
96  // Read Header Information
97  in.seekg(0, std::ifstream::beg);
98 
99  in.read(reinterpret_cast<char *>(&fVersion), sizeof(int32_t));
100  in.read(reinterpret_cast<char *>(&fTimeBase), sizeof(int32_t));
101  in.read(reinterpret_cast<char *>(&fnbEvents), sizeof(int32_t));
102  in.read(reinterpret_cast<char *>(&fnbBoards), sizeof(int32_t));
103  headerSize += 4*4; // 4 chucks of 4 Bytes
104 
105  // Read Board Headers

107  in.read(reinterpret_cast<char *>(fBoardHeaders), fnbBoards * sizeof(uint32_t));
108  headerSize += 4*fnbBoards;
109 
110  // Read rest of bytes into event buffer
111  fReadBuffer.resize(fFileSize - headerSize);
112  in.seekg(headerSize, std::ifstream::beg);
113  in.read(fReadBuffer.data(), fFileSize);
114  in.close();
115 
116  } catch(std::exception& e) {
117  std::cout<<"Caught "<<e.what()<<std::endl;
118  }

171 int TLstFile::GetRunNumber()
172 {
173  // Parse the run number from the current TMidasFile. This assumes a format of
174  // run#####_###.lst or run#####.lst.
175  if(fFilename.length() == 0) {
176  return 0;
177  }
178  std::size_t foundslash = fFilename.rfind('/');
179  std::size_t found = fFilename.rfind(".lst");
180  if(found == std::string::npos) {
181  return 0;
182  }
183  std::size_t found2 = fFilename.rfind('-');
184  if((found2 < foundslash && foundslash != std::string::npos) || found2 == std::string::npos) {
185  found2 = fFilename.rfind('_');
186  }
187  // printf("found 2 = %i\n",found2);
188  if(found2 < foundslash && foundslash != std::string::npos) {
189  found2 = std::string::npos;
190  }
191  std::string temp;
192  if(found2 == std::string::npos || fFilename.compare(found2 + 4, 4, ".lst") != 0) {
193  temp = fFilename.substr(found - 5, 5);
194  } else {
195  temp = fFilename.substr(found - 9, 5);
196  }
197  // printf(" %s \t %i \n",temp.c_str(),atoi(temp.c_str()));
198  return atoi(temp.c_str());
199 }
