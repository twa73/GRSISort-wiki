461 bool TNucleus::LoadTransitionFile()
462 {
463  if(TransitionList.GetSize() != 0) {
464  return false;
465  }
466  std::string filename;
467  filename = std::string(getenv("GRSISYS")) + "/libraries/TAnalysis/SourceData/";
468  std::string symbol = GetSymbol();
469  std::transform(symbol.begin(), symbol.end(), symbol.begin(), ::tolower);
470  filename.append(symbol);
471  filename.append(std::to_string(GetA()));
472  filename.append(".sou");
473  std::ifstream transfile;
474  transfile.open(filename.c_str());
475  if(!transfile.is_open()) {
476  printf("failed to open source file: %s\n", filename.c_str());
477  return false;
478  }
479  // printf("found %s\n",filename.c_str());
480 
481  std::string line;
482 
483  while(!getline(transfile, line).fail()) {
484  // printf("%i\t%s\n",counter++,line.c_str());
485  if(line.compare(0, 2, "//") == 0) {
486  continue;
487  }
488  if(line.compare(0, 1, "#") == 0) {
489  continue;
490  }
491  double temp;
492  auto* tran = new TTransition;
493  std::stringstream ss(line);
494  int counter = 0;
495  while(!(ss >> temp).fail()) {
496  counter++;
497  if(counter == 1) {
498  tran->SetEnergy(temp);
499  } else if(counter == 2) {
500  tran->SetEnergyUncertainty(temp);
501  } else if(counter == 3) {
502  tran->SetIntensity(temp);
503  } else if(counter == 4) {
504  tran->SetIntensityUncertainty(temp);
505  } else {
506  break;
507  }
508  }
509  AddTransition(tran);
510  }
511 
512  return true;
513 }
