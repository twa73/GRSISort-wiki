38 void TReaction::InitReaction(){
39 
40 // An initializing function that sets the energies and momenta of the beam and target in the lab and CM frame,
41 // as well as a few basic calculations.
42 
43  fTLab[0] = fTBeam; // target kinetic energy is ebeam
44  fELab[0] = fTLab[0] + fM[0]; // total E of beam in lab
45  fPLab[0] = sqrt(pow(fTLab[0],2)+2*fTLab[0]*fM[0]); // momentum of beam in lab
46  fVLab[0] = fPLab[0]/fELab[0]; // velocity of beam in lab
47  fGLab[0] = 1/sqrt(1-pow(fVLab[0],2)); // gamma factor of beam
48 
49  fTLab[1] = 0; // target kinetic energy is always 0
50  fELab[1] = fM[1]; // total E of target in lab
51  fPLab[1] = 0; // momentum of target in lab
52  fVLab[1] = 0; // velocity of target in lab
53  fGLab[1] = 1; // gamma factor of target
54 
55  fS = pow(fM[0],2)+pow(fM[1],2)+2*fELab[0]*fM[1]; // fELab[1] = fM[1]
56  fInvariantMass = sqrt(fS); // √s
57 
58  // CM motion
59  fCmE = fInvariantMass;
60  fCmTi = fCmE - fM[0] - fM[1];
61  fCmTf = fCmTi + fQVal;
62  fCmV = fPLab[0]/(fELab[0]+fM[1]);
63  fCmP = fCmV*fCmE;
64  fCmG = 1/sqrt(1-pow(fCmV,2));
65 
66  // take care of particles in CM frame using the reaction excitation energy
67  SetCmFrame(fExc);
68 
69  // options to make graphs draw nicely
70  gStyle->SetTitleYOffset(1.5);
71  gStyle->SetTitleXOffset(1.2);
72  gStyle->SetDrawOption("AC");
73 
74 }

83 void TReaction::SetCmFrame(double exc){
84 
85  // particles in CM frame
86  fPCm[0] = sqrt((fS-pow(fM[0]-fM[1],2)) * (fS-pow(fM[0]+fM[1],2)))/(2*sqrt(fS));
87  fPCm[1] = fPCm[0];
88  fPCm[2] = sqrt((fS-pow(exc+fM[2]-fM[3],2)) * (fS-pow(exc+fM[2]+fM[3],2)))/(2*sqrt(fS));
89  fPCm[3] = fPCm[2];
90 
91  for(int i=0; i<4; i++){
92  fECm[i] = sqrt(pow(fM[i],2)+pow(fPCm[i],2));
93  fTCm[i] = fECm[i]-fM[i];
94  fVCm[i] = fPCm[i]/fECm[i];
95  fGCm[i] = 1/sqrt(1-pow(fVCm[i],2));
96 
97  // max lab frame theta
98  if(i<2)
99  fThetaMax[i] = 0;
100  else{
101  double val = fPCm[i]/(fM[i]*fCmV*fCmG);
102  if(val<1)
103  fThetaMax[i] = asin(val);
104  else if(val<1.001) // catches elastic channels with small numerical rounding errors
105  fThetaMax[i] = PI/2;
106  else
107  fThetaMax[i] = PI;
108  }
109  }
110 }

210 double TReaction::ConvertThetaLabToCm(double theta_lab, int part){
211 
212  if(theta_lab>fThetaMax[part])
213  theta_lab = fThetaMax[part];
214 
215 // Uses the particle velocity in the CM frame, which makes it more complex
216  double gtan2 = pow(tan(theta_lab)*fCmG,2);
217  double x = fCmV/fVCm[part];
218  double expr = sqrt(1+gtan2*(1-pow(x,2)));
219  double theta_cm;
220 
221  // deals with double valued thetas in lab frame
222  if(tan(theta_lab)>=0)
223  theta_cm = acos((-x*gtan2 + expr)/(1 + gtan2));
224  else
225  theta_cm = acos((-x*gtan2 - expr)/(1 + gtan2));
226 
227  if(fInverse)
228  theta_cm = PI - theta_cm;
229 
230  if(part==3)
231  theta_cm = -theta_cm;
232 
233  return theta_cm;
234 }
