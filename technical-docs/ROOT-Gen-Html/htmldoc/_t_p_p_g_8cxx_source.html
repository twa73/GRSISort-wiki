131  /// Gets the last time that a status was given. If the EPpgPattern kJunk is passed, the
132  /// current status at the time "time" is looked for.

147  } else {

252  numberOfOffsets[(ppgIt->second->GetTimeStamp()) % 1000]++; // let's assume our offset is less than 10 us
253  switch(ppgIt->second->GetNewPPG()) {
254  case EPpgPattern::kBackground:

260  numberOfStateLengths[1][diff]++;

266  case EPpgPattern::kTapeMove:

269  break;
270  default: break;
271  }

278  cycleLength = numberOfCycleLength.first;
279  }

286  counter = it->second;
287  stateLength[i] = it->first;

291  counter = 0;

293  for(auto& numberOfOffset : numberOfOffsets) {

298  }

318  }

333 bool TPPG::OdbMatchesData(bool verbose)
334 {
335  if(!CalculateCycleFromData(verbose)) return false;
336 
337  // we got a cycle from the data, so we can compare it
338  if(fPPGCodes.size() != fOdbPPGCodes.size()) {
339  if(verbose) {
340  std::cout<<"Got: "<<fPPGCodes.size()<<" states from data, and "<<fOdbPPGCodes.size()<<" states from ODB"<<std::endl;
341  }
342  return false;
343  }
344  for(size_t i = 0; i < fPPGCodes.size(); ++i) {
345  // we only check the lowest nibble of the code, as the ODB code quite often is of the form 0xc00x (x = 1, 2, 4, or 8)
346  // and we need to convert the data durations (in ns) to the ODB duration (in microseconds)
347  if((fPPGCodes.at(i)&0xf) != (fOdbPPGCodes.at(i)&0xf) || static_cast<int>(fDurations.at(i)/1000) != fOdbDurations.at(i)) {
348  if(verbose) {
349  std::cout<<i<<": Mismatch between lowest nibble of codes ("<<hex(fPPGCodes.at(i),4)<<", "<<hex(fOdbPPGCodes.at(i),4)<<") or durations ("<<fDurations.at(i)<<", "<<fOdbDurations.at(i)<<")"<<std::endl;
350  }
351  return false;
352  }
353  }
354  // if we got here without finding anything wrong the odb matches the data
355  return true;

373 bool TPPG::CalculateCycleFromData(bool verbose)
374 {
375  /// Calculate the cycle from the data and store it.
376  if(fCycleSet) return true;
377 
378  // loop over all data and count how often we get each time difference between one state and the previous state
379  std::map<ULong64_t, int> numberOfStateLengths[4]; // to calculate the length of each state (tape move, background, beam on, and decay)
380  ULong64_t diff;
381  for(PPGMap_t::iterator ppgIt = MapBegin(); ppgIt != MapEnd(); ++ppgIt) {
382  switch(ppgIt->second->GetNewPPG()) {
383  case EPpgPattern::kBackground:
384  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kTapeMove);
385  numberOfStateLengths[0][diff]++;
386  break;
387  case EPpgPattern::kBeamOn:
388  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kBackground);
389  numberOfStateLengths[1][diff]++;
390  break;
391  case EPpgPattern::kDecay:
392  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kBeamOn);
393  numberOfStateLengths[2][diff]++;
394  break;
395  case EPpgPattern::kTapeMove:
396  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kDecay);
397  numberOfStateLengths[3][diff]++;
398  break;
399  default: break;
400  }
401  }
402  // set the durations to the one that occured the most often
403  // (codes are already set)
404  for(int i = 0; i < 4; ++i) {
405  int counter = 0;
406  if(verbose) std::cout<<"state "<<hex(fPPGCodes[i],4)<<" checking "<<numberOfStateLengths[i].size()<<" lengths:";
407  for(auto it : numberOfStateLengths[i]) {
408  if(it.second > counter) {
409  if(verbose) std::cout<<" "<<it.second<<">"<<counter<<" => fDurations["<<i<<"] = "<<it.first;
410  counter = it.second;
411  fDurations[i] = it.first;
412  } else if(verbose) {
413  std::cout<<" "<<it.second<<"<"<<counter<<" => fDurations["<<i<<"] != "<<it.first;
414  }
415  }
416  if(verbose) std::cout<<std::endl;
417  }
418 
419  fCycleSet = true;
420 
421  return true;
422 }

478 bool TPPG::Correct(bool verbose)
479 {
480  /// Correct the map of PPG events by checking that we always have a PPG event with the same status that occured
481  /// fCycleLength ago.
482 
483  // we use the cycle time to correct our ppg, so make sure it's set (GetCycleLength calculates and sets fCycleLength
484  // only if it's not already set)
485  GetCycleLength();
486 
487  if(verbose) {
488  // we can now use fNumberOfCycleLengths to see how many cycle lengths we have that are wrong
489  for(auto& fNumberOfCycleLength : fNumberOfCycleLengths) {
490  if(fNumberOfCycleLength.first < fCycleLength) {
491  continue;
492  }
493  if(fNumberOfCycleLength.first != fCycleLength) {
494  std::cout<<"Found "<<fNumberOfCycleLength.second<<" wrong cycle length(s) of "<<fNumberOfCycleLength.first<<" (correct is "<<fCycleLength<<")."<<std::endl;
495  }
496  }
497  }
498 
499  // loop over all ppg data, check that the last ppg data of the same type was fCycleLength ago
500  // if it's not, enter ppg data at that time (multiple times?)
501  for(auto it = MapBegin(); it != MapEnd(); it++) {
502  // skip the first cycle
503  if((*it).first < fCycleLength) {
504  continue;
505  }
506  // get time difference to previous ppg with same status
507  ULong64_t diff = (*it).second->GetTimeStamp() - GetLastStatusTime((*it).second->GetTimeStamp());
508  if(diff != fCycleLength) {
509  if(verbose) {
510  std::cout<<std::distance(MapBegin(), it)<<": found missing ppg at time "<<(*it).first<<" ("<<diff<<" != "<<fCycleLength<<")"<<std::endl;
511  }
512  // check that the previous ppg with the same status is a multiple of fCycleLength ago and that no other ppg is
513  // in the map that was fCycleLength ago
514  if(diff % fCycleLength != 0) {
515  if(verbose) {
516  std::cout<<DRED<<"PPG is messed up, cycle length is "<<fCycleLength<<", but the previous event with the same status was "<<diff<<" ago!"<<RESET_COLOR<<std::endl;
517  }
518  continue;
519  return false;
520  }
521  auto prev = fPPGStatusMap->find((*it).first - fCycleLength);
522  if(prev != fPPGStatusMap->end()) {
523  // check if the status of the current ppg and the old status of the ppg following the one fCycleLength ago
524  // are the same
525  // if they are, we change the status of the one fCycleLength ago to match the current status
526  if(it->second->GetNewPPG() == (++prev)->second->GetOldPPG()) {
527  (--prev)->second->SetNewPPG(it->second->GetNewPPG());
528  } else if(verbose) {
529  std::cout<<DBLUE<<"PPG at "<<(*it).first - fCycleLength<<" already exist with status "<<hex(static_cast<std::underlying_type<EPpgPattern>::type>(prev->second->GetNewPPG()))<<" (current status is "<<hex(static_cast<std::underlying_type<EPpgPattern>::type>(it->second->GetNewPPG()))<<")."<<RESET_COLOR<<std::endl;
530  }
531  continue;
532  }
533  // copy the current ppg data and correct it's time before inserting it into the map
534  auto* new_data = new TPPGData(*((*it).second));
535  ULong64_t new_ts = (*it).first - fCycleLength;
536  new_data->SetHighTimeStamp(new_ts >> 28);
537  new_data->SetLowTimeStamp(new_ts & 0x0fffffff);
538  if(verbose) {
539  std::cout<<"inserting new ppg data at "<<new_data->GetTimeStamp()<<std::endl;
540  }
541  it = fPPGStatusMap->insert(std::make_pair(new_data->GetTimeStamp(), new_data)).first;
542  --it;
543  }
544  }
545 
546  // now we reset fCycleLength and calculate it again
547  fCycleLength = 0;
548  fNumberOfCycleLengths.clear();
549  GetCycleLength();
550 
551  if(verbose) {
552  // we can now use fNumberOfCycleLengths to see how many cycle lengths we have that are wrong
553  for(auto& fNumberOfCycleLength : fNumberOfCycleLengths) {
554  if(fNumberOfCycleLength.first < fCycleLength) {
555  continue;
556  }
557  if(fNumberOfCycleLength.first != fCycleLength) {
558  std::cout<<"Found "<<fNumberOfCycleLength.second<<" wrong cycle length(s) of "<<fNumberOfCycleLength.first<<" (correct is "<<fCycleLength<<")."<<std::endl;
559  }
560  }
561  }
562 
563  return true;
564 }

625 ULong64_t TPPG::GetCycleLength()
626 {
627  /// Calculates cycle length from the most often occuring time difference between
628  /// two PPG events with the same code. The result is also stored in fCycleLength
629  /// (which prevents a second calculation of it the next time this function is called)
630  if(fCycleLength == 0) {
631  PPGMap_t::iterator ppgit;
632  for(ppgit = MapBegin(); ppgit != MapEnd(); ++ppgit) {
633  ULong64_t diff = ppgit->second->GetTimeStamp() - GetLastStatusTime(ppgit->second->GetTimeStamp());
634  fNumberOfCycleLengths[diff]++;
635  }
636  int counter = 0;
637  for(auto& fNumberOfCycleLength : fNumberOfCycleLengths) {
638  if(fNumberOfCycleLength.second > counter) {
639  counter = fNumberOfCycleLength.second;
640  fCycleLength = fNumberOfCycleLength.first;
641  }
642  }
643  }
644 
645  return fCycleLength;
646 }
