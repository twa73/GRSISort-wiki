130 ULong64_t TPPG::GetLastStatusTime(ULong64_t time,ppg_pattern pat,bool exact_flag) const {
131  ///Gets the last time that a status was given. If the ppg_pattern kJunk is passed, the
132  ///current status at the time "time" is looked for. If exact_flag is false, the bits of "pat"
133  ///are looked for and ignore the rest of the bits in the sotred statuses. If "exact_flag"
134  ///is true, the entire ppg pattern "pat" must be met.
135  if(MapIsEmpty()) {
136  printf("Empty\n");
137  return 0;
138  }
139 
140  PPGMap_t::iterator curppg_it = --(fPPGStatusMap->upper_bound(time));
141  PPGMap_t::iterator ppg_it;
142  if(pat == kJunk) {
143  for(ppg_it = curppg_it; ppg_it != fPPGStatusMap->begin(); --ppg_it) {
144  if(curppg_it->second->GetNewPPG() == ppg_it->second->GetNewPPG() && curppg_it != ppg_it ) {
145  return ppg_it->first;
146  }
147  }
148  } else {
149  for(ppg_it = curppg_it; ppg_it != fPPGStatusMap->begin(); --ppg_it) {
150  if(exact_flag) {
151  if(pat == ppg_it->second->GetNewPPG()) {
152  return ppg_it->first;
153  }
154  } else{
155  // printf("pat %x, ppg_it->first %lu, ppg_it->second->GetNewPPG() %x\n",pat,ppg_it->first,ppg_it->second->GetNewPPG());
156  if(pat & ppg_it->second->GetNewPPG()) {
157  return ppg_it->first;
158  }
159  }
160  }
161  }
162  //printf("No previous status\n");
163  return 0;
164 }

333 bool TPPG::Correct(bool verbose) {
334  ///Correct the map of PPG events by checking that we always have a PPG event with the same status that occured fCycleLength ago.
335 
336  //we use the cycle time to correct our ppg, so make sure it's set (GetCycleLength calculates and sets fCycleLength only if it's not already set)
337  GetCycleLength();
338 
339  if(verbose) {
340  //we can now use fNumberOfCycleLengths to see how many cycle lengths we have that are wrong
341  for(auto it = fNumberOfCycleLengths.begin(); it != fNumberOfCycleLengths.end(); ++it) {
342  if((*it).first < fCycleLength) {
343  continue;
344  }
345  if((*it).first != fCycleLength) {
346  printf("Found %d wrong cycle length(s) of %lld (correct is %lld).\n", (*it).second, (*it).first, fCycleLength);
347  }
348  }
349  }
350 
351  //loop over all ppg data, check that the last ppg data of the same type was fCycleLength ago
352  //if it's not, enter ppg data at that time (multiple times?)
353  for(auto it = MapBegin(); it != MapEnd(); it++) {
354  //skip the first cycle
355  if((*it).first < fCycleLength) {
356  continue;
357  }
358  //get time difference to previous ppg with same status
359  ULong64_t diff = (*it).second->GetTimeStamp() - GetLastStatusTime((*it).second->GetTimeStamp());
360  if(diff != fCycleLength) {
361  if(verbose) {
362  printf("%ld: found missing ppg at time %lu (%lld != %lld)\n", std::distance(MapBegin(),it),(*it).first, diff, fCycleLength);
363  }
364  //check that the previous ppg with the same status is a multiple of fCycleLength ago and that no other ppg is in the map that was fCycleLength ago
365  if(diff%fCycleLength != 0) {
366  if(verbose) {
367  printf(DRED "PPG is messed up, cycle length is %lld, but the previous event with the same status was %lld ago!" RESET_COLOR "\n", fCycleLength, diff);
368  }
369  continue;
370  return false;
371  }
372  auto prev = fPPGStatusMap->find((*it).first - fCycleLength);
373  if(prev != fPPGStatusMap->end()) {
374  //check if the status of the current ppg and the old status of the ppg following the one fCycleLength ago are the same
375  //if they are, we change the status of the one fCycleLength ago to match the current status
376  if(it->second->GetNewPPG() == (++prev)->second->GetOldPPG()) {
377  (--prev)->second->SetNewPPG(it->second->GetNewPPG());
378  } else if(verbose) {
379  printf(DBLUE "PPG at %lld already exist with status 0x%x (current status is 0x%x)." RESET_COLOR "\n", (*it).first - fCycleLength, prev->second->GetNewPPG(), it->second->GetNewPPG());
380  }
381  continue;
382  }
383  //copy the current ppg data and correct it's time before inserting it into the map
384  TPPGData* new_data = new TPPGData(*((*it).second));
385  ULong64_t new_ts = (*it).first - fCycleLength;
386  new_data->SetHighTimeStamp(new_ts>>28);
387  new_data->SetLowTimeStamp(new_ts&0x0fffffff);
388  if(verbose) {
389  printf("inserting new ppg data at %lld\n",new_data->GetTimeStamp());
390  }
391  it = fPPGStatusMap->insert(std::make_pair(new_data->GetTimeStamp(),new_data)).first;
392  --it;
393  }
394  }
395 
396  //now we reset fCycleLength and calculate it again
397  fCycleLength = 0;
398  fNumberOfCycleLengths.clear();
399  GetCycleLength();
400 
401  if(verbose) {
402  //we can now use fNumberOfCycleLengths to see how many cycle lengths we have that are wrong
403  for(auto it = fNumberOfCycleLengths.begin(); it != fNumberOfCycleLengths.end(); ++it) {
