208 void TPPG::Print(Option_t* opt) const
209 {
210  if(fOdbPPGCodes.size() != fOdbDurations.size()) {
211  std::cout<<"Mismatch between number of ppg codes ("<<fOdbPPGCodes.size()<<") and durations ("
212  <<fOdbDurations.size()<<")"<<std::endl;
213  } else {
214  if(fOdbPPGCodes.empty()) {
215  std::cout<<"No ODB cycle read!"<<std::endl;
216  } else {
217  std::cout<<"ODB cycle:"<<std::endl<<"Code Duration"<<std::endl;
218  for(size_t i = 0; i < fOdbPPGCodes.size(); ++i) {
219  std::cout<<"0x"<<std::hex<<std::setw(4)<<fOdbPPGCodes[i]<<std::dec<<" "<<fOdbDurations[i]
220  <<std::endl;
221  }
222  }
223  }
224  if(MapIsEmpty()) {
225  printf("Empty\n");
226  return;
227  }
228  if(TString(opt).Contains("all", TString::ECaseCompare::kIgnoreCase)) {
229  // print all ppg data
230  PPGMap_t::iterator ppgit;
231  printf("*****************************\n");
232  printf(" PPG STATUS \n");
233  printf("*****************************\n");
234  for(ppgit = MapBegin(); ppgit != MapEnd(); ppgit++) {
235  ppgit->second->Print();
236  }
237  return;
238  }
239  // print only an overview of the ppg
240  // can't call non-const GetCycleLength here, so we do the calculation with local variables here
241  PPGMap_t::iterator ppgIt;
242  std::map<EPpgPattern, int> status; // to calculate how often each different status occured
243  std::map<ULong64_t, int> numberOfCycleLengths; // to calculate the length of the whole cycle
244  std::map<ULong64_t, int>
245  numberOfStateLengths[4]; // to calculate the length of each state (tape move, background, beam on, and decay)
246  std::map<int, int> numberOfOffsets; // to calculate the offset on each timestamp
247  for(ppgIt = MapBegin(); ppgIt != MapEnd(); ++ppgIt) {
248  status[ppgIt->second->GetNewPPG()]++;
249  ULong64_t diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp());
250  numberOfCycleLengths[diff]++;
251  numberOfOffsets[(ppgIt->second->GetTimeStamp()) % 1000]++; // let's assume our offset is less than 10 us
252  switch(ppgIt->second->GetNewPPG()) {
253  case EPpgPattern::kBackground:
254  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kTapeMove);
255  numberOfStateLengths[0][diff]++;
256  break;
257  case EPpgPattern::kBeamOn:
258  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kBackground);
259  numberOfStateLengths[1][diff]++;
260  break;
261  case EPpgPattern::kDecay:
262  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kBeamOn);
263  numberOfStateLengths[2][diff]++;
264  break;
265  case EPpgPattern::kTapeMove:
266  diff = ppgIt->second->GetTimeStamp() - GetLastStatusTime(ppgIt->second->GetTimeStamp(), EPpgPattern::kDecay);
267  numberOfStateLengths[3][diff]++;
268  break;
269  default: break;
270  }
271  }
272  int counter = 0;
273  ULong64_t cycleLength = 0;
274  for(auto& numberOfCycleLength : numberOfCycleLengths) {
275  if(numberOfCycleLength.second > counter) {
276  counter = numberOfCycleLength.second;
277  cycleLength = numberOfCycleLength.first;
278  }
279  }
280  ULong64_t stateLength[4] = {0, 0, 0, 0};
281  for(int i = 0; i < 4; ++i) {
282  counter = 0;
283  for(auto it = numberOfStateLengths[i].begin(); it != numberOfStateLengths[i].end(); ++it) {
284  if(it->second > counter) {
285  counter = it->second;
286  stateLength[i] = it->first;
287  }
288  }
289  }
290  counter = 0;
291  int offset = 0;
292  for(auto& numberOfOffset : numberOfOffsets) {
293  if(numberOfOffset.second > counter) {
294  counter = numberOfOffset.second;
295  offset = numberOfOffset.first;
296  }
297  }
298 
299  // the print statement itself
300  printf("Cycle length is %lld in ns = %.3lf seconds.\n", cycleLength, cycleLength / 1e9);
301  printf("Cycle: %.3lf s tape move, %.3lf s background, %.3lf s beam on, and %.3lf s decay\n", stateLength[0] / 1e9,
302  stateLength[1] / 1e9, stateLength[2] / 1e9, stateLength[3] / 1e9);
303  printf("Offset is %d [ns]\n", offset);
304  printf("Got %ld PPG words:\n", fPPGStatusMap->size() - 1);
305  for(auto& statu : status) {
306  printf("\tfound status 0x%04x %d times\n", static_cast<std::underlying_type<EPpgPattern>::type>(statu.first), statu.second);
307  }
308 
309  if(TString(opt).Contains("missing", TString::ECaseCompare::kIgnoreCase)) {
310  // go through all expected ppg words
311  ULong64_t time = offset;
312  auto it = MapEnd();
313  --it;
314  ULong64_t lastTimeStamp = it->second->GetTimeStamp();
315  for(int cycle = 1; time < lastTimeStamp; ++cycle) {
316  if(GetLastStatusTime(time, EPpgPattern::kTapeMove) != time) {
317  printf("Missing tape move status at %12lld in %d. cycle, last tape move status came at %lld.\n", time, cycle,
318  GetLastStatusTime(time, EPpgPattern::kTapeMove));
319  }
320  if(GetLastStatusTime(time + cycleLength, EPpgPattern::kBackground) != time + stateLength[0]) {
321  printf("Missing background status at %12lld in %d. cycle, last background status came at %lld.\n",
322  time + stateLength[0], cycle, GetLastStatusTime(time + cycleLength, EPpgPattern::kBackground));
323  }
324  if(GetLastStatusTime(time + cycleLength, EPpgPattern::kBeamOn) != time + stateLength[0] + stateLength[1]) {
325  printf("Missing beam on status at %12lld in %d. cycle, last beam on status came at %lld.\n",
326  time + stateLength[0] + stateLength[1], cycle, GetLastStatusTime(time + cycleLength, EPpgPattern::kBeamOn));
327  }
328  if(GetLastStatusTime(time + cycleLength, EPpgPattern::kDecay) != time + stateLength[0] + stateLength[1] + stateLength[2]) {
329  printf("Missing decay status at %12lld in %d. cycle, last decay status came at %lld.\n",
330  time + stateLength[0] + stateLength[1] + stateLength[2], cycle,
331  GetLastStatusTime(time + cycleLength, EPpgPattern::kDecay));
332  }
333  time += cycleLength;
334  }
335  }

393 bool TPPG::Correct(bool verbose)
394 {
395  /// Correct the map of PPG events by checking that we always have a PPG event with the same status that occured
396  /// fCycleLength ago.
397 
398  // we use the cycle time to correct our ppg, so make sure it's set (GetCycleLength calculates and sets fCycleLength
399  // only if it's not already set)
400  GetCycleLength();
401 
402  if(verbose) {
403  // we can now use fNumberOfCycleLengths to see how many cycle lengths we have that are wrong
404  for(auto& fNumberOfCycleLength : fNumberOfCycleLengths) {
405  if(fNumberOfCycleLength.first < fCycleLength) {
406  continue;
407  }
408  if(fNumberOfCycleLength.first != fCycleLength) {
409  printf("Found %d wrong cycle length(s) of %lld (correct is %lld).\n", fNumberOfCycleLength.second,
410  fNumberOfCycleLength.first, fCycleLength);
411  }
412  }
413  }
414 
415  // loop over all ppg data, check that the last ppg data of the same type was fCycleLength ago
416  // if it's not, enter ppg data at that time (multiple times?)
417  for(auto it = MapBegin(); it != MapEnd(); it++) {
418  // skip the first cycle
419  if((*it).first < fCycleLength) {
420  continue;
421  }
422  // get time difference to previous ppg with same status
423  ULong64_t diff = (*it).second->GetTimeStamp() - GetLastStatusTime((*it).second->GetTimeStamp());
424  if(diff != fCycleLength) {
425  if(verbose) {
426  printf("%ld: found missing ppg at time %lu (%lld != %lld)\n", std::distance(MapBegin(), it), (*it).first,
427  diff, fCycleLength);
428  }
429  // check that the previous ppg with the same status is a multiple of fCycleLength ago and that no other ppg is
430  // in the map that was fCycleLength ago
431  if(diff % fCycleLength != 0) {
432  if(verbose) {
433  printf(DRED "PPG is messed up, cycle length is %lld, but the previous event with the same status was "
434  "%lld ago!" RESET_COLOR "\n",
435  fCycleLength, diff);
436  }
437  continue;
438  return false;
439  }
440  auto prev = fPPGStatusMap->find((*it).first - fCycleLength);
441  if(prev != fPPGStatusMap->end()) {
442  // check if the status of the current ppg and the old status of the ppg following the one fCycleLength ago
443  // are the same
444  // if they are, we change the status of the one fCycleLength ago to match the current status
445  if(it->second->GetNewPPG() == (++prev)->second->GetOldPPG()) {
446  (--prev)->second->SetNewPPG(it->second->GetNewPPG());
447  } else if(verbose) {
448  printf(DBLUE "PPG at %lld already exist with status 0x%x (current status is 0x%x)." RESET_COLOR "\n",
449  (*it).first - fCycleLength, static_cast<std::underlying_type<EPpgPattern>::type>(prev->second->GetNewPPG()),
450  static_cast<std::underlying_type<EPpgPattern>::type>(it->second->GetNewPPG()));
451  }
452  continue;
453  }
454  // copy the current ppg data and correct it's time before inserting it into the map
455  auto* new_data = new TPPGData(*((*it).second));
456  ULong64_t new_ts = (*it).first - fCycleLength;
457  new_data->SetHighTimeStamp(new_ts >> 28);
458  new_data->SetLowTimeStamp(new_ts & 0x0fffffff);
459  if(verbose) {
460  printf("inserting new ppg data at %lld\n", new_data->GetTimeStamp());
461  }
462  it = fPPGStatusMap->insert(std::make_pair(new_data->GetTimeStamp(), new_data)).first;
463  --it;
464  }
465  }
466 
467  // now we reset fCycleLength and calculate it again
468  fCycleLength = 0;
469  fNumberOfCycleLengths.clear();
470  GetCycleLength();
471 
472  if(verbose) {
473  // we can now use fNumberOfCycleLengths to see how many cycle lengths we have that are wrong
474  for(auto& fNumberOfCycleLength : fNumberOfCycleLengths) {
475  if(fNumberOfCycleLength.first < fCycleLength) {
476  continue;
477  }
478  if(fNumberOfCycleLength.first != fCycleLength) {
479  printf("Found %d wrong cycle length(s) of %lld (correct is %lld).\n", fNumberOfCycleLength.second,
480  fNumberOfCycleLength.first, fCycleLength);
481  }
482  }
483  }
484 
485  return true;
486 }
