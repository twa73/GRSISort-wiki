67 inline double DecGaus(double& x,double& sig,double& dec){return exp(x/dec)*TMath::Erfc(x/(sig*sqrt(2)) + sig/(dec*sqrt(2)));}

69 inline double DecGausCFD(double& x,double& sig,double& dec){return 0.5*(1 + exp((x/dec) + ((sig*sig)/(2*dec*dec)))*TMath::Erfc(((dec*x) + (sig*sig)) /(dec*sig*sqrt(2)))+TMath::Erf(x/(sig*sqrt(2))));}

71 inline double DecGausStep(double& x,double& sig,double& dec){return 1-DecGausCFD(x,sig,dec);}

73 inline double DecGausArea(double& sig,double& dec){return 2*dec*exp(-((sig*sig)/(2*dec*dec)));}

75 //An analytical approach to the X value of peak maximum, valid in the range sigma 0.1-10 decay 1-100
76 inline double DecGausMaxX(double& sig,double& dec){
77  double a=1.39065*pow(dec,0.785742);
78  a+=dec*-0.429206;
79  a+=dec*dec*0.000356727;
80  a+=dec*dec*dec*-3.08765e-8;
81 
82  double b=1.93679*pow(dec,0.920522);
83  b+=dec*-0.819016;
84  b+=dec*dec*-0.00515282;
85  b+=dec*dec*dec*4.87915e-05;
86  b+=dec*dec*dec*dec*-1.77445e-07;
87 
88  return (exp(-sig/a)-1)*b;

102 // Given the usual 0-1 sharing parameter returns the area of the combined peak
103 // Simply multiply by h_a
104 inline double CombArea(double& sig,double& dec,double& eta){
105  double decX_0=DecGausMaxX(sig,dec)*(1-eta);//The X value at which DecGaus is maximal
106  double yuni=UniGaus(decX_0,sig);//The Y value of UniGaus at that point
107  double xeta=eta/yuni;//UniGaus sharing fraction / UniGaus Y value
108  double zeta=(1-eta)/DecGaus(decX_0,sig,dec);// DecGaus sharing fraction / DecGaus Y value

173 inline double CombAreaTwo(double& sig1,double& sig2,double& TwoGausFrac,double& dec,double& eta){
174  double sigd=sig1*TwoGausFrac+sig2*(1-TwoGausFrac);
175  double decX_0=DecGausMaxX(sigd,dec)*(1-eta);
176  double xeta=eta/(UniGaus(decX_0,sig1)*TwoGausFrac+UniGaus(decX_0,sig2)*(1-TwoGausFrac));
177  double zeta=(1-eta)/DecGaus(decX_0,sigd,dec);
178  return (UniGausArea(sig1)*TwoGausFrac+UniGausArea(sig2)*(1-TwoGausFrac))*xeta + DecGausArea(sigd,dec)*zeta;
179 }

316  double operator() (double *x, double *p) {
317 
318  // DecGaus is not max @ 0 and does not have height = 1
319  // User p[gPeakSharing] specifies the fraction of the height that is UniGaus
320  // From this we calculate a new sharing parameter yeta & scaling factor hach such that:
321  // y = hach * ( yeta*UniGaus + (1-yeta)*DecGaus)
322  // Will have a max height ~1
323  // Thus heigh parameters and sharing parameters can be given in terms of simple observable height
324  // Much easier for fitting but more complicated in the functions.
325 
326  // Complex sharing parameters for the decay peak are calculated here to save multiplication of effort
327 
328 
329  double TwoGausFrac=p[gPeakSigmaC];
330  double sigd=p[gPeakSigma];
331  if(TestBit(k2Gaus))sigd*=(TwoGausFrac+p[gPeakSigmaB]*(1-TwoGausFrac));
332 
333  double a=DecGausMaxX(sigd,p[gPeakDecay])*(1-p[gPeakSharing]);
334  double X=x[0]+a;
335  double xeta;
336  if(TestBit(k2Gaus)){
337  double sB=p[gPeakSigma]*p[gPeakSigmaB];
338  xeta=p[gPeakSharing]/(UniGaus(a,p[gPeakSigma])*TwoGausFrac+UniGaus(a,sB)*(1-TwoGausFrac));
339  }else xeta=p[gPeakSharing]/UniGaus(a,p[gPeakSigma]);
340  double zeta=(1-p[gPeakSharing])/DecGaus(a,sigd,p[gPeakDecay]);
341  double yeta=xeta/(xeta+zeta);
342  double hach=xeta+zeta;
343 
344  // NOTE the following line removes
345  // complex sharing parameter normalisation behaviour
346  // Use OldCombArea and ignore "TrueCentroid"
347  //yeta=p[gPeakSharing];hach=1;X=x[0];
348 
349  // The latter functions take peak height but now peak height inputs may be ratio
350  // So first we convert.
351  double *P=p;
352  if(N>1){FixParameters(p,para,N,cBits);P=para;}
353 
354  double ret=0;
355  if(TestBit(kPeaks)){
356  if(TestBit(k2Gaus))ret+=hach*MulitPeakTwo(P,N,X,yeta);
357  else ret+=hach*MulitPeak(P,N,X,yeta);
358  }
359  //With new function it is possible to fix par[gPeakSigmaB]==1 and use MulitPeakTwo instead of MulitPeak
360 
361 
362  if(TestBit(kBack)){
363  ret+=P[gUltraPol0];
364  if(TestBit(kStep)){
365  if(TestBit(k2Gaus))ret+=P[gUltraStep]*MulitPeakBackUniTwo(P,N,X,yeta);
366  else ret+=P[gUltraStep]*MulitPeakBackUni(P,N,X,yeta);
367  }
368  if(TestBit(kPol2))ret+=x[0]*P[gUltraPol1]+x[0]*x[0]*P[gUltraOffsetOrPol2];
369  else ret+=(x[0]-P[gUltraOffsetOrPol2])*P[gUltraPol1];
370  }
371 
372  return ret;
373  }
