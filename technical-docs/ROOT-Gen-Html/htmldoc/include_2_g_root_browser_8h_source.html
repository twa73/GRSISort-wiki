60 /////////////////////////////////////////////////////////////////
61 ///
62 /// \class GRootBrowser
63 ///
64 /// This class creates a ROOT object browser, constitued by three main
65 /// tabs.
66 ///
67 /// All tabs can 'swallow' frames, thanks to the new method:
68 /// ExecPlugin(const char *name = 0, const char *fname = 0,
69 /// const char *cmd = 0, Int_t pos = kRight,
70 /// Int_t subpos = -1)
71 /// allowing to select plugins (can be a macro or a command)
72 /// to be executed, and where to embed the frame created by
73 /// the plugin (tab and tab element). Examples:
74 ///
75 /// create a new browser:
76 /// TBrowser b;
77 ///
78 /// create a new TCanvas in a new top right tab element:
79 /// b.ExecPlugin("Canvas", 0, "new TCanvas()");
80 ///
81 /// create a new top right tab element embedding the
82 /// TGMainFrame created by the macro 'myMacro.C':
83 /// b.ExecPlugin("MyPlugin", "myMacro.C");
84 ///
85 /// create a new bottom tab element embedding the
86 /// TGMainFrame created by the macro 'myMacro.C':
87 /// b.ExecPlugin("MyPlugin", "myMacro.C", 0, GRootBrowser::kBottom);
88 ///
89 /// this browser implementation can be selected via the env
90 /// 'Browser.Name' in .rootrc, (GRootBrowser or GRootBrowserLite)
91 /// the default being GRootBrowserLite (old browser)
92 /// a list of options (plugins) for the new GRootBrowser is also
93 /// specified via the env 'Browser.Options' in .rootrc, the default
94 /// being: FECI
95 /// Here is the list of available options:
96 /// F: File browser E: Text Editor H: HTML browser C: Canvas I: I/O
97 /// redirection P: Proof G: GL viewer
98 ///
99 /////////////////////////////////////////////////////////////////
