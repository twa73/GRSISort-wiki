151 int PeakSearch(TH1* hst, double sigma, double thresh,Option_t *opt) {
152  //TSpectrum spec(20);

154  TSpectrum::StaticSearch(hst,sigma,"Qnodraw",thresh);
155  TPolyMarker *pm = (TPolyMarker*)hst->GetListOfFunctions()->FindObject("TPolyMarker");
156  if (!pm)
157  {
158  printf("No TPolyMarker object in the list of functions for histogram %s\n",hst->GetName());
159  return 0;
160  }
161  TObjArray* testarray = (TObjArray*)hst->GetListOfFunctions()->FindObject("PeakLabels");
162  if (testarray)
163  {
164  hst->GetListOfFunctions()->Remove(testarray);
165  //delete testarray;
166  testarray->Delete(); //guarantees the text objects it points to are removed as well.
167  }
168 
169  TObjArray* array = new TObjArray();
170  array->SetName("PeakLabels");
171  int n = pm->GetN();
172  if(n<1)
173  return 0;
174  TText* text;
175  double *x = pm->GetX();
176  double *y = pm->GetY();
177  for (int i=0;i<n;i++)
178  {
179  text = new TText(x[i],y[i],Form("%.1f",x[i]));
180  text->SetTextSize(0.025);
181  text->SetTextAngle(90);
182  text->SetTextAlign(12);
183  text->SetTextFont(42);
184  text->SetTextColor(hst->GetLineColor());
185  array->Add(text);
186  }
187  hst->GetListOfFunctions()->Remove(pm);
188  //delete pm;
189  pm->Delete();
190  hst->GetListOfFunctions()->Add(array);
191  return n;
192 }
