76 void TFippsHit::Add(const TDetectorHit* hit)
77 {
78  const TFippsHit* fippsHit = dynamic_cast<const TFippsHit*>(hit);
79  if(fippsHit == nullptr) {
80  throw std::runtime_error("trying to add non-fipps hit to fipps hit!");
81  }
82  // add another griffin hit to this one (for addback),
83  // using the time and position information of the one with the higher energy
84  if(!CompareEnergy(this, fippsHit)) {
85  SetCfd(fippsHit->GetCfd());
86  SetTime(fippsHit->GetTime());
87  // SetPosition(fippsHit->GetPosition());
88  SetAddress(fippsHit->GetAddress());
89  }
90  SetEnergy(GetEnergy() + fippsHit->GetEnergy());
91  // this has to be done at the very end, otherwise GetEnergy() might not work
92  SetCharge(0);
93  // KValue is somewhate meaningless in addback, so I am using it as an indicator that a piledup hit was added-back RD
94  if(GetKValue() > fippsHit->GetKValue()) {
95  SetKValue(fippsHit->GetKValue());
96  }
97 }
