374 Double_t TSingleDecay::ActivityFunc(Double_t* dim, Double_t* par)
375 {
376  /// The general function for a decay chain
377  /// par[0] is the intensity
378  /// par[1*i] is the activity
379  Double_t result = 1.0;
380  UInt_t gencounter = 0;
381  Double_t tlow, thigh;
382  fTotalDecayFunc->GetRange(tlow, thigh);
383  Double_t t = dim[0] - tlow;
384  TSingleDecay* curDecay = this;
385  // Compute the first multiplication
386  while(curDecay != nullptr) {
387  ++gencounter;
388  // par[Generation] gets the decay rate for that decay since the parameters are stored
389  // as [intensity, decayrate1,decayrate2,...]
390  result *= par[curDecay->GetGeneration()];
391 
392  curDecay = curDecay->GetParentDecay();
393  }
394  if(gencounter != fGeneration) {
395  printf("We have Problems!\n");
396  return 0.0;
397  }
398  // Multiply by the initial intensity of the initial parent.
399  result *= par[0] / par[1];
400  // Now we need to deal with the second term
401  Double_t sum = 0.0;
402  curDecay = this;
403  while(curDecay != nullptr) {
404  Double_t denom = 1.0;
405  TSingleDecay* denomDecay = this;
406  while(denomDecay != nullptr) {
407  if(denomDecay != curDecay) {
408  // This term has problems if two or more rates are very similar. Will have to put a taylor expansion in here
409  // if this problem comes up. I believe the solution to this also depends on the number of nuclei in the
410  // chain
411  // that have similar rates. I think either of these issues is fairly rare, so I'll fix it when it comes up.
412  denom *= par[denomDecay->GetGeneration()] - par[curDecay->GetGeneration()];
413  }
414  denomDecay = denomDecay->GetParentDecay();
415  }
416 
417  sum += TMath::Exp(-par[curDecay->GetGeneration()] * t) / denom;
418 
419  curDecay = curDecay->GetParentDecay();
420  }
421  result *= sum;
422 
423  return result * GetEfficiency();
424 }

679 Double_t TDecay::DecayFit(Double_t* dim, Double_t* par)
680 {
681  /// This fits the total activity caused by the entire chain.
682  Double_t result = 0.0;
683  // Start the fit with flat background;
684  result = par[0];
685  // Parameters might be linked, so I have to sort this out here.
686  // We Must build parameter arrays for each fit.
687  for(auto& i : fChainList) {
688  Int_t par_counter = 0;
689  auto* tmppar = new Double_t[i->Size() + 1];
690  tmppar[par_counter++] = par[fFitFunc->GetParNumber(Form("Intensity_ChainId%d", i->GetChainId()))];
691  for(int j = 0; j < i->Size(); ++j) {
692  tmppar[par_counter++] = par[fFitFunc->GetParNumber(Form("DecayRate_DecayId%d", i->GetDecay(j)->GetDecayId()))];
693  }
694  result += i->EvalPar(dim, tmppar);
695  delete[] tmppar;
696  }
697 
698  return result;

704  TString opt1 = opt;

707  ROOT::Math::MinimizerOptions::SetDefaultMinimizer("Minuit2", "Combination");

712  }

714  SetParameters();
715 
716  // TFitResultPtr fitres = fithist->Fit(fFitFunc,Form("%sWLRS0",opt));
717  TFitResultPtr fitres;
718  if(opt1.Contains("G")) {
719  // Fit using LMFitter..Doesn't do residuals yet
720  TLMFitter fitter;
721  fitter.Fit(fithist, fFitFunc);
722  } else {
723  fitres = fFitFunc->Fit(fithist, Form("%sRIS", opt));
724  Double_t chi2 = fitres->Chi2();
725  Double_t ndf = fitres->Ndf();
726  printf("Chi2/ndf = %lf\n", chi2 / ndf);
727  }

732  i->GetDecay(0)->SetIntensity(fFitFunc->GetParameter(par_num));

736  i->GetDecay(j)->SetDecayRate(fFitFunc->GetParameter(par_num));

741  }
742 
743  return fitres;
