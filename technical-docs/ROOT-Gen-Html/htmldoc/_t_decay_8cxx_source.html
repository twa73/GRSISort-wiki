336 Double_t TSingleDecay::ActivityFunc(Double_t *dim, Double_t *par){
337  ///The general function for a decay chain
338  ///par[0] is the intensity
339  ///par[1*i] is the activity
340  Double_t result = 1.0;
341  UInt_t gencounter = 0;
342  Double_t tlow,thigh;
343  fTotalDecayFunc->GetRange(tlow,thigh);
344  Double_t t = dim[0] - tlow;
345  TSingleDecay *curDecay = this;
346  //Compute the first multiplication
347  while(curDecay){
348  ++gencounter;
349  //par[Generation] gets the decay rate for that decay since the parameters are stored
350  //as [intensity, decayrate1,decayrate2,...]
351  result *= par[curDecay->GetGeneration()];
352 
353  curDecay = curDecay->GetParentDecay();
354  }
355  if(gencounter != fGeneration){
356  printf("We have Problems!\n");
357  return 0.0;
358  }
359  //Multiply by the initial intensity of the initial parent.
360  result*=par[0]/par[1];
361  //Now we need to deal with the second term
362  Double_t sum = 0.0;
363  curDecay = this;
364  while(curDecay){
365  Double_t denom = 1.0;
366  TSingleDecay* denomDecay = this;
367  while(denomDecay){
368  if(denomDecay != curDecay){
369  //This term has problems if two or more rates are very similar. Will have to put a taylor expansion in here
370  //if this problem comes up. I believe the solution to this also depends on the number of nuclei in the chain
371  //that have similar rates. I think either of these issues is fairly rare, so I'll fix it when it comes up.
372  denom*=par[denomDecay->GetGeneration()] - par[curDecay->GetGeneration()];
373  }
374  denomDecay = denomDecay->GetParentDecay();
375  }
376 
377  sum+=TMath::Exp(-par[curDecay->GetGeneration()]*t)/denom;
378 
379  curDecay = curDecay->GetParentDecay();
380  }
381  result*=sum;
382 
383  return result*GetEfficiency();
384 
385 }
