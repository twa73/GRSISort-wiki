103 void TEpicsFrag::BuildScalerMap(TTree* tree)
104 {
105  if(tree == nullptr) {
106  std::cout<<DRED<<"Could not build map from tree"<<RESET_COLOR<<std::endl;
107  }
108  // Loop through the tree and insert the scalers into the map
109  fScalerMap.clear();
110  TEpicsFrag* my_frag = nullptr;
111  if(tree->SetBranchAddress("TEpicsFrag", &my_frag) == 0) {
112  for(int i = 0; i < tree->GetEntries(); ++i) {
113  tree->GetEntry(i);
114  if((static_cast<Long64_t>(my_frag->fDaqTimeStamp) - static_cast<Long64_t>(TRunInfo::Get()->RunStart())) < fSmallestTime) {
115  fSmallestTime = static_cast<Long64_t>(my_frag->fDaqTimeStamp) - static_cast<Long64_t>(TRunInfo::Get()->RunStart());
116  }
117  fScalerMap[static_cast<Long64_t>(my_frag->fDaqTimeStamp) -
118  static_cast<Long64_t>(TRunInfo::Get()->RunStart())] = *my_frag;
119  }
120  } else {
121  std::cout<<DRED<<"Could not build map from tree"<<RESET_COLOR<<std::endl;
122  }
123 }
