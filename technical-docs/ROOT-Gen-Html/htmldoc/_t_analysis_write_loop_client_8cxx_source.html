87 void TAnalysisWriteLoopClient::AddBranch(TClass* cls)
88 {
89  if(fDetMap.count(cls) == 0u) {
90  // This uses the ROOT dictionaries, so we need to lock the threads.
91  TThread::Lock();
92 
93  // Make a default detector of that type.
94  TDetector* det_p = reinterpret_cast<TDetector*>(cls->New());
95  fDefaultDets[cls] = det_p;
96 
97  // Add to our local map
98  auto det_pp = new TDetector*;
99  *det_pp = det_p;
100  fDetMap[cls] = det_pp;
101 
102  // check if this detector class exists in the global map
103  {
104  //std::cout<<std::endl<<Name()<<" locking"<<std::endl;
105  //std::lock_guard<std::mutex> lock(fGlobalMapMutex);
106  if(fGlobalDetMap.count(cls) == 0u) {
107  fGlobalDetMap[cls] = det_pp;
108  }
109  //std::cout<<std::endl<<Name()<<" done"<<std::endl;
110  }
111 
112  // Make a new branch.
113  TBranch* newBranch = fEventTree->Branch(cls->GetName(), cls->GetName(), det_pp);
114 
115  // Fill the new branch up to the point where the tree is filled.
116  // Explanation:
117  // When TTree::Fill is called, it calls TBranch::Fill for each
118  // branch, then increments the number of entries. We may be
119  // adding branches after other branches have already been filled.
120  // If the branch 'x' has been filled 100 times before the branch
121  // 'y' is created, then the next call to TTree::Fill will fill
122  // entry 101 of 'x', but entry 1 of 'y', rather than entry
123  // 101 of both.
124  // Therefore, we need to fill the new branch as many times as
125  // TTree::Fill has been called before.
126  std::lock_guard<std::mutex> lock(ttree_fill_mutex);
127  for(int i = 0; i < fEventTree->GetEntries(); i++) {
128  newBranch->Fill();
129  }
130 
131  if(fFirstClient) {
132  std::cout<<"\r"<<std::string(30, ' ')<<"\r"<<Name()<<": added \""<<cls->GetName()<<R"(" branch)"<<std::endl;
133  }
134 
135  // Unlock after we are done.
136  TThread::UnLock();
137  }
138 }

140 void TAnalysisWriteLoopClient::WriteEvent(std::shared_ptr<TUnpackedEvent>& event)
141 {
142  if(fEventTree != nullptr) {
143  // Clear pointers from previous writes.
144  // Note that we cannot just set this equal to nullptr,
145  // because ROOT would then construct a new object.
146  // This contradicts the ROOT documentation for TBranchElement::SetAddress,
147  // which suggests that a new object would be constructed only when setting the address,
148  // not when filling the TTree.
149  for(auto& elem : fDetMap) {
150  (*elem.second)->Clear();
151  }
152 
153  // Load current events
154  for(const auto& det : event->GetDetectors()) {
155  TClass* cls = det->IsA();
156  try {
157  **fDetMap.at(cls) = *(det.get());
158  } catch(std::out_of_range& e) {
159  AddBranch(cls);
160  **fDetMap.at(cls) = *(det.get());
161  }
162  (*fDetMap.at(cls))->ClearTransients();
163  }
164 
165  // Check if other clients found new detectors
166  {
167  //std::cout<<std::endl<<Name()<<" locking for AddBranch"<<std::endl;
168  //std::lock_guard<std::mutex> lock(fGlobalMapMutex);
169  if(fDetMap.size() != fGlobalDetMap.size()) {
170  // loop over global map of detectors and add those we don't have in our list
171  for(auto const& det : fGlobalDetMap) {
172  // AddBranch checks if detector exists
173  //std::cout<<std::endl<<Name()<<" Adding branch "<<det.first->GetName()<<std::endl;
174  AddBranch(det.first);
175  }
176  }
177  //std::cout<<std::endl<<Name()<<" done with AddBranch"<<std::endl;
178  }
179

181  // std::lock_guard<std::mutex> lock(ttree_fill_mutex);
182  fEventTree->Fill();
183 
184  // write file every 100000 popped events (this sends the events to the server)
185  if(fItemsPopped%100000 == 0) {
186  std::lock_guard<std::mutex> lock(ttree_fill_mutex);
187  fOutputFile->Write();
188  }
