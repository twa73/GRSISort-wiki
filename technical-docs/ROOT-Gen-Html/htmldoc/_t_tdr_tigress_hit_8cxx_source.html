97 void TTdrTigressHit::Add(const TDetectorHit* hit)
98 {
99  // add another TDR tigress hit to this one (for addback),
100  // using the time and position information of the one with the higher energy
101  const TTdrTigressHit* tigressHit = dynamic_cast<const TTdrTigressHit*>(hit);
102  if(tigressHit == nullptr) {
103  throw std::runtime_error("trying to add non-tdr-tigress hit to tdr-tigress hit!");
104  }
105 
106  if(!CompareEnergy(this, tigressHit)) {
107  SetCfd(tigressHit->GetCfd());
108  SetTime(tigressHit->GetTime());
109  // SetPosition(tigressHit->GetPosition());
110  SetAddress(tigressHit->GetAddress());
111  } else {
112  SetTime(GetTime());
113  }
114  SetEnergy(GetEnergy() + tigressHit->GetEnergy());
115  // this has to be done at the very end, otherwise GetEnergy() might not work
116  SetCharge(0);
117  // Add all of the pileups.This should be changed when the max number of pileups changes
118  if((NPileUps() + tigressHit->NPileUps()) < 4) {
119  SetNPileUps(NPileUps() + tigressHit->NPileUps());
120  } else {
121  SetNPileUps(3);
122  }
123  if((PUHit() + tigressHit->PUHit()) < 4) {
124  SetPUHit(PUHit() + tigressHit->PUHit());
125  } else {
126  SetPUHit(3);
127  }
128  // KValue is somewhate meaningless in addback, so I am using it as an indicator that a piledup hit was added-back RD
129  if(GetKValue() > tigressHit->GetKValue()) {
130  SetKValue(tigressHit->GetKValue());
131  }
132 }
