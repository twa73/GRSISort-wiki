308  void parse(int argc, char** argv, bool firstPass)
309  {
310  /// this version takes argc and argv, parses them, and sets only those
311  /// that have the matching firstPass flag set
312  /// this allows us to parse command line arguments in two stages, one
313  /// to get the normal options and file names (from which the run info
314  /// and analysis options are read), and a second stage where only the
315  /// run info and analysis option flags are parsed
316  bool double_dash_encountered = false;
317 
318  int iarg = 1;
319  while(iarg < argc) {
320  std::string arg = argv[iarg++];
321 
322  if(arg.at(0) != '-' || double_dash_encountered) {
323  handle_default_option(argc, argv, iarg, firstPass);
324  } else if(arg.substr(0, 2) == "--") {
325  handle_long_flag(argc, argv, iarg, firstPass);
326  } else {
327  handle_short_flag(argc, argv, iarg, firstPass);
328  }
329  }
330 
331  for(auto& val : values) {
332  if(val->is_required() && !val->is_present()) {
333  std::stringstream ss;
334  ss<<R"(Required argument ")"<<val->flag_name()<<R"(" is not present)";
335  throw ParseError(ss.str());
336  }
337  }
338  }

340  void parse_file(std::string& filename)
341  {
342  std::ifstream infile(filename);
343 
344  std::string line;
345  while(std::getline(infile, line)) {
346  size_t colon = line.find(':');
347  bool has_colon = (colon != std::string::npos);
348 
349  std::string flag;
350  std::string remainder;
351  if(has_colon) {
352  flag = line.substr(0, colon);
353  std::stringstream(flag) >> flag; // Strip out whitespace
354  if(flag.length() == 1) {
355  flag = "-" + flag;
356  } else {
357  flag = "--" + flag;
358  }
359  remainder = line.substr(colon + 1, line.length());
360  } else {
361  flag = "";
362  remainder = line;
363  }
364 
365  std::vector<std::string> args;
366  std::stringstream ss(remainder);
367  std::string tmparg;
368  while(ss >> tmparg) {
369  args.push_back(tmparg);

371

374  item.parse(flag, args, true, true); // parse "firstPass" items only
375  item.parse(flag, args, false, true); // parse "!firstPass" items only
376  } else {
377  for(auto& arg : args) {
378  ArgParseItem& item = get_item(arg);
379  item.parse(arg, std::vector<std::string>{arg}, true); // parse "firstPass" items only
380  item.parse(arg, std::vector<std::string>{arg}, false); // parse "!firstPass" items only
381  }
382  }

422  {
423  std::string arg = argv[iarg - 1];
424  std::vector<std::string> flag_args;
425  std::string flag;
426  size_t equals_index = arg.find('=');
427  if(equals_index == std::string::npos) {
428  // flag followed by list of flag_args

445  void handle_short_flag(int argc, char** argv, int& iarg, bool firstPass)
446  {
447  std::string arg = argv[iarg - 1];
448  std::string flag = arg.substr(0, 2);
449  ArgParseItem& item = get_item(flag);
450  if(item.num_arguments() == 0) {
451  // Each character is a boolean flag
452  for(unsigned int ichar = 1; ichar < arg.length(); ichar++) {
453  std::string tmpflag = "-" + arg.substr(ichar, 1);
454  std::vector<std::string> flag_args;
455  get_item(tmpflag).parse(tmpflag, flag_args, firstPass);
456  }
457  } else {
458  if(arg.length() == 2) {
459  // Next arguments passed to the program get passed to the flag.
460  std::vector<std::string> flag_args = argument_list(argc, argv, iarg, item.num_arguments());
461  item.parse(flag, flag_args, firstPass);
462  } else {
463  // Everything past the first character is argument to the flag.
464  std::vector<std::string> flag_args{arg.substr(2)};
465  item.parse(flag, flag_args, firstPass);
466  }
467  }
