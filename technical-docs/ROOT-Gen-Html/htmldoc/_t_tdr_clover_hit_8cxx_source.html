97 void TTdrCloverHit::Add(const TDetectorHit* hit)
98 {
99  // add another TDR clover hit to this one (for addback),
100  // using the time and position information of the one with the higher energy
101  const TTdrCloverHit* cloverHit = dynamic_cast<const TTdrCloverHit*>(hit);
102  if(cloverHit == nullptr) {
103  throw std::runtime_error("trying to add non-clover hit to clover hit!");
104  }
105 
106  if(!CompareEnergy(this, cloverHit)) {
107  SetCfd(cloverHit->GetCfd());
108  SetTime(cloverHit->GetTime());
109  // SetPosition(cloverHit->GetPosition());
110  SetAddress(cloverHit->GetAddress());
111  } else {
112  SetTime(GetTime());
113  }
114  SetEnergy(GetEnergy() + cloverHit->GetEnergy());
115  // this has to be done at the very end, otherwise GetEnergy() might not work
116  SetCharge(0);
117  // Add all of the pileups.This should be changed when the max number of pileups changes
118  if((NPileUps() + cloverHit->NPileUps()) < 4) {
119  SetNPileUps(NPileUps() + cloverHit->NPileUps());
120  } else {
121  SetNPileUps(3);
122  }
123  if((PUHit() + cloverHit->PUHit()) < 4) {
124  SetPUHit(PUHit() + cloverHit->PUHit());
125  } else {
126  SetPUHit(3);
127  }
128  // KValue is somewhate meaningless in addback, so I am using it as an indicator that a piledup hit was added-back RD
129  if(GetKValue() > cloverHit->GetKValue()) {
130  SetKValue(cloverHit->GetKValue());
131  }
132 }
