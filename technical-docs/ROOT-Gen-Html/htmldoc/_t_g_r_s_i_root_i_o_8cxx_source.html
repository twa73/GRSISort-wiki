365 int TGRSIRootIO::GetRunNumber(std::string fileName) {
366  if(fileName.length()==0) {
367  return 0;
368  }
369  std::size_t found = fileName.rfind(".root");
370  if(found == std::string::npos) {
371  return 0;
372  }
373  std::size_t found2 = fileName.rfind('-');
374  //printf("found 2 = %i\n",found2);
375 
376  if(found2 == std::string::npos)
377  found2 = fileName.rfind('_');
378  std::string temp;
379  if(found2 == std::string::npos || fileName.compare(found2+4,5,".root") !=0 ) {
380  temp = fileName.substr(found-5,5);
381  }
382  else {
383  temp = fileName.substr(found-9,5);
384  }
385  return atoi(temp.c_str());
