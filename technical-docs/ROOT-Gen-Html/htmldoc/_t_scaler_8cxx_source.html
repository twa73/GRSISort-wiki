119 std::vector<UInt_t> TScaler::GetScaler(UInt_t address, ULong64_t time) const
120 {
121  /// Returns the vector of scaler values for address "address" at the time "time".
122  /// If the time is after the last entry, we return the last entry, otherwise we return the following entry (or the
123  /// current entry if the time is an exact match).
124  if(fTree == nullptr || fEntries == 0) {
125  printf("Empty\n");
126  return std::vector<UInt_t>(0);
127  }
128  if(!fScalerMap.empty()) {
129  // Check that this address exists
130  if(fScalerMap.find(address) == fScalerMap.end()) {
131  return std::vector<UInt_t>();
132  }
133  auto currIt = fScalerMap.find(address)->second.lower_bound(time);
134  // if the time is after our last entry, we return the last entry
135  if(currIt == fScalerMap.find(address)->second.end()) {
136  return std::prev(fScalerMap.find(address)->second.end())->second;
137  }
138  // The lower_bound function returns an iterator to the NEXT map element or the current map element if the time is
139  // a perfect match.
140  return currIt->second;
141  }
142  // loop through the tree and find the right entry
143  for(Long64_t entry = 0; entry < fEntries; ++entry) {
144  fTree->GetEntry(entry);
145  // got the exact time match, so we return the current value
146  if(fScalerData->GetAddress() == address && fScalerData->GetTimeStamp() == time) {
147  return fScalerData->GetScaler();
148  }
149  // got a scaler that is later than our requested time, so we need to find the previous time and return that one
150  if(fScalerData->GetAddress() == address && fScalerData->GetTimeStamp() > time) {
151  for(--entry; entry >= 0; --entry) {
152  if(fScalerData->GetAddress() == address) {
153  return fScalerData->GetScaler();
154  }
155  }
156  // if we failed to find any previous data we return a vector of zeros
157  return std::vector<UInt_t>(4, 0);
158  }
159  }
160  // failed to find any matching time so we just return the very last value (or should we search for the last value of
161  // this address?)
162  return fScalerData->GetScaler();
163 }

174 UInt_t TScaler::GetScalerDifference(UInt_t address, ULong64_t time, size_t index) const
175 {
176  /// Returns the difference between "index"th scaler value for address "address" at the time "time" and the previous
177  /// scaler value.
178  /// If the time is after our last entry, we return the last entry divided by the number of entries,
179  /// if this is before the first scaler, we just return the first scaler, otherwise we return the scaler minus the
180  /// previous scaler.
181  if(fTree == nullptr || fEntries == 0) {
182  printf("Empty\n");
183  return 0;
184  }
185  if(!fScalerMap.empty()) {
186  // Check that this address exists
187  if(fScalerMap.find(address) == fScalerMap.end()) {
188  return 0;
189  }
190  auto currIt = fScalerMap.find(address)->second.lower_bound(time);
191  // if the time is after our last entry, we return the last entry divided by the number of entries
192  if(currIt == fScalerMap.find(address)->second.end()) {
193  return std::prev(fScalerMap.find(address)->second.end())->second[index] /
194  fScalerMap.find(address)->second.size();
195  }
196  // if this is the before or at the first scaler, we just return the first scaler
197  if(currIt == fScalerMap.find(address)->second.begin()) {
198  return fScalerMap.find(address)->second.begin()->second[index];
199  }
200  // otherwise we return the scaler minus the previous scaler
201  return (currIt->second[index]) - (std::prev(currIt)->second[index]);
202  }
203  // loop through the tree and find the right entry
204  for(Long64_t entry = 0; entry < fEntries; ++entry) {
205  fTree->GetEntry(entry);
206  // got the exact time match, so we look for the previous value
207  if(fScalerData->GetAddress() == address && fScalerData->GetTimeStamp() == time) {
208  UInt_t currentValue = fScalerData->GetScaler(index);
209  for(--entry; entry >= 0; --entry) {
210  if(fScalerData->GetAddress() == address) {
211  return currentValue - fScalerData->GetScaler(index);
212  }
213  }
214  return currentValue;
215  }
216  // got a scaler that is later than our requested time, so we need to find the previous time and return that one
217  if(fScalerData->GetAddress() == address && fScalerData->GetTimeStamp() > time) {
218  for(--entry; entry >= 0; --entry) {
219  if(fScalerData->GetAddress() == address) {
220  // got the current entry, so we look for the previous value
221  UInt_t currentValue = fScalerData->GetScaler(index);
222  for(--entry; entry >= 0; --entry) {
223  if(fScalerData->GetAddress() == address) {
224  return currentValue - fScalerData->GetScaler(index);
225  }
226  }
227  // failed to find a previous entry, so we return the current value
228  return currentValue;
229  }
230  }
231  // if we failed to find any previous data we return zero
232  return 0;
233  }
234  }
235  // failed to find any matching time, so we return 0 (???)
236  return 0;
237 }

507 ULong64_t TScaler::GetTimePeriod(UInt_t address)
508 {
509  /// Get time period of scaler readouts for address "address" by calculating all time differences and choosing the one
510  /// that occurs most often.
511  /// Returns 0 if the address doesn't exist in the map.
512  if(fTimePeriod[address] == 0) {
513  ULong64_t previousValue = 0;
514  for(Long64_t entry = 0; entry < fEntries; ++entry) {
515  fTree->GetEntry(entry);
516  if(fScalerData->GetAddress() == address) {
517  // fill the difference between the current and the next scaler (if we found a previous value and that one is
518  // smaller than the current one)
519  if(previousValue != 0 && previousValue < fScalerData->GetTimeStamp()) {
520  // compare timestamp of current element with that of the previous element
521  ULong64_t diff = fScalerData->GetTimeStamp() - previousValue;
522  fNumberOfTimePeriods[address][diff]++;
523  }
524  previousValue = fScalerData->GetTimeStamp();
525  }
526  }
527  int counter = 0;
528  for(auto it = fNumberOfTimePeriods[address].begin(); it != fNumberOfTimePeriods[address].end(); ++it) {
529  if(it->second > counter) {
530  counter = it->second;
531  fTimePeriod[address] = it->first;
532  }
533  }
534  }
535 
536  return fTimePeriod[address];
537 }
