174 bool GPeak::InitParams(TH1* fithist)
175 {
176  if(fithist == nullptr) {
177  printf("No histogram is associated yet, no initial guesses made\n");
178  return false;
179  }
180  // printf("%s called.\n",__PRETTY_FUNCTION__); fflush(stdout);
181  // Makes initial guesses at parameters for the fit. Uses the histogram to
182  Double_t xlow, xhigh;
183  GetRange(xlow, xhigh);
184 
185  // Int_t bin = fithist->GetXaxis()->FindBin(GetParameter("centroid"));
186  Int_t binlow = fithist->GetXaxis()->FindBin(xlow);
187  Int_t binhigh = fithist->GetXaxis()->FindBin(xhigh);
188 
189  Double_t highy = fithist->GetBinContent(binlow);
190  Double_t lowy = fithist->GetBinContent(binhigh);
191  for(int x = 1; x < 5; x++) {
192  highy += fithist->GetBinContent(binlow - x);
193  lowy += fithist->GetBinContent(binhigh + x);
194  }
195  highy = highy / 5.0;
196  lowy = lowy / 5.0;
197 
198  // Double_t yhigh = fithist->GetBinContent(binhigh);
199  // Double_t ylow = fithist->GetBinContent(binlow);
200  if(lowy > highy) {
201  std::swap(lowy, highy);
202  }
203 
204  double largestx = 0.0;
205  double largesty = 0.0;
206  int i = binlow;
207  for(; i <= binhigh; i++) {
208  if(fithist->GetBinContent(i) > largesty) {
209  largesty = fithist->GetBinContent(i);
210  largestx = fithist->GetXaxis()->GetBinCenter(i);
211  }
212  }
213 
214  // - par[0]: height of peak
215  // - par[1]: cent of peak
216  // - par[2]: sigma
217  // - par[3]: R: relative height of skewed gaus to gaus
218  // - par[4]: beta: "skewedness" of the skewed gaussin
219  // - par[5]: step: size of stepfunction step.
220 
221  // - par[6]: base bg height.
222 
223  // limits.
224  TF1::SetParLimits(0, 0, largesty * 2);
225  TF1::SetParLimits(1, xlow, xhigh);
226  TF1::SetParLimits(2, 0.1, xhigh - xlow);
227  TF1::SetParLimits(3, 0.0, 40);
228  TF1::SetParLimits(4, 0.01, 5);
229  double step = ((highy - lowy) / largesty) * 50;
230 
231  // TF1::SetParLimits(5,step-step*.1,step+.1*step);
232  TF1::SetParLimits(5, 0.0, step + step);
233 
234  // double slope = (yhigh-ylow)/(xhigh-xlow);
235  // double offset = yhigh-slope*xhigh;
236  double offset = lowy;
237  TF1::SetParLimits(6, offset - 0.5 * offset, offset + offset);
238  // TF1::SetParLimits(7,-2*slope,2*slope);
239 
240  // Make initial guesses
241  TF1::SetParameter(0, largesty); // fithist->GetBinContent(bin));
242  TF1::SetParameter(1, largestx); // GetParameter("centroid"));
243  TF1::SetParameter(2, (largestx * .01) / 2.35); // 2,(xhigh-xlow)); //2.0/binWidth); //
244  TF1::SetParameter(3, 5.);
245  TF1::SetParameter(4, 1.);
246  TF1::SetParameter(5, step);
247  TF1::SetParameter(6, offset);
248  // TF1::SetParameter(7,slope);
249 
250  TF1::SetParError(0, 0.10 * largesty);
251  TF1::SetParError(1, 0.25);
252  TF1::SetParError(2, 0.10 * ((largestx * .01) / 2.35));
253  TF1::SetParError(3, 5);
254  TF1::SetParError(4, 0.5);
255  TF1::SetParError(5, 0.10 * step);
256  TF1::SetParError(6, 0.10 * offset);
257 
258  // TF1::Print();
259 
260  SetInitialized();
261  return true;

451 void GPeak::DrawResiduals(TH1* hist) const
452 {
453  if(hist == nullptr) {
454  return;
455  }
456  if(fChi2 < 0.000000001) {
457  printf("No fit performed\n");
458  return;
459  }
460  Double_t xlow, xhigh;
461  GetRange(xlow, xhigh);
462  Int_t nbins = hist->GetXaxis()->GetNbins();
463  auto* res = new Double_t[nbins];
464  auto* bin = new Double_t[nbins];
465  Int_t points = 0;
466  for(int i = 1; i <= nbins; i++) {
467  if(hist->GetBinCenter(i) <= xlow || hist->GetBinCenter(i) >= xhigh) {
468  continue;
469  }
470  res[points] = (hist->GetBinContent(i) - Eval(hist->GetBinCenter(i))) + GetParameter("Height") / 2;
471  bin[points] = hist->GetBinCenter(i);
472  points++;
473  }
474  new GCanvas();
475  auto* residuals = new TGraph(points, bin, res);
476  residuals->Draw("*AC");
477  delete[] res;
478  delete[] bin;
479 }
