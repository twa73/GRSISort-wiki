129 double FitPositionOptimisation(TH1* fHistIn,int& fPeak,int& fLeft,int& fRight){
130  //Determine a good fitting range near the suggested fLeft and fRight
131 
132  //Find the local maximum from the given peak
133  double fHH=FindLocalMax(fHistIn,fPeak,fLeft,fRight);
134 
135  TH1* fHist=SmoothOffBack(fHistIn);
136 
137  double fMaxY=fHH*5;
138 
139  double fMin=fMaxY;
140  for(int j=fLeft;j<=fRight;j++)if(fHist->GetBinContent(j)<fMin)fMin=fHist->GetBinContent(j);
141  fMin*=5;
142 
143  //Go along taking an average and when it starts to go back up, kill the progression
144  double fMinR=fMaxY*2;
145  for(int j=fPeak+5;j<=fRight;j++){
146  double sAverage=0;
147  for(int l=0;l<5;l++)sAverage+=abs(fHist->GetBinContent(j+l));
148  if(sAverage<fMinR){fMinR=sAverage;}
149  if(sAverage>fMinR+(fMaxY-fMin)*0.05){fRight=j;}
150  }
151 
152  double fMinL=fMaxY*2;
153  for(int j=fPeak-5;j>=fLeft;j--){
154  double sAverage=0;
155  for(int l=0;l<5;l++)sAverage+=abs(fHist->GetBinContent(j-l));
156  if(sAverage<fMinL){fMinL=sAverage;}
157  if(sAverage>fMinL+(fMaxY-fMin)*0.05){fLeft=j;}
158  }
159  delete fHist;
160  return fHH;
161 }
