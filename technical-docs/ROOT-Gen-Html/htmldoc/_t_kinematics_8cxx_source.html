339 double TKinematics::GetBeamEnergy(double LabAngle, double LabEnergy)
340 {
341  // Returns the beam energy given the lab angle and energy of the ejectile
342  double ProjectileMass = fM[0];
343  double TargetMass = fM[1];
344 
345  double ts = pow(TargetMass, 2);
346  double ps = pow(ProjectileMass, 2);
347  double cs = pow(cos(LabAngle), 2);
348  double es = pow(LabEnergy, 2);
349  double te = TargetMass * LabEnergy;
350 
351  return (-8 * ProjectileMass - 4 * TargetMass + LabEnergy / cs - 2 * LabEnergy * tan(LabAngle) +
352  sqrt(16 * ts * cs +
353  LabEnergy * (LabEnergy / cs * pow(cos(LabAngle) - sin(LabAngle), 4) +
354  8 * TargetMass * (3 + sin(2 * LabAngle)))) /
355  cos(LabAngle) +
356  sqrt((24 * ts * LabEnergy * cs + 32 * ps * TargetMass * pow(cos(LabAngle), 4) +
357  2 * TargetMass * es * pow(cos(LabAngle) - sin(LabAngle), 4) +
358  16 * ts * LabEnergy * pow(cos(LabAngle), 3) * sin(LabAngle) -
359  8 * ps * LabEnergy * cs * (sin(2 * LabAngle) - 1) +
360  2 * ps * cos(3 * LabAngle) * sqrt(2 * (4 * ts + 12 * te + es) + (8 * ts - 2 * es) * cos(2 * LabAngle) +
361  LabEnergy * (LabEnergy / cs + 8 * TargetMass * sin(2 * LabAngle) -
362  4 * LabEnergy * tan(LabAngle))) +
363  2 * cos(LabAngle) * (3 * ps + te - te * sin(2 * LabAngle)) *
364  sqrt(2 * (4 * ts + 12 * te + es) + (8 * ts - 2 * es) * cos(2 * LabAngle) +
365  LabEnergy *
366  (LabEnergy / cs + 8 * TargetMass * sin(2 * LabAngle) - 4 * LabEnergy * tan(LabAngle)))) /
367  (pow(cos(LabAngle), 4) * TargetMass))) /
368  8.;
369 }

371 void TKinematics::Initial()
372 {
373  // An initializing function that sets the energies and momenta of the beam and target in the lab and CM frame,
374  // as well as a few basic calculations.
375  fT[0] = fEBeam; // KE of beam in lab
376  fT[1] = 0; // KE of target in lab
377  fE[0] = E_tm(fT[0], fM[0]); // total E of beam in lab
378  fE[1] = E_tm(fT[1], fM[1]); // total E of target in lab
379  fP[0] = sqrt(fT[0] * fT[0] + 2 * fT[0] * fM[0]); // momentum of beam in lab
380  fP[1] = 0; // momentum of target in lab
381  fV[0] = V_pe(fP[0], fE[0]); // velocity of beam in lab
382  fV[1] = V_pe(fP[1], fE[1]); // velocity of target in lab
383 
384  fEcm[0] = GetCmEnergy(fEBeam) / 2 + (fM[0] * fM[0] - fM[1] * fM[1]) / (2 * GetCmEnergy(fEBeam));
385  fEcm[1] = GetCmEnergy(fEBeam) / 2 - (fM[0] * fM[0] - fM[1] * fM[1]) / (2 * GetCmEnergy(fEBeam));
386  fTcm[0] = fEcm[0] - fM[0];
387  fTcm[1] = fEcm[1] - fM[1];
388  fPcm[0] = Pcm_em(fEcm[0], fM[0]);
389  fPcm[1] = Pcm_em(fEcm[1], fM[1]);
390  fVcm[0] = V_pe(fPcm[0], fEcm[0]);
391  fVcm[1] = V_pe(fPcm[1], fEcm[1]);
392 
393  fBeta_cm = (fP[0] - fP[1]) / (fE[0] + fE[1]);
394  fBetacm[0] = betacm_tm(fTcm[0], fM[0]);
395  fBetacm[1] = -betacm_tm(fTcm[1], fM[1]);
396  fGamma_cm = 1 / sqrt(1 - fBeta_cm * fBeta_cm);
397  fTCm_i = GetCmEnergy(fEBeam) - fM[0] - fM[1];
398  fTCm_f = fTCm_i + fQValue;
399  // printf("\n\nfM[0] = %.05f \tfM[1] = %.05f \tfM[2] = %.05f \tfM[3] = %.05f \n",fM[0],fM[1],fM[2],fM[3]);

405 void TKinematics::FinalCm()
406 {
407  // Calculates the recoil and ejectile energies and momenta in the CM frame
408 
409  // angle of proton in cm system
410  if(fParticle[2] == nullptr && fParticle[3] == nullptr) {
411  fM[2] = fParticle[1]->GetMass();
412  fM[3] = fParticle[0]->GetMass();
413  }
414  fTcm[2] = fTCm_f / 2 * (fTCm_f + 2 * fM[3]) / GetCmEnergy(fEBeam);
415  fTcm[3] = fTCm_f / 2 * (fTCm_f + 2 * fM[2]) / GetCmEnergy(fEBeam);
416  fEcm[2] = E_tm(fTcm[2], fM[2]);
417  fEcm[3] = E_tm(fTcm[3], fM[3]);
418  fPcm[2] = Pcm_em(fEcm[2], fM[2]);
419  fPcm[3] = Pcm_em(fEcm[3], fM[3]);
420  fVcm[2] = V_pe(fPcm[2], fEcm[2]);
421  fVcm[3] = V_pe(fPcm[3], fEcm[3]);
422  fBetacm[2] = -betacm_tm(fTcm[2], fM[2]);
423  fBetacm[3] = betacm_tm(fTcm[3], fM[3]);
424 }
