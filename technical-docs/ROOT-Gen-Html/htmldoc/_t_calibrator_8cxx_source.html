133  if(option.Contains("draw", TString::kIgnoreCase)) {

333 std::map<double, double> TCalibrator::Match(std::vector<double> peaks, std::vector<double> source)
334 {
335  std::map<double, double> map;
336  std::sort(peaks.begin(), peaks.end());
337  std::sort(source.begin(), source.end());
338 
339  // Peaks are the fitted points.
340  // source are the known values
341 
342  std::vector<bool> filled(source.size());
343  std::fill(filled.begin(), filled.begin() + peaks.size(), true);
344 
345  // std::cout<<"Num peaks: "<<peaks.size()<<std::endl;
346  // std::cout<<"Num source: "<<source.size()<<std::endl;
347 
348  TLinearFitter fitter(1, "1 ++ x");
349 
350  for(size_t num_data_points = peaks.size(); num_data_points > 0; num_data_points--) {
351  double best_chi2 = DBL_MAX;
352  for(auto peak_values : combinations(peaks, num_data_points)) {
353  // Add a (0,0) point to the calibration.
354  peak_values.push_back(0);
355  for(auto source_values : combinations(source, num_data_points)) {
356  source_values.push_back(0);
357 
358  if(peaks.size() > 3) {
359  double max_err = 0.02;
360  double pratio = peak_values.front() / peak_values.at(peak_values.size() - 2);
361  double sratio = source_values.front() / source_values.at(source_values.size() - 2);
362  // std::cout<<"ratio: "<<pratio<<" - "<<sratio<<" = "<<std::abs(pratio-sratio)<<std::endl;
363  if(std::abs(pratio - sratio) > max_err) {
364  // std::cout<<"skipping"<<std::endl;
365  continue;
366  }
367  }
368 
369  fitter.ClearPoints();
370  fitter.AssignData(source_values.size(), 1, peak_values.data(), source_values.data());
371  fitter.Eval();
372 
373  if(fitter.GetChisquare() < best_chi2) {
374  map.clear();
375  for(size_t i = 0; i < num_data_points; i++) {
376  map[peak_values[i]] = source_values[i];
377  }
378  best_chi2 = fitter.GetChisquare();
379  }
380  }
381  }
382 
383  // Remove one peak value from the best fit, make sure that we reproduce (0,0) intercept.
384  if(map.size() > 2) {
385  std::vector<double> peak_values;
386  std::vector<double> source_values;
387  for(auto& item : map) {
388  peak_values.push_back(item.first);
389  source_values.push_back(item.second);
390  }
391 
392  for(size_t skipped_point = 0; skipped_point < source_values.size(); skipped_point++) {
393  std::swap(peak_values[skipped_point], peak_values.back());
394  std::swap(source_values[skipped_point], source_values.back());
395 
396  fitter.ClearPoints();
397  fitter.AssignData(source_values.size() - 1, 1, peak_values.data(), source_values.data());
398  fitter.Eval();
399 
400  if(std::abs(fitter.GetParameter(0)) > 10) {
401  map.clear();
402  break;
403  }
404 
405  std::swap(peak_values[skipped_point], peak_values.back());
406  std::swap(source_values[skipped_point], source_values.back());
407  }
408  }
409 
410  if(!map.empty()) {
411  return map;
412  }
413  }
414 
415  map.clear();
416  return map;
