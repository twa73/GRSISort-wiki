432 TVector3 TIfin::GetPosition(int DetNbr, int CryNbr, double dist)
433 {
434  // Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
435  // This is calculated to the most likely interaction point within the crystal.
436  if(DetNbr > 16) {
437  return TVector3(0, 0, 1);
438  }
439 
440  TVector3 CloverPosition(gCloverPosition[DetNbr]);
441 
442  // XAxis: Array's Port
443  // YAxis: Upwards
444  // ZAxis: Points in the neutron beam direction
445 
446  // Interaction points may eventually be set externally. May make these members of each crystal, or pass from
447  // waveforms.
448  Double_t cp = 17.678;//25.0; // Crystal Center Point mm. (diameter 50mm)
449  Double_t id = 40.0; // Crystal interaction depth mm. (length 80mm)
450  // Set Theta's of the center of each DETECTOR face
451  ////Define one Detector position
452  TVector3 CrystalPosition;
453  switch(CryNbr) {
454  case 0: CrystalPosition.SetXYZ(-cp, cp, id); break; // BLUE
455  case 1: CrystalPosition.SetXYZ(cp, cp, id); break; // GREEN
456  case 2: CrystalPosition.SetXYZ(cp, -cp, id); break; // RED
457  case 3: CrystalPosition.SetXYZ(-cp, -cp, id); break; // WHITE
458  default: CrystalPosition.SetXYZ(0, 0, 1); break;
459  };
460  // Rotate counterclockwise from the downstream position
461  CrystalPosition.RotateY(CloverPosition.Theta());
462  // Rotate around the neutron beam
463  CrystalPosition.RotateZ(CloverPosition.Phi());
464  // Set distance of detector from target
465  CloverPosition.SetMag(dist);
466 
467  return (CloverPosition + CrystalPosition);
468 }

514 Double_t TIfin::CTCorrectedEnergy(const TIfinHit* const hit_to_correct, const TIfinHit* const other_hit,
515  Bool_t time_constraint)
516 {
517  /// Corrects the energy of the hit to correct by ADDING the uncorrected energy of the other hit times the parameter for this combination
518  /// This is different to the very similar TGriffin function that SUBTRACTS instead of ADDING.
519  /// If time_constraint is true it also checks that the two hits are within the addback window to each other.
520  if((hit_to_correct == nullptr) || (other_hit == nullptr)) {
521  std::cerr<<"One of the hits is invalid in TIfin::CTCorrectedEnergy"<<std::endl;
522  return 0;
523  }
524 
525  if(time_constraint) {
526  // Figure out if this passes the selected window
527  if(TMath::Abs(other_hit->GetTime() - hit_to_correct->GetTime()) >
528  TGRSIOptions::AnalysisOptions()->AddbackWindow()) { // placeholder
529  return hit_to_correct->GetEnergy();
530  }
531  }
532 
533  if(hit_to_correct->GetDetector() != other_hit->GetDetector()) {
534  return hit_to_correct->GetEnergy();
535  }
536  static bool been_warned[256] = {false};
537  double fixed_energy = hit_to_correct->GetEnergy();
538  try {
539  if(hit_to_correct->GetChannel() != nullptr) {
540  fixed_energy += hit_to_correct->GetChannel()->GetCTCoeff().at(other_hit->GetCrystal()) * other_hit->GetNoCTEnergy();
541  }
542  } catch(const std::out_of_range& oor) {
543  int id = 16 * hit_to_correct->GetDetector() + 4 * hit_to_correct->GetCrystal() + other_hit->GetCrystal();
544  if(!been_warned[id]) {
545  been_warned[id] = true;
546  std::cerr<<DRED<<"Missing CT correction for Det: "<<hit_to_correct->GetDetector()
547  <<" Crystals: "<<hit_to_correct->GetCrystal()<<" "<<other_hit->GetCrystal()<<" (id "<<id<<")"<<std::endl;
548  }
549  return hit_to_correct->GetEnergy();
550  }
551 
552  return fixed_energy;
553 }
