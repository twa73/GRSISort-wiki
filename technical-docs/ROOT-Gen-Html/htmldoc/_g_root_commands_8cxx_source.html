181  y = hist->GetBinContent(hist->GetXaxis()->FindBin(i_x));

185  text = new TText(x[i], y, Form("%.1f", x[i]));

194  pm->Delete();
195  hist->GetListOfFunctions()->Add(array);
196  return n;
