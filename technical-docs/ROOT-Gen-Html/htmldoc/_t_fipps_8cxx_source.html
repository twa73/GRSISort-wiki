409 TVector3 TFipps::GetPosition(int DetNbr, int CryNbr, double dist)
410 {
411  // Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
412  // This is calculated to the most likely interaction point within the crystal.
413  if(DetNbr > 16) {
414  return TVector3(0, 0, 1);
415  }
416 
417  TVector3 temp_pos(gCloverPosition[DetNbr]);
418 
419  // The axis represent:
420  // XAxis: Array's Port
421  // YAxis: Upwards
422  // ZAxis: Points in the neutron beam direction
423  TVector3 XAxis(1,0,0); TVector3 YAxis(0,1,0); TVector3 ZAxis(0, 0, 1);
424 
425  // Interaction points may eventually be set externally. May make these members of each crystal, or pass from
426  // waveforms.
427  Double_t cp = 25.0; // Crystal Center Point mm. (diameter 50mm)
428  Double_t id = 40.0; // Crystal interaction depth mm. (length 80mm)
429  // Set Theta's of the center of each DETECTOR face
430  ////Define one Detector position
431  TVector3 shift; TVector3 DetectorCenter;
432  switch(CryNbr) {
433  case 0: shift.SetXYZ(-cp, cp, id); break;
434  case 1: shift.SetXYZ(cp, cp, id); break;
435  case 2: shift.SetXYZ(cp, -cp, id); break;
436  case 3: shift.SetXYZ(-cp, -cp, id); break;
437  default: shift.SetXYZ(0, 0, 1); break;
438  };
439  DetectorCenter.SetXYZ(0, 0, 1);
440  // Rotate in the XZ plain to where the detector is
441  TVector3 XZProjection = temp_pos - (temp_pos*YAxis)*YAxis;
442  Double_t YRotationAngle = XZProjection.Angle(ZAxis);
443  shift.RotateY(YRotationAngle);
444  DetectorCenter.RotateY(YRotationAngle);
445 
446  // Rotate upto the angle of the detector
447  Double_t RotationAngle = DetectorCenter.Angle(temp_pos);
448  TVector3 RotationAxis = DetectorCenter.Cross(temp_pos).Unit();
449  shift.Rotate( RotationAngle, RotationAxis );
450 
451  temp_pos.SetMag(dist);
452 
453  return (temp_pos + shift);
454 }

525 {

527  if(hit_vec.size() < 2) {
528  SetCrossTalk(true);

530  }

534 
535  if(TGRSIOptions::AnalysisOptions()->IsCorrectingCrossTalk()) {

540  hit_vec.at(j)->SetEnergy(TFipps::CTCorrectedEnergy(static_cast<TFippsHit*>(hit_vec.at(j)), static_cast<TFippsHit*>(hit_vec.at(i))));
