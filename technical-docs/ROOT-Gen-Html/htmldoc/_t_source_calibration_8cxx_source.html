469  if(fData != nullptr) delete fData;
470  if(fEfficiency != nullptr) delete fEfficiency;
471  fData = new TGraphErrors(map.size());
472  fData->SetLineColor(2);
473  fData->SetMarkerColor(2);
474  fEfficiency = new TGraphErrors(map.size());
475  fEfficiency->SetLineColor(2);
476  fEfficiency->SetMarkerColor(2);
477  int i = 0;
478  for(auto iter = map.begin(); iter != map.end();) {
479  // more readable variable names
480  auto peakPos = std::get<0>(iter->first);
481  auto peakPosErr = std::get<1>(iter->first);
482  auto peakArea = std::get<2>(iter->first);
483  auto peakAreaErr = std::get<3>(iter->first);
484  auto energy = std::get<0>(iter->second);
485  auto energyErr = std::get<1>(iter->second);
486  auto intensity = std::get<2>(iter->second);
487  auto intensityErr = std::get<3>(iter->second);
488  // drop this peak if the uncertainties in area or position are too large
489  if(peakPosErr > 0.1*peakPos || peakAreaErr > peakArea) {
490  if(fVerboseLevel > 1) std::cout<<"Dropping peak with position "<<peakPos<<" +- "<<peakPosErr<<", area "<<peakArea<<" +- "<<peakAreaErr<<", energy "<<energy<<", intensity "<<intensity<<std::endl;
491  map.erase(iter++);
492  } else {
493  fData->SetPoint(i, peakPos, energy);
494  fData->SetPointError(i, peakPosErr, energyErr);
495  // we use the source energy for the x-values, and the ratio of peak area and intensity for y
496  fEfficiency->SetPoint(i, energy, peakArea/intensity);
497  fEfficiency->SetPointError(i, energyErr, peakArea/intensity*TMath::Sqrt(TMath::Power(peakAreaErr/peakArea,2)+TMath::Power(intensityErr/intensity,2)));
498  if(fVerboseLevel > 2) std::cout<<"Using peak with position "<<peakPos<<" +- "<<peakPosErr<<", area "<<peakArea<<" +- "<<peakAreaErr<<", energy "<<energy<<", intensity "<<intensity<<std::endl;
499  ++iter;
500  ++i;
501  }
502  }
503  // remove poly markers that weren't used for the fit
504  TList* functions = fProjection->GetListOfFunctions();
505  TPolyMarker* pm = static_cast<TPolyMarker*>(functions->FindObject("TPolyMarker"));
506  double* oldX = pm->GetX();
507  double* oldY = pm->GetY();
508  int size = pm->GetN();
509  std::vector<double> newX;
510  std::vector<double> newY;
511  for(i = 0; i < size; ++i) {
512  for(auto point : map) {
513  if(TMath::Abs(oldX[i] - std::get<0>(point.first)) < fSigma) {
514  newX.push_back(oldX[i]);
515  newY.push_back(oldY[i]);
516  break;
517  }
518  }
519  }
520  pm->SetPolyMarker(newX.size(), newX.data(), newY.data());
521  // remove fit functions for unused peaks
522  TIter iter(functions);
523  TObject* item;
524  while((item = iter.Next()) != nullptr) {
525  if(item->IsA() == TF1::Class() || item->IsA() == GPeak::Class()) { // if the item is a TF1 or GPeak we see if we can find the centroid in the map of used peaks
526  double centroid = 0.;
527  if(item->IsA() == TF1::Class()) centroid = static_cast<TF1*>(item)->GetParameter(1);
528  else centroid = static_cast<GPeak*>(item)->Centroid();
529  bool found = false;
530  for(auto point : map) {
531  if(TMath::Abs(centroid - std::get<0>(point.first)) < fSigma) {
532  found = true;
533  break;
534  }
535  }
536  // remove the TF1 if it wasn't found in the map
537  if(!found) {
538  functions->Remove(item);
539  }
540  }
541  }
542 }
