211 Bool_t TRunInfo::ParseInputData(const char* inputdata, Option_t* opt)
212 {
213  // A helper function to parse the run info file.
214 
215  std::istringstream infile(inputdata);
216  std::string line;
217  int linenumber = 0;
218 
219  // Parse the info file.
220  while(!std::getline(infile, line).fail() ) {
221  linenumber++;
222  trim(&line);
223  size_t comment = line.find("//");
224  if(comment != std::string::npos) {
225  line = line.substr(0, comment);
226  }
227  if(line.length() == 0u) {
228  continue;
229  }
230 
231  size_t ntype = line.find(':');
232  if(ntype == std::string::npos) {
233  continue; // no seperator, not useful.
234  }
235 
236  std::string type = line.substr(0, ntype);
237  line = line.substr(ntype + 1, line.length());
238  trim(&line);
239  int j = 0;
240  while(type[j] != 0) {
241  char c = *(type.c_str() + j);
242  c = toupper(c);
243  type[j++] = c;
244  }
245  if(type.compare("CAL") == 0 || type.compare("CALFILE") == 0) {
246  // TODO Make this work again, using priorities
247  // TGRSIOptions::AddInputCalFile(line);
248  } else if(type.compare("MID") == 0 || type.compare("MIDAS") == 0 || type.compare("MIDASFILE") == 0) {
249  // TODO Make this work again, using priorities
250  // TGRSIOptions::AddInputMidasFile(line);
251  } else if(type.compare("ARRAYPOS") == 0 || type.compare("HPGEPOS") == 0) {
252  std::istringstream ss(line);
253  double temp_double;
254  ss >> temp_double;
255  Get()->SetHPGeArrayPosition(temp_double);
256  } else if(type.compare("BADCYCLE") == 0) {
257  std::istringstream ss(line);
258  int tmp_int;
259  while(!(ss >> tmp_int).fail() ) {
260  Get()->AddBadCycle(tmp_int);
261  }
262  }
263  }
264 
265  if(strcmp(opt, "q") != 0) {
266  std::cout<<"parsed "<<linenumber<<" lines."<<std::endl;
267  std::cout<<DBLUE "\tArray Position (mm) = "<<DRED<<TRunInfo::HPGeArrayPosition()<<RESET_COLOR<<std::endl;
268  }
269  return true;

421 void TRunInfo::Add(TRunInfo* runinfo)
422 {
423  // add new run to list of runs (and check if the current run needs to be added)
424  if(fRunList.empty()) {
425  std::pair<int, int> currentPair = std::make_pair(fRunNumber, fSubRunNumber);
426  fRunList.push_back(currentPair);
427  }
428  std::pair<int, int> newPair = std::make_pair(runinfo->fRunNumber, runinfo->fSubRunNumber);
429  // check for dual entries
430  if(std::find(fRunList.begin(), fRunList.end(), newPair) != fRunList.end()) {
431  std::cerr<<DYELLOW<<"Warning, adding run "<<std::setfill('0')<<std::setw(5)<<newPair.first<<"_"<<std::setw(3)<<newPair.second<<std::setfill(' ')<<" again!"<<RESET_COLOR<<std::endl;
432  return;
433  }
434  fRunList.push_back(newPair);
435 
436  //std::cout<<"adding run "<<runinfo->fRunNumber<<", sub run "<<runinfo->fSubRunNumber<<" to run "<<fRunNumber<<", sub run "<<fSubRunNumber<<std::endl;
437  // add the run length together
438  if(runinfo->fRunLength > 0) {
439  if(fRunLength > 0) {
440  fRunLength += runinfo->fRunLength;
441  } else {
442  fRunLength = runinfo->fRunLength;
443  }
444  }
445 
446  if(runinfo->fRunNumber != fRunNumber) {
447  // check if the added run is an increment of the current run number (if the run number is set)
448  if(fRunNumber != 0) {
449  if(runinfo->fRunNumber + 1 == fRunNumber) {
450  //std::cout<<"found second run ("<<runinfo->fRunNumber<<") before current run ("<<fRunNumber<<")"<<std::endl;
451  // use runinfo as first run and the current run as last
452  fFirstRunNumber = runinfo->fRunNumber;
453  fLastRunNumber = fRunNumber;
454  fRunStart = runinfo->fRunStart; // no need to set run stop
455  } else if(runinfo->fRunNumber - 1 == fRunNumber) {
456  //std::cout<<"found second run ("<<runinfo->fRunNumber<<") after current run ("<<fRunNumber<<")"<<std::endl;
457  // use runinfo as last run and the current run as first
458  fFirstRunNumber = fRunNumber;
459  fLastRunNumber = runinfo->fRunNumber;
460  fRunStop = runinfo->fRunStop; // no need to set run start
461  } else {
462  //std::cout<<"found second run ("<<runinfo->fRunNumber<<") non-consecutive to run ("<<fRunNumber<<")"<<std::endl;
463  // run start and stop don't make a lot of sense with non-consecutive runs (?)
464  fRunStart = 0.;
465  fRunStop = 0.;
466  // still need to keep some kind of information about the runs (e.g. to create filenames)
467  // by keeping first and last the exact same it is still obvious that these are not consecutive runs
468  fFirstRunNumber = fRunNumber;
469  fLastRunNumber = fRunNumber;
470  }
471  // the run number is meaningful only when the run numbers are the same
472  fRunNumber = 0;
473  fSubRunNumber = -1;
474  } else if(fFirstRunNumber != 0 && fLastRunNumber != 0 && fFirstRunNumber != fLastRunNumber) {
475  // if we already have a (good) range of runs, check if runinfo fits at the beginning or the end
476  if(runinfo->fRunNumber + 1 == fFirstRunNumber) {
477  //std::cout<<"found another run ("<<runinfo->fRunNumber<<") before first run ("<<fFirstRunNumber<<")"<<std::endl;
478  // use runinfo as first run
479  fFirstRunNumber = runinfo->fRunNumber;
480  fRunStart = runinfo->fRunStart;
481  } else if(runinfo->fRunNumber - 1 == fLastRunNumber) {
482  //std::cout<<"found another run ("<<runinfo->fRunNumber<<") after last run ("<<fLastRunNumber<<")"<<std::endl;
483  // use runinfo as last run
484  fLastRunNumber = runinfo->fRunNumber;
485  fRunStop = runinfo->fRunStop;
486  } else if(runinfo->fRunNumber == fFirstRunNumber || runinfo->fRunNumber == fLastRunNumber) {
487  //std::cout<<"found another sub(?) run part of runs ("<<fFirstRunNumber<<" - "<<fLastRunNumber<<")"<<std::endl;
488  // found probably another subrun of a run already added
489  // since we do not keep track of all subruns we have to assume this is in order
490  // so we only update the run start or stop if necessary
491  //std::cout<<"changing run start/stop from "<<std::setw(16)<<fRunStart<<"/"<<std::setw(16)<<fRunStop<<" to ";
492  if(fRunStop < runinfo->fRunStop) fRunStop = runinfo->fRunStop;
493  if(fRunStart > runinfo->fRunStart) fRunStart = runinfo->fRunStart;
494  //std::cout<<std::setw(16)<<fRunStart<<"/"<<std::setw(16)<<fRunStop<<std::endl;
495  } else {
496  //std::cout<<"found another run ("<<runinfo->fRunNumber<<") non-consecutive to runs ("<<fFirstRunNumber<<" - "<<fLastRunNumber<<")"<<std::endl;
497  // run start and stop don't make a lot of sense with non-consecutive runs (?)
498  fRunStart = 0.;
499  fRunStop = 0.;
500  // still need to keep some kind of information about the runs (e.g. to create filenames)
501  // by keeping first and last the exact same it is still obvious that these are not consecutive runs
502  fLastRunNumber = fFirstRunNumber;
503  }
504  } else {
505  // the run number is zero, and we do not have a (good) range, so there is nothing to do.
506  //std::cout<<"found another run ("<<runinfo->fRunNumber<<") non-consecutive run ("<<fFirstRunNumber<<" - "<<fLastRunNumber<<")"<<std::endl;
507  }
508  } else if(fSubRunNumber != -1) {
509  // check if the added sub run is an increment of the current run number
510  if(runinfo->fSubRunNumber + 1 == fSubRunNumber) {
511  //std::cout<<"found second sub run ("<<runinfo->fSubRunNumber<<") before current sub run ("<<fSubRunNumber<<")"<<std::endl;
512  // if the run numbers are the same and we have subsequent sub runs we can update the run start
513  fRunStart = runinfo->fRunStart;
514  fFirstSubRunNumber = runinfo->fSubRunNumber;
515  fLastSubRunNumber = fSubRunNumber;
516  } else if(runinfo->fSubRunNumber - 1 == fSubRunNumber) {
517  //std::cout<<"found second sub run ("<<runinfo->fSubRunNumber<<") after current sub run ("<<fSubRunNumber<<")"<<std::endl;
518  // if the run numbers are the same and we have subsequent sub runs we can update the run stop
519  fRunStop = runinfo->fRunStop;
520  fFirstSubRunNumber = fSubRunNumber;
521  fLastSubRunNumber = runinfo->fSubRunNumber;
522  } else {
523  //std::cout<<"found second sub run ("<<runinfo->fSubRunNumber<<") non-consecutive to current sub run ("<<fSubRunNumber<<")"<<std::endl;
524  // with multiple non-sequential subruns added, the sub run number and start/stop have no meaning anymore
525  fRunStart = 0.;
526  fRunStop = 0.;
527  }
528  // sub run number is only meaningful if it's the only sub run
529  fSubRunNumber = -1;
530  } else {
531  // we have the same run with a range of sub-runs already added, so check if this once fits at the end or the beginning
532  if(runinfo->fSubRunNumber + 1 == fFirstSubRunNumber) {
533  //std::cout<<"found another sub run ("<<runinfo->fSubRunNumber<<") before first sub run ("<<fFirstSubRunNumber<<")"<<std::endl;
534  // use runinfo as first run
535  fFirstSubRunNumber = runinfo->fSubRunNumber;
536  fRunStart = runinfo->fRunStart;
537  } else if(runinfo->fSubRunNumber - 1 == fLastSubRunNumber) {
538  //std::cout<<"found another sub run ("<<runinfo->fSubRunNumber<<") after last sub run ("<<fLastSubRunNumber<<")"<<std::endl;
539  // use runinfo as last run
540  fLastSubRunNumber = runinfo->fSubRunNumber;
541  fRunStop = runinfo->fRunStop;
542  } else {
543  //std::cout<<"found another sub run ("<<runinfo->fSubRunNumber<<") non-consecutive to sub runs ("<<fFirstSubRunNumber<<" - "<<fLastSubRunNumber<<")"<<std::endl;
544  // with multiple non-sequential subruns added, the sub run number and start/stop have no meaning anymore
545  fRunStart = 0.;
546  fRunStop = 0.;
547  fFirstSubRunNumber = -1;
548  fLastSubRunNumber = -1;
549  }
