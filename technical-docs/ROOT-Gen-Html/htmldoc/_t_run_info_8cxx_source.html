192 Bool_t TRunInfo::ParseInputData(const char* inputdata, Option_t* opt)
193 {
194  // A helper function to parse the run info file.
195 
196  std::istringstream infile(inputdata);
197  std::string line;
198  int linenumber = 0;
199 
200  // Parse the info file.
201  while(!std::getline(infile, line).fail() ) {
202  linenumber++;
203  trim(&line);
204  size_t comment = line.find("//");
205  if(comment != std::string::npos) {
206  line = line.substr(0, comment);
207  }
208  if(line.length() == 0u) {
209  continue;
210  }
211 
212  size_t ntype = line.find(':');
213  if(ntype == std::string::npos) {
214  continue; // no seperator, not useful.
215  }
216 
217  std::string type = line.substr(0, ntype);
218  line = line.substr(ntype + 1, line.length());
219  trim(&line);
220  int j = 0;
221  while(type[j] != 0) {
222  char c = *(type.c_str() + j);
223  c = toupper(c);
224  type[j++] = c;
225  }
226  if(type.compare("CAL") == 0 || type.compare("CALFILE") == 0) {
227  // TODO Make this work again, using priorities
228  // TGRSIOptions::AddInputCalFile(line);
229  } else if(type.compare("MID") == 0 || type.compare("MIDAS") == 0 || type.compare("MIDASFILE") == 0) {
230  // TODO Make this work again, using priorities
231  // TGRSIOptions::AddInputMidasFile(line);
232  } else if(type.compare("ARRAYPOS") == 0 || type.compare("HPGEPOS") == 0) {
233  std::istringstream ss(line);
234  double temp_double;
235  ss >> temp_double;
236  Get()->SetHPGeArrayPosition(temp_double);
237  } else if(type.compare("BADCYCLE") == 0) {
238  std::istringstream ss(line);
239  int tmp_int;
240  while(!(ss >> tmp_int).fail() ) {
241  Get()->AddBadCycle(tmp_int);
242  }
243  }
244  }
245 
246  if(strcmp(opt, "q") != 0) {
247  printf("parsed %i lines.\n", linenumber);
248  printf(DBLUE "\tArray Position (mm) = " DRED "%lf" RESET_COLOR "\n", TRunInfo::HPGeArrayPosition());
249  }
250  return true;
251 }
