169  /// The Process() function is called for each entry in the tree (or possibly
170  /// keyed object in the case of PROOF) to be processed. The entry argument
171  /// specifies which entry in the currently loaded tree is to be processed.
172  /// It can be passed to either TGRSISelector::GetEntry() or TBranch::GetEntry()
173  /// to read either all or the required parts of the data. When processing
174  /// keyed objects with PROOF, the object is already loaded and is available
175  /// via the fObject pointer.
176  ///
177  /// This function should contain the "body" of the analysis. It can contain
178  /// simple or elaborate selection criteria, run algorithms on the data
179  /// of the event and typically fill histograms.
180  ///
181  /// The processing can be stopped by calling Abort().
182  ///
183  /// Use fStatus to set the return value of TTree::Process().
184  ///
185  /// The return value is currently not used
186 
187  static TFile* current_file = nullptr;
188  if(current_file != fChain->GetCurrentFile()) {
189  current_file = fChain->GetCurrentFile();
190  std::cout<<"Starting to sort: "<<current_file->GetName()<<std::endl;
191  TChannel::ReadCalFromFile(current_file);
192  TGRSIOptions::AnalysisOptions()->ReadFromFile(current_file);
193  }
194 
195  fChain->GetEntry(entry);
196  try {

216 void TGRSISelector::Terminate()
217 {
218  /// The Terminate() function is the last function to be called during
219  /// a query. It always runs on the client, it can be used to present
220  /// the results graphically or save the results to file.
221  TGRSIOptions* options = TGRSIOptions::Get();
222  if(fRunInfo == nullptr) {
223  fRunInfo = TRunInfo::Get();
224  std::cout<<"replaced null run info with:"<<std::endl;
225  fRunInfo->Print();
226  }
227  Int_t runNumber = fRunInfo->RunNumber();
228  Int_t subRunNumber = fRunInfo->SubRunNumber();
229 
230  TFile* outputFile;
231  if(runNumber != 0 && subRunNumber != -1) {
232  // both run and subrun number set => single file processed
233  std::cout<<"Using run "<<runNumber<<" subrun "<<subRunNumber<<std::endl;
234  outputFile = new TFile(Form("%s%05d_%03d.root", fOutputPrefix.c_str(), runNumber, subRunNumber), "RECREATE");
235  if(!outputFile->IsOpen()) {
236  std::cerr<<"Failed to open output file "<<Form("%s%05d_%03d.root", fOutputPrefix.c_str(), runNumber, subRunNumber)<<"!"<<std::endl<<std::endl;
237  return;
238  }
239  } else if(runNumber != 0) {
240  // multiple subruns of a single run
241  std::cout<<"Using run "<<runNumber<<" subruns "<<fRunInfo->FirstSubRunNumber()<<" - "<<fRunInfo->LastSubRunNumber()<<std::endl;
242  outputFile = new TFile(Form("%s%05d_%03d-%03d.root", fOutputPrefix.c_str(), runNumber, fRunInfo->FirstSubRunNumber(), fRunInfo->LastSubRunNumber()), "RECREATE");
243  if(!outputFile->IsOpen()) {
244  std::cerr<<"Failed to open output file "<<Form("%s%05d_%03d-%03d.root", fOutputPrefix.c_str(), runNumber, fRunInfo->FirstSubRunNumber(), fRunInfo->LastSubRunNumber())<<"!"<<std::endl<<std::endl;
245  return;
246  }
247  } else {
248  // multiple runs
249  std::cout<<"Using runs "<<fRunInfo->FirstRunNumber()<<" - "<<fRunInfo->LastRunNumber()<<std::endl;
250  outputFile = new TFile(Form("%s%05d-%05d.root", fOutputPrefix.c_str(), fRunInfo->FirstRunNumber(), fRunInfo->LastRunNumber()), "RECREATE");
251  if(!outputFile->IsOpen()) {
252  std::cerr<<"Failed to open output file "<<Form("%s%05d-%05d.root", fOutputPrefix.c_str(), fRunInfo->FirstRunNumber(), fRunInfo->LastRunNumber())<<"!"<<std::endl<<std::endl;
253  return;
254  }
255  }
256 
257  outputFile->cd();
258  fOutput->Write();
259  fRunInfo->Write();
260  if(fPpg != nullptr) {
261  fPpg->Write();
262  } else {
263  std::cerr<<"failed to find TPPG, can't write it!"<<std::endl;
264  }
265  options->AnalysisOptions()->WriteToFile(outputFile);
266  TChannel::WriteToRoot();
267  outputFile->Close();
268 }
