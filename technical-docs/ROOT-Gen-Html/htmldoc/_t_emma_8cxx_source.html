74 void TEmma::AddFragment(const std::shared_ptr<const TFragment>& frag, TChannel* chan)
75 {
76  if (frag == nullptr || chan == nullptr) {
77  return;
78  }
79  TEmmaHit dethit( * frag);
80  if (chan->GetMnemonic()->SubSystem() == TMnemonic::EMnemonic::kI) { // IC ADC Data
81  fEmmaICHits.push_back(std::move(dethit));
82  } else if (chan->GetMnemonic()->SubSystem() == TMnemonic::EMnemonic::kS) { // Si at focal plane
83  fEmmaSiHits.push_back(std::move(dethit));
84  } else if (chan->GetMnemonic()->SubSystem() == TMnemonic::EMnemonic::kT) { // EMMA focal plane Trigger
85  fEmmaTriggerHits.push_back(std::move(dethit));
86  } else if (chan->GetMnemonic()->SubSystem() == TMnemonic::EMnemonic::kP) { // PGAC
87  if (chan->GetMnemonic()->CollectedCharge() == TMnemonic::EMnemonic::kN) { // Anode data
88  switch (chan->GetMnemonic()->OutputSensor()) {
89  case TMnemonic::EMnemonic::kX:
90  fEmmaAnodeHits.push_back(std::move(dethit)); // Anode ADC data
91  break;
92 
93  case TMnemonic::EMnemonic::kT:
94  dethit.SetTdcNumber(frag->GetSegment());
95  fEmmaTdcHits.push_back(std::move(dethit)); // PGAC Anode TDC data
96  break;
97 
98  default:
99  break;
100  }
101  } else if (chan->GetMnemonic()->CollectedCharge() == TMnemonic::EMnemonic::kP) { // PGAC Cathode TDC data
102  switch (chan->GetMnemonic()->OutputSensor()) {
103  case TMnemonic::EMnemonic::kL:
104  dethit.SetTdcNumber(10);
105  break;
106 
107  case TMnemonic::EMnemonic::kR:
108  dethit.SetTdcNumber(11);
109  break;
110 
111  case TMnemonic::EMnemonic::kU:
112  dethit.SetTdcNumber(12);
113  break;
114 
115  case TMnemonic::EMnemonic::kD:
116  dethit.SetTdcNumber(13);
117  break;
118 
119  default:
120  break;
121  };
122  fEmmaTdcHits.push_back(std::move(dethit));
123  }
124  } else if (chan->GetMnemonic()->SubSystem() == TMnemonic::EMnemonic::kO) { // ORTEC SSBs at target position
125  fEmmaSSBHits.push_back(std::move(dethit));
126  } else return;
127 }

129 TVector3 TEmma::GetPosition(double left, double right, double top, double bottom, double delayL, double delayR, double delayT, double delayB )
130 {
131  // Calculates recoil position from PGAC TDC data including delays
132  double Xdiff = (left+delayL) - (right+delayR);
133  double Xsum = (left) + (right);
134  double Ydiff = (bottom+delayB) - (top+delayT);
135  double Ysum = (bottom) + (top);
136 
137  double Xpos = ( Xdiff / Xsum )*fXlength;
138  double Ypos = ( Ydiff / Ysum )*fYlength;
139 
140  return TVector3(Xpos, Ypos, 1);
141 }

203 void TEmma::BuildHits()
204 {
205  // Everything below is reproduced from the EMMA sort code given to me by Nick Esker
206  // The EMMA PGAC has all cathode signals chained together in the X and Y direction with readouts at each end fed into a TDC.
207  // Build hits subtracts the trigger time (from an anode wire) and returns a left/right/up/down value which is used in GetPosition()
208 
209  std::vector <double> tdcArray;
210  std::vector <double> icArray;
211  if (fEmmaTdcHits.size() > 4 ) { // Require a Good hit to contain only the PGAC TDC signals
212  TEmmaHit * hit = new TEmmaHit();
213  for (size_t i = 0; i < fEmmaTdcHits.size(); ++i) {
214  hit->SetTimeStamp(fEmmaTdcHits[i].GetTimeStamp());
215  hit->SetAddress(fEmmaTdcHits[i].GetAddress());
216  if (fEmmaTdcHits[i].GetTdcNumber() < 10) tdcArray.push_back(fEmmaTdcHits[i].GetEnergy());
217  if (fEmmaTdcHits[i].GetTdcNumber() == 10) hit->SetLeft(fEmmaTdcHits[i].GetEnergy());
218  if (fEmmaTdcHits[i].GetTdcNumber() == 11) hit->SetRight(fEmmaTdcHits[i].GetEnergy());
219  if (fEmmaTdcHits[i].GetTdcNumber() == 12) hit->SetTop(fEmmaTdcHits[i].GetEnergy());
220  if (fEmmaTdcHits[i].GetTdcNumber() == 13) hit->SetBottom(fEmmaTdcHits[i].GetEnergy());
221  }
222 
223  if (tdcArray.size() != 0) {
224  fAnodeTrigger = * std::min_element(tdcArray.begin(), tdcArray.end());
225  if (hit->GetLeft() != 0 && hit->GetRight() != 0 && hit->GetTop() != 0 && hit->GetBottom() != 0 && fAnodeTrigger != 0) {
226 
227  hit->SetLeft((hit->GetLeft() - fAnodeTrigger));
228  hit->SetRight((hit->GetRight() - fAnodeTrigger));
229  hit->SetTop((hit->GetTop() - fAnodeTrigger));
230  hit->SetBottom((hit->GetBottom() - fAnodeTrigger));
231  hit->SetAnodeTrigger(fAnodeTrigger);
232  fHits.push_back(std::move(hit));
233  } else {
234  printf("TDC Array Failed \n");
235  fFail = 0;
236  if (hit->GetLeft() == 0) fFail++;
237  if (hit->GetRight() == 0) fFail++;
238  if (hit->GetTop() == 0) fFail++;
239  if (hit->GetBottom() == 0) fFail++;
240  hit->SetFailedFill(fFail);
241  fHits.push_back(std::move(hit));
242  }
243  } else {
244  return;
245  }
246  }
247  return;
248 }
