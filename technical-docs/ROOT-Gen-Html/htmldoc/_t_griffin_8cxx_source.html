146 Int_t TGriffin::GetAddbackMultiplicity() {
147  // Automatically builds the addback hits using the fAddbackCriterion (if the size of the fAddbackHits vector is zero) and return the number of addback hits.
148  if(fGriffinHits.size() == 0) {
149  return 0;
150  }
151  //if the addback has been reset, clear the addback hits
152  if((fGriffinBits & kIsAddbackSet) == 0x0) {
153  fAddbackHits.clear();
154  }
155  if(fAddbackHits.size() == 0) {
156  // use the first griffin hit as starting point for the addback hits
157  fAddbackHits.push_back(fGriffinHits[0]);
158  fAddbackFrags.push_back(1);
159 
160  // loop over remaining griffin hits
161  size_t i, j;
162  for(i = 1; i < fGriffinHits.size(); ++i) {
163  // check for each existing addback hit if this griffin hit should be added
164  for(j = 0; j < fAddbackHits.size(); ++j) {
165  if(fAddbackCriterion(fAddbackHits[j], fGriffinHits[i])) {
166  fAddbackHits[j].Add(&(fGriffinHits[i]));
167  fAddbackFrags[j]++;
168  break;
169  }
170  }
171  if(j == fAddbackHits.size()) {
172  fAddbackHits.push_back(fGriffinHits[i]);
173  fAddbackFrags.push_back(1);
174  }
175  }
176  SetBitNumber(kIsAddbackSet, true);
177  }
178 
179  return fAddbackHits.size();
180 }

236 TVector3 TGriffin::GetPosition(int DetNbr,int CryNbr, double dist ) {
237  //Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
238  //This is calculated to the most likely interaction point within the crystal.
239  if(DetNbr>16)
240  return TVector3(0,0,1);
241 
242  TVector3 temp_pos(gCloverPosition[DetNbr]);
243 
244  //Interaction points may eventually be set externally. May make these members of each crystal, or pass from waveforms.
245  Double_t cp = 26.0; //Crystal Center Point mm.
246  Double_t id = 45.0;//45.0; //Crystal interaction depth mm.
247  //Set Theta's of the center of each DETECTOR face
248  ////Define one Detector position
249  TVector3 shift;
250  switch(CryNbr) {
251  case 0:
252  shift.SetXYZ(-cp,cp,id);
253  break;
254  case 1:
255  shift.SetXYZ(cp,cp,id);
256  break;
257  case 2:
258  shift.SetXYZ(cp,-cp,id);
259  break;
260  case 3:
261  shift.SetXYZ(-cp,-cp,id);
262  break;
263  default:
264  shift.SetXYZ(0,0,1);
265  break;
266  };
267  shift.RotateY(temp_pos.Theta());
268  shift.RotateZ(temp_pos.Phi());
269 
270  temp_pos.SetMag(dist);
271 
272  return (temp_pos + shift);
273
