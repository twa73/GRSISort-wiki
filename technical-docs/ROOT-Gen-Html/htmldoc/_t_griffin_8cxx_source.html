407 Short_t TGriffin::GetAddbackMultiplicity(const EGainBits& gain_type)
408 {
409  // Automatically builds the addback hits using the fAddbackCriterion (if the size of the fAddbackHits vector is zero)
410  // and return the number of addback hits.
411  if(!IsCrossTalkSet(gain_type)) {
412  // Calculate Cross Talk on each hit
413  FixCrossTalk(gain_type);
414  }
415  auto& hit_vec = GetHitVector(gain_type);
416  auto& ab_vec = GetAddbackVector(gain_type);
417  auto& frag_vec = GetAddbackFragVector(gain_type);
418  if(hit_vec.empty()) {
419  return 0;
420  }
421  // if the addback has been reset, clear the addback hits
422  if(!IsAddbackSet(gain_type)) {
423  for(auto hit : ab_vec) {
424  delete hit;
425  }
426  ab_vec.clear();
427  frag_vec.clear();
428  }
429  if(ab_vec.empty()) {
430  CreateAddback(hit_vec, ab_vec, frag_vec);
431  SetAddback(gain_type, true);
432  }
433 
434  return ab_vec.size();
435 }

457 void TGriffin::AddFragment(const std::shared_ptr<const TFragment>& frag, TChannel* chan)
458 {
459  // Builds the GRIFFIN Hits directly from the TFragment. Basically, loops through the hits for an event and sets
460  // observables.
461  // This is done for both GRIFFIN and it's suppressors.
462  if(frag == nullptr || chan == nullptr) {
463  return;
464  }
465  const TMnemonic* mnemonic = chan->GetMnemonic();
466  if(mnemonic == nullptr) {
467  std::cerr<<"Trying to add fragment to TGriffin w/o mnemonic in TChannel!"<<std::endl;
468  return;
469  }
470 
471  if(mnemonic->SubSystem() != TMnemonic::EMnemonic::kG) {
472  std::cerr<<__PRETTY_FUNCTION__<<": not a GRIFFIN detector: "<<static_cast<std::underlying_type<TMnemonic::EMnemonic>::type>(mnemonic->SubSystem())<<std::endl;
473  return;
474  }
475 
476  // only create the hit if we have a HPGe signal (A or B), not a suppressor (S)
477  switch(mnemonic->OutputSensor()) {
478  case TMnemonic::EMnemonic::kA:
479  {
480  TGriffinHit* hit = new TGriffinHit(*frag);
481  GetHitVector(EGainBits::kLowGain).push_back(hit);
482  }
483  break;
484  case TMnemonic::EMnemonic::kB:
485  {
486  TGriffinHit* hit = new TGriffinHit(*frag);
487  GetHitVector(EGainBits::kHighGain).push_back(hit);
488  }
489  break;
490  default:
491  std::cout<<"output sensor "<<static_cast<std::underlying_type<TMnemonic::EMnemonic>::type>(mnemonic->OutputSensor())<<", skipping it"<<std::endl;
492  break;
493  };
494 }

496 TVector3 TGriffin::GetPosition(int DetNbr, int CryNbr, double dist)
497 {
498  // Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
499  // This is calculated to the most likely interaction point within the crystal.
500  if(DetNbr > 16) {

590 Double_t TGriffin::CTCorrectedEnergy(const TGriffinHit* const hit_to_correct, const TGriffinHit* const other_hit,
591  Bool_t time_constraint)
592 {
593  if((hit_to_correct == nullptr) || (other_hit == nullptr)) {
594  std::cerr<<"One of the hits is invalid in TGriffin::CTCorrectedEnergy"<<std::endl;
595  return 0;
596  }
597 
598  if(time_constraint) {
599  // Figure out if this passes the selected window
600  if(TMath::Abs(other_hit->GetTime() - hit_to_correct->GetTime()) >
601  TGRSIOptions::AnalysisOptions()->AddbackWindow()) { // placeholder
602  return hit_to_correct->GetEnergy();
603  }
604  }
605 
606  if(hit_to_correct->GetDetector() != other_hit->GetDetector()) {
607  return hit_to_correct->GetEnergy();
608  }
609  static bool been_warned[256] = {false};
610  double fixed_energy = hit_to_correct->GetEnergy();
611  try {
612  if(hit_to_correct->GetChannel() != nullptr) {
613  fixed_energy -= hit_to_correct->GetChannel()->GetCTCoeff().at(other_hit->GetCrystal()) * other_hit->GetNoCTEnergy();
614  }
615  } catch(const std::out_of_range& oor) {
616  int id = 16 * hit_to_correct->GetDetector() + 4 * hit_to_correct->GetCrystal() + other_hit->GetCrystal();
617  if(!been_warned[id]) {
618  been_warned[id] = true;
619  std::cerr<<DRED<<"Missing CT correction for Det: "<<hit_to_correct->GetDetector()
620  <<" Crystals: "<<hit_to_correct->GetCrystal()<<" "<<other_hit->GetCrystal()<<" (id "<<id<<")"<<RESET_COLOR<<std::endl;
621  }
622  return hit_to_correct->GetEnergy();
623  }
624 
625  return fixed_energy;
626 }

639 {
