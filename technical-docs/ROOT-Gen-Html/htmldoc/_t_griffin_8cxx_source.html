274 Int_t TGriffin::GetAddbackMultiplicity(const Int_t &gain_type) {
275  // Automatically builds the addback hits using the fAddbackCriterion (if the size of the fAddbackHits vector is zero) and return the number of addback hits.
276  if(!IsCrossTalkSet(gain_type)){
277  //Calculate Cross Talk on each hit
278  FixCrossTalk(gain_type);
279  }
280  auto hit_vec = GetHitVector(gain_type);
281  auto ab_vec = GetAddbackVector(gain_type);
282  auto frag_vec = GetAddbackFragVector(gain_type);
283  if(hit_vec->size() == 0) {
284  return 0;
285  }
286  //if the addback has been reset, clear the addback hits
287  if(!IsAddbackSet(gain_type)) {
288  ab_vec->clear();
289  }
290  if(ab_vec->size() == 0) {
291  // use the first griffin hit as starting point for the addback hits
292  ab_vec->push_back(hit_vec->at(0));
293  frag_vec->push_back(1);
294 
295  // loop over remaining griffin hits
296  size_t i, j;
297  for(i = 1; i < hit_vec->size(); ++i) {
298  // check for each existing addback hit if this griffin hit should be added
299  for(j = 0; j < ab_vec->size(); ++j) {
300  if(fAddbackCriterion(ab_vec->at(j), hit_vec->at(i))) {
301  ab_vec->at(j).Add(&(hit_vec->at(i)));
302  frag_vec->at(j)++;
303  break;
304  }
305  }
306  if(j == ab_vec->size()) {
307  ab_vec->push_back(hit_vec->at(i));
308  frag_vec->push_back(1);
309  }
310  }
311  SetAddback(gain_type,true);
312  }
313 
314 
315  return ab_vec->size();
316 }

362 TVector3 TGriffin::GetPosition(int DetNbr,int CryNbr, double dist ) {
363  //Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
364  //This is calculated to the most likely interaction point within the crystal.
365  if(DetNbr>16)
366  return TVector3(0,0,1);
367 
368  TVector3 temp_pos(gCloverPosition[DetNbr]);
369 
370  //Interaction points may eventually be set externally. May make these members of each crystal, or pass from waveforms.
371  Double_t cp = 26.0; //Crystal Center Point mm.
372  Double_t id = 45.0;//45.0; //Crystal interaction depth mm.
373  //Set Theta's of the center of each DETECTOR face
374  ////Define one Detector position
375  TVector3 shift;
376  switch(CryNbr) {
377  case 0:
378  shift.SetXYZ(-cp,cp,id);
379  break;
380  case 1:
381  shift.SetXYZ(cp,cp,id);
382  break;
383  case 2:
384  shift.SetXYZ(cp,-cp,id);
385  break;
386  case 3:
387  shift.SetXYZ(-cp,-cp,id);
388  break;
389  default:
390  shift.SetXYZ(0,0,1);
391  break;
392  };
393  shift.RotateY(temp_pos.Theta());
394  shift.RotateZ(temp_pos.Phi());
395 
396  temp_pos.SetMag(dist);
397 
398  return (temp_pos + shift);

477  auto hit_vec = GetHitVector(gain_type);
478  if(hit_vec->size() < 2) {
479  SetCrossTalk(gain_type,true);
480  return;
481  }
