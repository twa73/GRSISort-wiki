97 void TGriffinHit::Add(const TDetectorHit* hit)
98 {
99  const TGriffinHit* griffinHit = dynamic_cast<const TGriffinHit*>(hit);
100  if(griffinHit == nullptr) {
101  throw std::runtime_error("trying to add non-griffin hit to griffin hit!");
102  }
103  // add another griffin hit to this one (for addback),
104  // using the time and position information of the one with the higher energy
105  if(!CompareEnergy(this, griffinHit)) {
106  SetCfd(griffinHit->GetCfd());
107  SetTime(griffinHit->GetTime());
108  // SetPosition(griffinHit->GetPosition());
109  SetAddress(griffinHit->GetAddress());
110  } else {
111  SetTime(GetTime());
112  }
113  SetEnergy(GetEnergy() + griffinHit->GetEnergy());
114  // this has to be done at the very end, otherwise GetEnergy() might not work
115  SetCharge(0);
116  // Add all of the pileups.This should be changed when the max number of pileups changes
117  if((NPileUps() + griffinHit->NPileUps()) < 4) {
118  SetNPileUps(NPileUps() + griffinHit->NPileUps());
119  } else {
120  SetNPileUps(3);
121  }
122  if((PUHit() + griffinHit->PUHit()) < 4) {
123  SetPUHit(PUHit() + griffinHit->PUHit());
124  } else {
125  SetPUHit(3);
126  }
127  // KValue is somewhate meaningless in addback, so I am using it as an indicator that a piledup griffinHit was added-back RD
128  if(GetKValue() > griffinHit->GetKValue()) {
129  SetKValue(griffinHit->GetKValue());
130  }
131 }
