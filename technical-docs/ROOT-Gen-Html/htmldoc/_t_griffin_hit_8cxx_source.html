137 void TGriffinHit::Add(const TGriffinHit *hit) {
138  // add another griffin hit to this one (for addback),
139  // using the time and position information of the one with the higher energy
140  if(!CompareEnergy(this,hit)) {
141  this->SetCfd(hit->GetCfd());
142  this->SetTime(hit->GetTime());
143  this->SetPosition(hit->GetPosition());
144  this->SetAddress(hit->GetAddress());
145  }
146 
147  this->SetEnergy(this->GetEnergy() + hit->GetEnergy());
148  //this has to be done at the very end, otherwise this->GetEnergy() might not work
149  this->SetCharge(0);
150 }
