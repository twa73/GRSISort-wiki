121 void TGriffinHit::Add(const TGriffinHit *hit) {
122  // add another griffin hit to this one (for addback),
123  // using the time and position information of the one with the higher energy
124  if(!CompareEnergy(this,hit)) {
125  this->SetCfd(hit->GetCfd());
126  this->SetTime(hit->GetTime());
127  //this->SetPosition(hit->GetPosition());
128  this->SetAddress(hit->GetAddress());
129  }
130  this->SetEnergy(this->GetEnergy() + hit->GetEnergy());
131  //this has to be done at the very end, otherwise this->GetEnergy() might not work
132  this->SetCharge(0);
133  //Add all of the pileups.This should be changed when the max number of pileups changes
134  if((this->NPileUps() + hit->NPileUps()) < 4){
135  this->SetNPileUps(this->NPileUps() + hit->NPileUps());
136  }
137  else{
138  this->SetNPileUps(3);
139  }
140  if((this->PUHit() + hit->PUHit()) < 4){
141  this->SetPUHit(this->PUHit() + hit->PUHit());
142  }
143  else{
144  this->SetPUHit(3);
145  }
146 }
