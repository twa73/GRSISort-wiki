48 Double_t TGRSIDetectorHit::GetTime(const UInt_t& correction_flag,Option_t* opt) const {
49  if(IsTimeSet())
50  return fTime;
51 
52  TChannel* chan = GetChannel();
53  if(!chan) {
54  Error("GetTime","No TChannel exists for address 0x%08x",GetAddress());
55  return 10.*(static_cast<Double_t>((GetTimeStamp()) + gRandom->Uniform()));
56  }
57 
58  switch(chan->GetDigitizerType()) {
59  Double_t dTime;
60  case TMnemonic::kGRF16:
61  dTime = (GetTimeStamp()&(~0x3ffff))*10. + (GetCfd() + gRandom->Uniform())/1.6;//CFD is in 10/16th of a nanosecond
62  return dTime - 10.*(chan->GetTZero(GetEnergy()));
63  case TMnemonic::kGRF4G:
64  dTime = GetTimeStamp()*10. + (fCfd>>22) + ((fCfd & 0x3fffff) + gRandom->Uniform())/256.;
65  return dTime - 10.*(chan->GetTZero(GetEnergy()));
66  default:
67  dTime = static_cast<Double_t>((GetTimeStamp()) + gRandom->Uniform());
68  return 10.*(dTime - chan->GetTZero(GetEnergy()));
69  }
70  return 0.;
71 }
