95 void TGRSIDetectorInformation::Set()
96 {
97  /// Sets the run info. This figures out what systems are available.
98  for(auto iter = TChannel::GetChannelMap()->begin(); iter != TChannel::GetChannelMap()->end(); iter++) {
99  std::string channelname = iter->second->GetName();
100 
101  // check if we have an old TIG digitizer, in that case sort by trigger ID (instead of time stamp)
102  // don't need to check again if we already found one, so include check for fSortByTriggerId being true
103  if(!fSortByTriggerId && (iter->second->GetDigitizerType() == EDigitizer::kTIG10 || iter->second->GetDigitizerType() == EDigitizer::kTIG64)) {
104  fSortByTriggerId = true;
105  }
106  // detector system type.
107  // for more info, see: https://www.triumf.info/wiki/tigwiki/index.php/Detector_Nomenclature
108  switch(static_cast<const TGRSIMnemonic*>(iter->second->GetMnemonic())->System()) {
109  case TGRSIMnemonic::ESystem::kTigress:
110  SetTigress();
111  break;
112  case TGRSIMnemonic::ESystem::kSharc:
113  SetSharc();
114  break;
115  case TGRSIMnemonic::ESystem::kTriFoil:
116  SetTriFoil();
117  break;
118  case TGRSIMnemonic::ESystem::kRF:
119  SetRF();
120  break;
121  case TGRSIMnemonic::ESystem::kCSM:
122  SetCSM();
123  break;
124  case TGRSIMnemonic::ESystem::kTip:
125  SetTip();
126  break;
127  case TGRSIMnemonic::ESystem::kGriffin:
128  SetGriffin();
129  break;
130  case TGRSIMnemonic::ESystem::kSceptar:
131  SetSceptar();
132  break;
133  case TGRSIMnemonic::ESystem::kPaces:
134  SetPaces();
135  break;
136  case TGRSIMnemonic::ESystem::kLaBr:
137  SetDante();
138  break;
139  case TGRSIMnemonic::ESystem::kZeroDegree:
140  SetZeroDegree();
141  break;
142  case TGRSIMnemonic::ESystem::kDescant:
143  SetDescant();
144  break;
145  case TGRSIMnemonic::ESystem::kBgo:
146  SetBgo();

149  SetEmma();
150  break;

152  SetGeneric();

154  default:

156  if(system.compare("SP") == 0) {

160  SetBambino();
