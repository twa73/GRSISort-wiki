146 Double_t TAB3Peak::BackgroundFunction(Double_t *dim, Double_t *par)
147 {
148  Double_t x = dim[0]; // channel number used for fitting
149  Double_t height = par[0]; // height of photopeak
150  Double_t c = par[1]; // Peak Centroid of non skew gaus

152  Double_t step = par[7]; // Size of the step function;
