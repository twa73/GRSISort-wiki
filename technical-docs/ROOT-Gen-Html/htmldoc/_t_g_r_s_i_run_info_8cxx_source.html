405 Bool_t TGRSIRunInfo::ParseInputData(const char *inputdata,Option_t *opt) {
406  //A helper function to parse the run info file.
407 
408  std::istringstream infile(inputdata);
409  std::string line;
410  int linenumber = 0;
411 
412  //Parse the info file.
413  while(std::getline(infile, line)) {
414  linenumber++;
415  trim(&line);
416  size_t comment = line.find("//");
417  if(comment != std::string::npos) {
418  line = line.substr(0, comment);
419  }
420  if(!line.length())
421  continue;
422 
423  size_t ntype = line.find(":");
424  if(ntype == std::string::npos) //no seperator, not useful.
425  continue;
426 
427  std::string type = line.substr(0, ntype);
428  line = line.substr(ntype + 1, line.length());
429  trim(&line);
430  int j = 0;
431  while (type[j]) {
432  char c = *(type.c_str() + j);
433  c = toupper(c);
434  type[j++] = c;
435  }
436  if( type.compare("BW")==0 || type.compare("BUILDWINDOW")==0 ) {
437  std::istringstream ss(line);
438  long int temp_bw; ss >> temp_bw;
439  Get()->SetBuildWindow(temp_bw);
440  } else if( type.compare("WF")==0 || type.compare("WAVEFORMFIT")==0) {
441  std::istringstream ss(line);
442  bool temp_wff; ss >> temp_wff;
443  Get()->SetWaveformFitting(temp_wff);
444  } else if( type.compare("MW")==0 || type.compare("MOVINGWINDOW")==0) {
445  std::istringstream ss(line);
446  bool temp_mw; ss >> temp_mw;
447  Get()->SetMovingWindow(temp_mw);
448  } else if( type.compare("ABW")==0 || type.compare("ADDBACKWINDOW")==0 || type.compare("ADDBACK")==0 ) {
449  std::istringstream ss(line);
450  double temp_abw; ss >> temp_abw;
451  Get()->SetAddBackWindow(temp_abw);
452  } else if( type.compare("CAL")==0 || type.compare("CALFILE")==0 ) {
453  TGRSIOptions::AddInputCalFile(line);
454  } else if( type.compare("MID")==0 || type.compare("MIDAS")==0 || type.compare("MIDASFILE")==0 ) {
455  TGRSIOptions::AddInputMidasFile(line);
456  } else if( type.compare("ARRAYPOS")==0 || type.compare("HPGEPOS")==0) {
457  std::istringstream ss(line);
458  double temp_double; ss >> temp_double;
459  Get()->SetHPGeArrayPosition(temp_double);
460  } else if( type.compare("DESCANTANCILLARY") == 0) {
461  std::istringstream ss(line);
462  int temp_int; ss >> temp_int;
463  Get()->SetDescantAncillary(temp_int);
464  }
465  }
466 
467  if(strcmp(opt,"q")) {
468  printf("parsed %i lines.\n",linenumber);
469  printf(DBLUE"\tBuild Window (10 ns) = " DRED "%lu" RESET_COLOR "\n",TGRSIRunInfo::BuildWindow());
470  printf(DBLUE"\tMoving Window = " DRED "%s" RESET_COLOR "\n",TGRSIRunInfo::IsMovingWindow() ? "TRUE" : "FALSE");
471  printf(DBLUE"\tAddBack Window (ns) = " DRED "%.01f" RESET_COLOR "\n",TGRSIRunInfo::AddBackWindow());
472  printf(DBLUE"\tArray Position (mm) = " DRED "%lf" RESET_COLOR "\n",TGRSIRunInfo::HPGeArrayPosition());
473  printf(DBLUE"\tWaveform Fitting = " DRED "%s" RESET_COLOR "\n",TGRSIRunInfo::IsWaveformFitting() ? "TRUE" : "FALSE");
474  }
475  return true;
476 }
