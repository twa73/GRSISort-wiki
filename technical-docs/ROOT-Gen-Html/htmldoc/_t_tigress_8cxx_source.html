117 Int_t TTigress::GetAddbackMultiplicity() {
118  // Automatically builds the addback hits using the addback_criterion
119  // (if the size of the addback_hits vector is zero) and return the number of addback hits.
120  if(fTigressHits.size() == 0) {
121  return 0;
122  }
123  //if the addback has been reset, clear the addback hits
124  if(!TestBit(kAddbackSet)) {
125  fAddbackHits.clear();
126  } else {
127  return fAddbackHits.size();
128  }
129 
130  // use the first tigress hit as starting point for the addback hits
131  fAddbackHits.push_back(fTigressHits[0]);
132  fAddbackHits.back().SumHit(&(fAddbackHits.back()));//this sets the last position
133  fAddbackFrags.push_back(1);
134 
135  // loop over remaining tigress hits
136  size_t i, j;
137  for(i = 1; i < fTigressHits.size(); ++i) {
138  // check for each existing addback hit if this tigress hit should be added
139  for(j = 0; j < fAddbackHits.size(); ++j) {
140  if(fAddbackCriterion(fAddbackHits[j], fTigressHits[i])) {
141  fAddbackHits[j].SumHit(&(fTigressHits[i]));
142  fAddbackFrags[j]++;
143  break;
144  }
145  }
146  if(j == fAddbackHits.size()) {
147  fAddbackHits.push_back(fTigressHits[i]);
148  fAddbackHits.back().SumHit(&(fAddbackHits.back()));//this sets the last position
149  fAddbackFrags.push_back(1);
150  }
151  }
152  SetBit(kAddbackSet, true);
153 
154  return fAddbackHits.size();
155 }
