102 Int_t TTigress::GetAddbackMultiplicity() {
103  // Automatically builds the addback hits using the addback_criterion (if the size of the addback_hits vector is zero) and return the number of addback hits.
104  if(fTigressHits.size() == 0) {
105  return 0;
106  }
107  //if the addback has been reset, clear the addback hits
108  if((fTigressBits & kIsAddbackSet) == 0x0) {
109  fAddbackHits.clear();
110  }
111  if(fAddbackHits.size() == 0) {
112  // use the first tigress hit as starting point for the addback hits
113  fAddbackHits.push_back(fTigressHits[0]);
114  fAddbackHits.back().SumHit(&(fAddbackHits.back()));//this sets the last position
115  fAddbackFrags.push_back(1);
116 
117  // loop over remaining tigress hits
118  size_t i, j;
119  for(i = 1; i < fTigressHits.size(); ++i) {
120  // check for each existing addback hit if this tigress hit should be added
121  for(j = 0; j < fAddbackHits.size(); ++j) {
122  if(fAddbackCriterion(fAddbackHits[j], fTigressHits[i])) {
123  fAddbackHits[j].SumHit(&(fTigressHits[i]));
124  fAddbackFrags[j]++;
125  break;
126  }
127  }
128  if(j == fAddbackHits.size()) {
129  fAddbackHits.push_back(fTigressHits[i]);
130  fAddbackHits.back().SumHit(&(fAddbackHits.back()));//this sets the last position
131  fAddbackFrags.push_back(1);
132  }
133  }
134  SetBitNumber(kIsAddbackSet, true);
135  }
136 
137  return fAddbackHits.size();
138 }
