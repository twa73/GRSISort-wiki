112 bool TZeroDegreeHit::AnalyzeWaveform()
113 {
114  /// Calculates the cfd time from the waveform
115  bool error = false;
116  if(fWaveform.empty()) {
117  return false; // Error!
118  }
119 
120  std::vector<Int_t> baselineCorrections(8, 0);
121  std::vector<Short_t> smoothedWaveform;
122 
123  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as
124  // integers)
125  unsigned int interpolationSteps = 256;
126  int delay = 2;
127  double attenuation = 24. / 64.;
128  int halfsmoothingwindow = 0; // 2*halfsmoothingwindow + 1 = number of samples in moving window.
129 
130  // baseline algorithm: correct each adc with average of first two samples in that adc
131  for(size_t i = 0; i < 8 && i < fWaveform.size(); ++i) {
132  baselineCorrections[i] = fWaveform[i];
133  }
134  for(size_t i = 8; i < 16 && i < fWaveform.size(); ++i) {
135  baselineCorrections[i - 8] =
136  ((baselineCorrections[i - 8] + fWaveform[i]) + ((baselineCorrections[i - 8] + fWaveform[i]) > 0 ? 1 : -1)) >>
137  1;
138  }
139  for(size_t i = 0; i < fWaveform.size(); ++i) {
140  fWaveform[i] -= baselineCorrections[i % 8];
141  }
142 
143  SetCfd(CalculateCfd(attenuation, delay, halfsmoothingwindow, interpolationSteps));
144 
145  SetCharge(CalculatePartialSum().back());
146 
147  return !error;
148 }

159 Int_t TZeroDegreeHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfsmoothingwindow,
160  unsigned int interpolationSteps, std::vector<Short_t>& monitor)
161 {
162  /// Used when calculating the CFD from the waveform
163 
164  Short_t monitormax = 0;
165 
166  bool armed = false;
167 
168  Int_t cfd = 0;
169 
170  std::vector<Short_t> smoothedWaveform;
171 
172  if(fWaveform.empty()) {
173  return INT_MAX; // Error!
174  }
175 
176  if(static_cast<unsigned int>(fWaveform.size()) > delay + 1) {
177 
178  if(halfsmoothingwindow > 0) {
179  smoothedWaveform = TZeroDegreeHit::CalculateSmoothedWaveform(halfsmoothingwindow);
180  } else {
181  smoothedWaveform = fWaveform;
182  }
183 
184  monitor.resize(smoothedWaveform.size() - delay);
185  monitor[0] = attenuation * smoothedWaveform[delay] - smoothedWaveform[0];
186  if(monitor[0] > monitormax) {
187  armed = true;
188  monitormax = monitor[0];
189  }
190 
191  for(unsigned int i = delay + 1; i < smoothedWaveform.size(); ++i) {
192  monitor[i - delay] = attenuation * smoothedWaveform[i] - smoothedWaveform[i - delay];
193  if(monitor[i - delay] > monitormax) {
194  armed = true;
195  monitormax = monitor[i - delay];
196  } else {
197  if(armed && monitor[i - delay] < 0) {
198  armed = false;
199  if(monitor[i - delay - 1] - monitor[i - delay] != 0) {
200  // Linear interpolation.
201  cfd = (i - delay - 1) * interpolationSteps +
202  (monitor[i - delay - 1] * interpolationSteps) / (monitor[i - delay - 1] - monitor[i - delay]);
203  } else {
204  // Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
205  cfd = 0;
206  }
207  }
208  }
209  }
210  } else {
211  monitor.resize(0);
212  }
213 
214  if(TGRSIOptions::Get()->Debug()) {
215  fCfdMonitor = monitor;
216  }
217 
218  // correct for remainder between the 100MHz timestamp and the 125MHz start of the waveform
219  // we save this in the upper bits, otherwise we can't correct the waveform themselves
220  cfd = (cfd & 0x3fffff) | (static_cast<Int_t>(fCfd) & 0x7c00000);
221 
222  return cfd;
223 }

225 std::vector<Short_t> TZeroDegreeHit::CalculateSmoothedWaveform(unsigned int halfsmoothingwindow)
226 {

228

230  return std::vector<Short_t>(); // Error!
231  }
232
