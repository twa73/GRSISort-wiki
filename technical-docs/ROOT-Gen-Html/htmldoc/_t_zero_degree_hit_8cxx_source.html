105 bool TZeroDegreeHit::AnalyzeWaveform() {
106  ///Calculates the cfd time from the waveform
107  bool error = false;
108  if(fWaveform.empty()) {
109  return false; //Error!
110  }
111 
112  std::vector<Int_t> baselineCorrections (8, 0);
113  std::vector<Short_t> smoothedWaveform;
114 
115  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as integers)
116  unsigned int interpolationSteps = 256;
117  int delay = 2;
118  double attenuation = 24./64.;
119  int halfsmoothingwindow = 0; //2*halfsmoothingwindow + 1 = number of samples in moving window.
120 
121  // baseline algorithm: correct each adc with average of first two samples in that adc
122  for(size_t i = 0; i < 8 && i < fWaveform.size(); ++i) {
123  baselineCorrections[i] = fWaveform[i];
124  }
125  for(size_t i = 8; i < 16 && i < fWaveform.size(); ++i) {
126  baselineCorrections[i-8] = ((baselineCorrections[i-8] + fWaveform[i]) + ((baselineCorrections[i-8] + fWaveform[i]) > 0 ? 1 : -1)) >> 1;
127  }
128  for(size_t i = 0; i < fWaveform.size(); ++i) {
129  fWaveform[i] -= baselineCorrections[i%8];
130  }
131 
132  SetCfd(CalculateCfd(attenuation, delay, halfsmoothingwindow, interpolationSteps));
133 
134  SetCharge(CalculatePartialSum().back());
135 
136  return !error;
137 }

146 Int_t TZeroDegreeHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfsmoothingwindow, unsigned int interpolationSteps, std::vector<Short_t> &monitor) {
147  ///Used when calculating the CFD from the waveform
148 
149  Short_t monitormax = 0;
150 
151  bool armed = false;
152 
153  Int_t cfd = 0;
154 
155  std::vector<Short_t> smoothedWaveform;
156 
157  if(fWaveform.empty()) {
158  return INT_MAX; //Error!
159  }
160 
161  if((unsigned int)fWaveform.size() > delay+1) {
162 
163  if(halfsmoothingwindow > 0) {
164  smoothedWaveform = TZeroDegreeHit::CalculateSmoothedWaveform(halfsmoothingwindow);
165  } else {
166  smoothedWaveform = fWaveform;
167  }
168 
169  monitor.resize(smoothedWaveform.size()-delay);
170  monitor[0] = attenuation*smoothedWaveform[delay]-smoothedWaveform[0];
171  if(monitor[0] > monitormax) {
172  armed = true;
173  monitormax = monitor[0];
174  }
175 
176  for(unsigned int i = delay + 1; i < smoothedWaveform.size(); ++i) {
177  monitor[i-delay] = attenuation*smoothedWaveform[i]-smoothedWaveform[i-delay];
178  if(monitor[i-delay] > monitormax) {
179  armed=true;
180  monitormax = monitor[i-delay];
181  } else {
182  if(armed == true && monitor[i-delay] < 0) {
183  armed = false;
184  if(monitor[i-delay-1]-monitor[i-delay] != 0) {
185  //Linear interpolation.
186  cfd = (i-delay-1)*interpolationSteps + (monitor[i-delay-1]*interpolationSteps)/(monitor[i-delay-1]-monitor[i-delay]);
187  }
188  else {
189  //Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
190  cfd = 0;
191  }
192  }
193  }
194  }
195  } else {
196  monitor.resize(0);
197  }
198 
199  if(TGRSIOptions::Get()->Debug()) {
200  fCfdMonitor = monitor;
201  }
202 
203  // correct for remainder between the 100MHz timestamp and the 125MHz start of the waveform
204  // we save this in the upper bits, otherwise we can't correct the waveform themselves
205  cfd = (cfd & 0x3fffff) | (fCfd & 0x7c00000);
206 
207  return cfd;
208 
209 }
