68 bool TZeroDegreeHit::AnalyzeWaveform() {
69  ///Calculates the cfd time from the waveform
70  bool error = false;
71  std::vector<Short_t> *waveform = GetWaveform();
72  if(waveform->empty()) {
73  return false; //Error!
74  }
75 
76  std::vector<Int_t> baselineCorrections (8, 0);
77  std::vector<Short_t> smoothedWaveform;
78 
79  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as integers)
80  unsigned int interpolationSteps = 256;
81  int delay = 8;
82  double attenuation = 24./64.;
83  int halfsmoothingwindow = 0; //2*halfsmoothingwindow + 1 = number of samples in moving window.
84 
85  // baseline algorithm: correct each adc with average of first two samples in that adc
86  for(size_t i = 0; i < 8 && i < waveform->size(); ++i) {
87  baselineCorrections[i] = (*waveform)[i];
88  }
89  for(size_t i = 8; i < 16 && i < waveform->size(); ++i) {
90  baselineCorrections[i-8] = ((baselineCorrections[i-8] + (*waveform)[i]) + ((baselineCorrections[i-8] + (*waveform)[i]) > 0 ? 1 : -1)) >> 1;
91  }
92  for(size_t i = 0; i < waveform->size(); ++i) {
93  (*waveform)[i] -= baselineCorrections[i%8];
94  }
95 
96  this->SetCfd(CalculateCfd(attenuation, delay, halfsmoothingwindow, interpolationSteps));
97 
98  return !error;
99 }

108 Int_t TZeroDegreeHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfsmoothingwindow, unsigned int interpolationSteps, std::vector<Short_t> &monitor) {
109  ///Used when calculating the CFD from the waveform
110 
111  Short_t monitormax = 0;
112  std::vector<Short_t> *waveform = GetWaveform();
113 
114  bool armed = false;
115 
116  Int_t cfd = 0;
117 
118  std::vector<Short_t> smoothedWaveform;
119 
120  if(waveform->empty()) {
121  return INT_MAX; //Error!
122  }
123 
124  if((unsigned int)waveform->size() > delay+1) {
125 
126  if(halfsmoothingwindow > 0) {
127  smoothedWaveform = TZeroDegreeHit::CalculateSmoothedWaveform(halfsmoothingwindow);
128  } else {
129  smoothedWaveform = *waveform;
130  }
131 
132  monitor.resize(smoothedWaveform.size()-delay);
133  monitor[0] = attenuation*smoothedWaveform[delay]-smoothedWaveform[0];
134  if(monitor[0] > monitormax) {
135  armed = true;
136  monitormax = monitor[0];
137  }
138 
139  for(unsigned int i = delay + 1; i < smoothedWaveform.size(); ++i) {
140  monitor[i-delay] = attenuation*smoothedWaveform[i]-smoothedWaveform[i-delay];
141  if(monitor[i-delay] > monitormax) {
142  armed=true;
143  monitormax = monitor[i-delay];
144  } else {
145  if(armed == true && monitor[i-delay] < 0) {
146  armed = false;
147  if(monitor[i-delay-1]-monitor[i-delay] != 0) {
148  //Linear interpolation.
149  cfd = (i-delay-1)*interpolationSteps + (monitor[i-delay-1]*interpolationSteps)/(monitor[i-delay-1]-monitor[i-delay]);
150  }
151  else {
152  //Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
153  cfd = 0;
154  }
155  }
156  }
157  }
158  } else {
159  monitor.resize(0);
160  }
161 
162  return cfd;
