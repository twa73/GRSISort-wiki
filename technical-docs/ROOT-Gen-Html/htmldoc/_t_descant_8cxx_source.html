145 TVector3 TDescant::GetPosition(int DetNbr, double dist)
146 {
147  // Gets the position vector for detector DetNbr
148  // dist is only used when detectors are in the ancillary positions.
149 
150  if(TRunInfo::GetDetectorInformation() != nullptr && static_cast<TGRSIDetectorInformation*>(TRunInfo::GetDetectorInformation())->DescantAncillary()) {
151  if(DetNbr > 8) {
152  return TVector3(0, 0, 1);

154  TVector3 temp_pos(gAncillaryPosition[DetNbr]);

156  return temp_pos;
