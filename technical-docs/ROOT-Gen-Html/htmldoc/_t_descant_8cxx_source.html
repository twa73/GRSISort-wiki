169 void TDescant::AddFragment(TFragment* frag, MNEMONIC* mnemonic) {
170  ///Builds the DESCANT Hits directly from the TFragment. Basically, loops through the data for an event and sets observables.
171  ///This is done for both DESCANT and it's suppressors.
172  if(frag == NULL || mnemonic == NULL) {
173  return;
174  }
175 
176  for(size_t i = 0; i < frag->Charge.size(); ++i) {
177  TDescantHit hit;
178  hit.SetAddress(frag->ChannelAddress);
179  hit.SetTimeStamp(frag->GetTimeStamp());
180  hit.SetCfd(frag->GetCfd(i));
181  hit.SetCharge(frag->GetCharge(i));
182  hit.SetZc(frag->GetZc(i));
183  hit.SetCcShort(frag->GetCcShort(i));
184  hit.SetCcLong(frag->GetCcLong(i));
185 
186  if(TDescant::SetWave()){
187  if(frag->wavebuffer.size() == 0) {
188  //printf("Warning, TDescant::SetWave() set, but data waveform size is zero!\n");
189  }
190  if(0) {
191  std::vector<Short_t> x;
192  //Need to reorder waveform data for S1507 data from December 2014
193  //All pairs of samples are swapped.
194  //The first two samples are also delayed by 8.
195  //We choose to throw out the first 2 samples (junk) and the last 6 samples (convience)
196  x = frag->wavebuffer;
197  size_t length = x.size() - (x.size()%8);
198  Short_t temp;
199 
200  if(length > 8) {
201  for(size_t i = 0; i < length-8; i+=8) {
202  x[i] = x[i+9];
203  x[i+1] = x[i+8];
204  temp = x[i+2];
205  x[i+2] = x[i+3];
206  x[i+3] = temp;
207  temp = x[i+4];
208  x[i+4] = x[i+5];
209  x[i+5] = temp;
210  temp = x[i+6];
211  x[i+6] = x[i+7];
212  x[i+7] = temp;
213  }
214  x.resize(length-8);
215  }
216  hit.SetWaveform(x);
217  }
218  else {
219  hit.SetWaveform(frag->wavebuffer);
220  }
221  if(hit.GetWaveform()->size() > 0) {
222  // printf("Analyzing waveform, current cfd = %d, psd = %d\n",hit.GetCfd(),hit.GetPsd());
223  hit.AnalyzeWaveform();
224  // bool analyzed = hit.AnalyzeWaveform();
225  // printf("%s analyzed waveform, cfd = %d, psd = %d\n",analyzed ? "successfully":"unsuccessfully",hit.GetCfd(),hit.GetPsd());
226  }
227  }
228 
229  AddHit(&hit);
230  }
