77 double GetCentroidTPeak(TH1* hst, double xmin, double xmax)
78 {
79  // for this to be robust, we have to be a bit smarter
80  // particularly, we need the fit window to be smarter
81  double guesscentroid = GetCentroidMaxBin(hst,xmin,xmax);
82  double fwhm = FWHM(hst,xmin,xmax);
83 
84  // initial fit
85  TPeak* peak;
86  hst->GetXaxis()->SetRangeUser(xmin,xmax);
87  if(guesscentroid-1.5*fwhm>xmin && guesscentroid+1.5*fwhm<xmax) {
88  peak = new TPeak(guesscentroid,guesscentroid-1.5*fwhm,guesscentroid+1.5*fwhm);
89  } else {
90  peak = new TPeak(guesscentroid,xmin,xmax);
91  }
92  peak->Fit(hst,"Q");
93  double centroid = peak->GetCentroid();
94  hst->GetXaxis()->UnZoom();
95  return centroid;

107 double GetRoughGain(TH1* h, double largepeak, double mindistance, bool twopeaks=kFALSE, double largepeak2 = 0.0)
108 {
109  TSpectrum* spec2 = new TSpectrum(2);
110 
111  // search for big keV peak (or peaks)
112  spec2->Search(h);
113  double posbigb;
114  if(!twopeaks) {
115  if(spec2->GetNPeaks()==2) {
116  // if we get two peaks, check the height of the peaks to find the actual tallest one
117  // using GetMaximum here instead of GetPositionY() because sometimes TSpectrum is weird.
118  h->GetXaxis()->SetRangeUser(spec2->GetPositionX()[0]-mindistance,spec2->GetPositionX()[0]+mindistance);
119  double mag0 = h->GetMaximum();
120  h->GetXaxis()->SetRangeUser(spec2->GetPositionX()[1]-mindistance,spec2->GetPositionX()[1]+mindistance);
121  double mag1 = h->GetMaximum();
122  h->GetXaxis()->UnZoom();
123  posbigb = spec2->GetPositionX()[0];
124  if(mag1>mag0) posbigb = spec2->GetPositionX()[1];
125  } else {
126  posbigb = spec2->GetPositionX()[0];
127  }
128  } else {
129  if(spec2->GetNPeaks()!=2) {
130  posbigb = spec2->GetPositionX()[0];
131  std::cout <<"Warning: Two peaks were specified, but not found. Setting only peak found to " <<largepeak <<std::endl;
132  } else {
133  if(std::min(largepeak,largepeak2) == largepeak) {
134  posbigb = std::min(spec2->GetPositionX()[0],spec2->GetPositionX()[1]);
135  } else {
136  posbigb = std::max(spec2->GetPositionX()[0],spec2->GetPositionX()[1]);
137  }
138  }
139  }
140  // calculate rough gain
141  double gain = posbigb/largepeak;
142  return gain;
143 }

237 void create_gainmatch_graphs(const char* histFileName, int minchannel, int maxchannel, std::vector<int> channelsToSkip, double largepeak, std::vector<double> peaks, EType type = EType::kTSpectrum, double mindistance=20e3, bool roughenergy=kTRUE, bool twopeaks=kFALSE, double largepeak2 = 0.0)
238 {
239  // minchannel and max channel are inclusive values. most loops go from i=min to i<=maxchannel
240 
241  // largepeak is the value of largest peak in spectrum (in keV)
242 
243  // the values in the vector peaks do not have to be exact values, as the point here is not to run an energy calibration
244  // advice: do not use peaks that have another peak near-by
245 
246  // mindistance should be in the same units as your original histogram (probably charge)
247  // it is the window size for peak searching
248 
249  // if you want a rough enery match, your largepeak and peaks values need to be in keV
250  // set roughenergy to true if you want a rough energy match as well
251 
252  /*---------------------------------------------------------------*/
253 
254  TFile* f = new TFile(histFileName,"update");
255  if(f == nullptr) {
256  printf("Failed to open file '%s'!\n",histFileName);
257  return;
258  }
259  if(!f->IsOpen()) {
260  printf("Failed to open file '%s'!\n",histFileName);
261  return;
262  }
263 
264  // fit histograms, create graphs, save slopes and offsets
265  TGraph** g = new TGraph*[maxchannel+1];
266 
267  // create and save the graphs
268  for(int i=minchannel;i<=maxchannel;i++) {
269  std::cout <<"\t" <<i <<std::endl;
270  bool skipChannel = kFALSE;
271  for(auto skip : channelsToSkip) {
272  if(i == skip) skipChannel = kTRUE;
273  }
274  if(skipChannel) continue;
275  std::cout <<"Creating graph for channel " <<i <<std::endl;
276  TH1F* h = (TH1F*) f->Get(Form("hst%i",i)); // grab histogram
277  if(h==0) {
278  std::cout <<"Error: hst" <<i <<" does not exist. Please check your calib_hst file and your inputs to GainMatchGRIFFIN." <<std::endl;
279  continue;
280  }
281 
282  // calculate rough gain
283  double gain = GetRoughGain(h,largepeak,mindistance,twopeaks,largepeak2);
284  // std::cout <<"Rough gain for channel " <<i <<" is " <<gain <<std::endl;
285 
286  // calculate rough positions of peaks
287  std::vector<double> peakguesses;
288  for(auto peak : peaks) {
289  peakguesses.push_back(gain*peak);
290  }
291 
292  // gainmatch
293  g[i] = gainmatch_peaks(h,peaks,peakguesses,mindistance,type);
294 
295  // check the first histogram and change the peak values if necessary
296  if(i==minchannel) {
297  if(roughenergy) g[i]->Fit("pol1","q","goff");
298  for(int j=0;j<g[i]->GetN();j++) {
299  double x,y;
300  g[i]->GetPoint(j,x,y);
301  if(!roughenergy) {
302  peaks[j] = x;
303  g[i]->SetPoint(j,x,x);
304  } else {
305  peaks[j] = g[i]->GetFunction("pol1")->Eval(x);
306  g[i]->SetPoint(j,x,peaks[j]);
307  }
308  }
309  }
310  // format graph
311  if((i)%4==1) g[i]->SetMarkerColor(kBlue);
312  if((i)%4==2) g[i]->SetMarkerColor(kGreen);
313  if((i)%4==3) g[i]->SetMarkerColor(kRed);
314  if((i)%4==4) g[i]->SetMarkerColor(kBlack);
315  g[i]->SetMarkerStyle(20+(i)/4);
316  const char* hsttitle = h->GetTitle();
317  g[i]->SetTitle(hsttitle);
318  g[i]->SetName(Form("graph%i",i));
319  g[i]->Write();
320 
321  }
322  f->Close();
323 }

431  }

776 // this function calculates the full-width at X maximum, where x is given as a percent.
777 double FWXM(TH1* h, double xmin, double xmax, double percent = 0.5)
778 {
779  h->GetXaxis()->SetRangeUser(xmin,xmax);
780  double max = h->GetMaximum();
781  int maxbin = h->GetMaximumBin();
782  double bincontent = max;
783  int binbelow = maxbin;
784  double low,high;
785 
786  // rough estimation of background
787  int lowbin = h->FindBin(xmin);
788  int highbin = h->FindBin(xmax);
789  double bg = (h->GetBinContent(highbin)+h->GetBinContent(lowbin))/2.;
790 
791  // find lower limit for FWHM
792  while(bincontent>(max-bg)*percent+bg) {
793  --binbelow;
794  if(binbelow<lowbin) {
795  std::cout <<"Couldn't find good lower limit, returning 0." <<std::endl;
796  return 0.0;
797  }
798  bincontent = h->GetBinContent(binbelow);
799  }
800  double x1,y1,x2,y2,y3,m;
801  y3 = (max-bg)*percent+bg;
802  x1 = h->GetBinCenter(binbelow);
803  x2 = h->GetBinCenter(binbelow+1);
804  y1 = h->GetBinContent(binbelow);
805  y2 = h->GetBinContent(binbelow+1);
806  m = (y2-y1)/(x2-x1);
807  low = (y3-y1)/m+x1;
808 
809  binbelow = maxbin;
810  bincontent = max;
811  // find upper limit for FWHM
812  while(bincontent>(max-bg)*percent+bg) {
813  ++binbelow;
814  if(binbelow>highbin) {
815  std::cout <<"Couldn't find good upper limit, returning 0." <<std::endl;
816  return 0.0;
817  }
818  bincontent = h->GetBinContent(binbelow);
819  }
820  y3 = max/2.;
821  x1 = h->GetBinCenter(binbelow);
822  x2 = h->GetBinCenter(binbelow-1);
823  y1 = h->GetBinContent(binbelow);
824  y2 = h->GetBinContent(binbelow-1);
825  m = (y2-y1)/(x2-x1);
826  high = (y3-y1)/m+x1;
827 
828  h->GetXaxis()->UnZoom();
829 
830  return (high-low);
831 }
