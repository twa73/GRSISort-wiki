256 Int_t TTdrClover::GetAddbackMultiplicity()
257 {
258  /// Automatically builds the addback hits using the fAddbackCriterion (if the size of the fAddbackHits vector is zero)
259  /// and returns the number of addback hits.
260  auto& hit_vec = GetHitVector();
261  auto& ab_vec = GetAddbackVector();
262  auto& frag_vec = GetAddbackFragVector();
263  if(hit_vec.empty()) {
264  return 0;
265  }
266  // if the addback has been reset, clear the addback hits
267  if(!IsAddbackSet()) {
268  ab_vec.clear();
269  frag_vec.clear();
270  }
271  if(ab_vec.empty()) {
272  CreateAddback(hit_vec, ab_vec, frag_vec);
273  SetAddback(true);
274  }
275 
276  return ab_vec.size();
277 }

370 TVector3 TTdrClover::GetPosition(int DetNbr, int CryNbr, double dist)
371 {
372  // Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
373  // This is calculated to the most likely interaction point within the crystal.
374  if(DetNbr > 16) {
375  return TVector3(0, 0, 1);
376  }
377 
378  TVector3 temp_pos(gCloverPosition[DetNbr]);
379 
380  // Interaction points may eventually be set externally. May make these members of each crystal, or pass from
381  // waveforms.
382  Double_t cp = 26.0; // Crystal Center Point mm.
383  Double_t id = 45.0; // 45.0; //Crystal interaction depth mm.
384  // Set Theta's of the center of each DETECTOR face
385  ////Define one Detector position
386  TVector3 shift;
387  switch(CryNbr) {
388  case 0: shift.SetXYZ(-cp, cp, id); break;
389  case 1: shift.SetXYZ(cp, cp, id); break;
390  case 2: shift.SetXYZ(cp, -cp, id); break;
391  case 3: shift.SetXYZ(-cp, -cp, id); break;
392  default: shift.SetXYZ(0, 0, 1); break;
393  };
394  shift.RotateY(temp_pos.Theta());
395  shift.RotateZ(temp_pos.Phi());
396 
397  temp_pos.SetMag(dist);
398 
399  return (temp_pos + shift);
400 }
