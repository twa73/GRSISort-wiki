354 void GRootObjectManager::ExtractObjects(TCollection *list) {
355  TIter iter(list);
356  TObject *object = iter.Next(); //iter.Next() returns nullptr if there is no next iterator
357  while(object) {
358  //printf("object->GetName() = %s \n",object->GetName());
359  if(object->InheritsFrom("TFolder")) {
360  ExtractObjects(((TFolder*)object)->GetListOfFolders());
361  } else if(object->InheritsFrom("TDirectoryFile")) {
362  ExtractObjectsFromFile((TDirectoryFile*)object);
363  } else if(object->InheritsFrom("TDirectory")) {
364  ExtractObjects(((TDirectory*)object)->GetList());
365  } else {
366  AddObject(object);
367  }
368  object = iter.Next();
369  }
370  return;
371 }
