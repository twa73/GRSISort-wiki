163 bool TFragment::IsDetector(const char * prefix, Option_t *opt) const {
164  //Checks to see if the current fragment constains the same "prefix", for example "GRG"
165  //The option determines whether the channel should be:
166  // - C : The core of a segmented detector
167  // - S : The segments of a segmented detector
168  // - A : Low gain output of a detector
169  // - B : High gain output of a detectora
170  // If C or S are not given, default to C
171  // If A or B are not given, default to A
172  //Note that multiple options add to the output, so "CAB" would return the core with both high and low gain
173  //One should eventually add N,P,T options as well.
174  std::string pre = prefix;
175  TString option = opt;
176  std::string channame = this->GetName();
177  if(channame.length()<9)
178  return false;
179 
180  option.ToUpper();
181  //Could also do everything below with MNEMONIC Struct. This limits the amount of string processing that needs to be done
182  //Because it returns false after every potential failure while the mnemonic class sets all of the strings, and then checks
183  //for conditions.
184  if(!channame.compare(0,pre.length(),pre)) { //channame.BeginsWith(pre)){
185  if(option.Length()<1) //no option.
186  return true;
187  if(channame.length()>8) {
188  if(option.Contains("B") && (std::toupper(channame[9])==std::toupper('B')))
189  return true;
190  else if(option.Contains("A") && (std::toupper(channame[9])==std::toupper('A')))
191  return true;
192  }
193  if(option.Contains("C") && !channame.compare(7,2,"00"))
194  return true;
195  else if(option.Contains("S") && channame.compare(7,2,"00"))
196  return true;
197  } else
198  return false;
199 
200  return false;
201 }

204 Int_t TFragment::GetSharcMesyBoard() const {
205  int slave = (fAddress & 0x00f00000) >> 20;
206  int port = (fAddress & 0x00000f00) >> 8;
207  int channel = (fAddress & 0x000000ff);
208 
209  //printf("slave = 0x%08x port = 0x%08x channel = 0x%08x\n",slave,port,channel);
210 
211  if(slave!=0x1 && slave!=0x2)
212  return -1;
213  if(channel==0x1f) {
214  return (slave-1)*16 + (port-1)*2 + 1;
215  } else if(channel==0x3f) {

217  }
218  return -1;
219 }
