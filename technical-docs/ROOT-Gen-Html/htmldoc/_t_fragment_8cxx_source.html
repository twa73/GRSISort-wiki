181 bool TFragment::IsDetector(const char* prefix, Option_t* opt) const
182 {
183  // Checks to see if the current fragment constains the same "prefix", for example "GRG"
184  // The option determines whether the channel should be:
185  // - C : The core of a segmented detector
186  // - S : The segments of a segmented detector
187  // - A : Low gain output of a detector
188  // - B : High gain output of a detectora
189  // If C or S are not given, default to C
190  // If A or B are not given, default to A
191  // Note that multiple options add to the output, so "CAB" would return the core with both high and low gain
192  // One should eventually add N,P,T options as well.
193  std::string pre = prefix;
194  TString option = opt;
195  std::string channame = GetName();
196  if(channame.length() < 9) {
197  return false;
198  }
199 
200  option.ToUpper();
201  // Could also do everything below with TMnemonic. This limits the amount of string processing that needs to be done
202  // Because it returns false after every potential failure while the mnemonic class sets all of the strings, and then
203  // checks
204  // for conditions.
205  if(channame.compare(0, pre.length(), pre) == 0) { // channame.BeginsWith(pre)){
206  if(option.Length() < 1) { // no option.
207  return true;
208  }
209  if(channame.length() > 8) {
210  if(option.Contains("B") && (std::toupper(channame[9]) == std::toupper('B'))) {
211  return true;
212  }
213  if(option.Contains("A") && (std::toupper(channame[9]) == std::toupper('A'))) {
214  return true;
215  }
216  }
217  if(option.Contains("C") && (channame.compare(7, 2, "00") == 0)) {
218  return true;
219  }
220  if(option.Contains("S") && (channame.compare(7, 2, "00") != 0)) {
221  return true;
222  }
223  } else {
224  return false;
225  }
226 
227  return false;
228 }

230 Int_t TFragment::GetSharcMesyBoard() const
231 {
232  int slave = (fAddress & 0x00f00000) >> 20;
233  int port = (fAddress & 0x00000f00) >> 8;
234  int channel = (fAddress & 0x000000ff);
235 
236  // printf("slave = 0x%08x port = 0x%08x channel = 0x%08x\n",slave,port,channel);
237 
238  if(slave != 0x1 && slave != 0x2) {
239  return -1;
240  }
241  if(channel == 0x1f) {
242  return (slave - 1) * 16 + (port - 1) * 2 + 1;
243  }
244  if(channel == 0x3f) {
245  return (slave - 1) * 16 + (port - 1) * 2 + 2;
246  }
247  return -1;
248 }
