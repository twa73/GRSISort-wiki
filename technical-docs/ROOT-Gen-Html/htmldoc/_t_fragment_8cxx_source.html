270 bool TFragment::IsDetector(const char * prefix, Option_t *opt) const {
271  //Checks to see if the current fragment constains the same "prefix", for example "GRG"
272  //The option determines whether the channel should be:
273  // - C : The core of a segmented detector
274  // - S : The segments of a segmented detector
275  // - A : Low gain output of a detector
276  // - B : High gain output of a detectora
277  // If C or S are not given, default to C
278  // If A or B are not given, default to A
279  //Note that multiple options add to the output, so "CAB" would return the core with both high and low gain
280  //One should eventually add N,P,T options as well.
281  std::string pre = prefix;
282  TString option = opt;
283  std::string channame = this->GetName();
284  if(channame.length()<9)
285  return false;
286 
287  option.ToUpper();
288  //Could also do everything below with MNEMONIC Struct. This limits the amount of string processing that needs to be done
289  //Because it returns false after every potential failure while the mnemonic class sets all of the strings, and then checks
290  //for conditions.
291  if(!channame.compare(0,pre.length(),pre)) { //channame.BeginsWith(pre)){
292  if(option.Length()<1) //no option.
293  return true;
294  if(channame.length()>8) {
295  if(option.Contains("B") && (std::toupper(channame[9])==std::toupper('B')))
296  return true;
297  else if(option.Contains("A") && (std::toupper(channame[9])==std::toupper('A')))
298  return true;
299  }
300  if(option.Contains("C") && !channame.compare(7,2,"00"))
301  return true;
302  else if(option.Contains("S") && channame.compare(7,2,"00"))
303  return true;
304  } else
305  return false;
306 
307  return false;
308 }
