116 bool TSceptarHit::AnalyzeWaveform() {
117  //Calculates the cfd time from the waveform
118  bool error = false;
119  if(fWaveform.empty()) {
120  return false; //Error!
121  }
122 
123  std::vector<Int_t> baselineCorrections (8, 0);
124  std::vector<Short_t> smoothedWaveform;
125 
126  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as integers)
127  unsigned int interpolationSteps = 256;
128  int delay = 8;
129  double attenuation = 24./64.;
130  int halfsmoothingwindow = 0; //2*halfsmoothingwindow + 1 = number of samples in moving window.
131 
132  // baseline algorithm: correct each adc with average of first two samples in that adc
133  for(size_t i = 0; i < 8 && i < fWaveform.size(); ++i) {
134  baselineCorrections[i] = fWaveform[i];
135  }
136  for(size_t i = 8; i < 16 && i < fWaveform.size(); ++i) {
137  baselineCorrections[i-8] = ((baselineCorrections[i-8] + fWaveform[i]) + ((baselineCorrections[i-8] + fWaveform[i]) > 0 ? 1 : -1)) >> 1;
138  }
139  for(size_t i = 0; i < fWaveform.size(); ++i) {
140  fWaveform[i] -= baselineCorrections[i%8];
141  }
142 
143  SetCfd(CalculateCfd(attenuation, delay, halfsmoothingwindow, interpolationSteps));
144 
145  return !error;
146 }

155 Int_t TSceptarHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfsmoothingwindow, unsigned int interpolationSteps, std::vector<Short_t> &monitor) {
156  //Used when calculating the CFD from the waveform
157 
158  Short_t monitormax = 0;
159 
160  bool armed = false;
161 
162  Int_t cfd = 0;
163 
164  std::vector<Short_t> smoothedWaveform;
165 
166  if(fWaveform.empty()) {
167  return INT_MAX; //Error!
168  }
169 
170  if((unsigned int)fWaveform.size() > delay+1) {
171 
172  if(halfsmoothingwindow > 0) {
173  smoothedWaveform = TSceptarHit::CalculateSmoothedWaveform(halfsmoothingwindow);
174  } else {
175  smoothedWaveform = fWaveform;
176  }
177 
178  monitor.resize(smoothedWaveform.size()-delay);
179  monitor[0] = attenuation*smoothedWaveform[delay]-smoothedWaveform[0];
180  if(monitor[0] > monitormax) {
181  armed = true;
182  monitormax = monitor[0];
183  }
184 
185  for(unsigned int i = delay + 1; i < smoothedWaveform.size(); ++i) {
186  monitor[i-delay] = attenuation*smoothedWaveform[i]-smoothedWaveform[i-delay];
187  if(monitor[i-delay] > monitormax) {
188  armed=true;
189  monitormax = monitor[i-delay];
190  } else {
191  if(armed == true && monitor[i-delay] < 0) {
192  armed = false;
193  if(monitor[i-delay-1]-monitor[i-delay] != 0) {
194  //Linear interpolation.
195  cfd = (i-delay-1)*interpolationSteps + (monitor[i-delay-1]*interpolationSteps)/(monitor[i-delay-1]-monitor[i-delay]);
196  }
197  else {
198  //Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
199  cfd = 0;
200  }
201  }
202  }
203  }
204  } else {
205  monitor.resize(0);
206  }
207 
208  return cfd;
209 
210 }
