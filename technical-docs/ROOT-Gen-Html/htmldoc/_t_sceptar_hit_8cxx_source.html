72 bool TSceptarHit::AnalyzeWaveform() {
73  //Calculates the cfd time from the waveform
74  bool error = false;
75  std::vector<Short_t> *waveform = GetWaveform();
76  if(waveform->empty()) {
77  return false; //Error!
78  }
79 
80  std::vector<Int_t> baselineCorrections (8, 0);
81  std::vector<Short_t> smoothedWaveform;
82 
83  // all timing algorithms use interpolation with this many steps between two samples (all times are stored as integers)
84  unsigned int interpolationSteps = 256;
85  int delay = 8;
86  double attenuation = 24./64.;
87  int halfsmoothingwindow = 0; //2*halfsmoothingwindow + 1 = number of samples in moving window.
88 
89  // baseline algorithm: correct each adc with average of first two samples in that adc
90  for(size_t i = 0; i < 8 && i < waveform->size(); ++i) {
91  baselineCorrections[i] = (*waveform)[i];
92  }
93  for(size_t i = 8; i < 16 && i < waveform->size(); ++i) {
94  baselineCorrections[i-8] = ((baselineCorrections[i-8] + (*waveform)[i]) + ((baselineCorrections[i-8] + (*waveform)[i]) > 0 ? 1 : -1)) >> 1;
95  }
96  for(size_t i = 0; i < waveform->size(); ++i) {
97  (*waveform)[i] -= baselineCorrections[i%8];
98  }
99 
100  SetCfd(CalculateCfd(attenuation, delay, halfsmoothingwindow, interpolationSteps));
101 
102  return !error;
103 }

112 Int_t TSceptarHit::CalculateCfdAndMonitor(double attenuation, unsigned int delay, int halfsmoothingwindow, unsigned int interpolationSteps, std::vector<Short_t> &monitor) {
113  //Used when calculating the CFD from the waveform
114 
115  Short_t monitormax = 0;
116  std::vector<Short_t> *waveform = GetWaveform();
117 
118  bool armed = false;
119 
120  Int_t cfd = 0;
121 
122  std::vector<Short_t> smoothedWaveform;
123 
124  if(waveform->empty()) {
125  return INT_MAX; //Error!
126  }
127 
128  if((unsigned int)waveform->size() > delay+1) {
129 
130  if(halfsmoothingwindow > 0) {
131  smoothedWaveform = TSceptarHit::CalculateSmoothedWaveform(halfsmoothingwindow);
132  } else {
133  smoothedWaveform = *waveform;
134  }
135 
136  monitor.resize(smoothedWaveform.size()-delay);
137  monitor[0] = attenuation*smoothedWaveform[delay]-smoothedWaveform[0];
138  if(monitor[0] > monitormax) {
139  armed = true;
140  monitormax = monitor[0];
141  }
142 
143  for(unsigned int i = delay + 1; i < smoothedWaveform.size(); ++i) {
144  monitor[i-delay] = attenuation*smoothedWaveform[i]-smoothedWaveform[i-delay];
145  if(monitor[i-delay] > monitormax) {
146  armed=true;
147  monitormax = monitor[i-delay];
148  } else {
149  if(armed == true && monitor[i-delay] < 0) {
150  armed = false;
151  if(monitor[i-delay-1]-monitor[i-delay] != 0) {
152  //Linear interpolation.
153  cfd = (i-delay-1)*interpolationSteps + (monitor[i-delay-1]*interpolationSteps)/(monitor[i-delay-1]-monitor[i-delay]);
154  }
155  else {
156  //Should be impossible, since monitor[i-delay-1] => 0 and monitor[i-delay] > 0
157  cfd = 0;
158  }
159  }
160  }
161  }
162  } else {
163  monitor.resize(0);
164  }
165 
166  return cfd;
