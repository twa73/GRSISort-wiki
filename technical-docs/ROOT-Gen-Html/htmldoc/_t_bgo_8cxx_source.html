149 TVector3 TBgo::GetPosition(int DetNbr, int CryNbr, double dist)
150 {
151  // Gets the position vector for a crystal specified by CryNbr within Clover DetNbr at a distance of dist mm away.
152  // This is calculated to the most likely interaction point within the crystal.
153  if(DetNbr > 16) return TVector3(0, 0, 1);
154 
155  TVector3 temp_pos(gScintPosition[DetNbr]);
156 
157  // Interaction points may eventually be set externally. May make these members of each crystal, or pass from
158  // waveforms.
159  Double_t cp = 26.0; // Crystal Center Point mm.
160  Double_t id = 45.0; // 45.0; //Crystal interaction depth mm.
161  // Set Theta's of the center of each DETECTOR face
162  ////Define one Detector position
163  TVector3 shift;
164  switch(CryNbr) {
165  case 0: shift.SetXYZ(-cp, cp, id); break;
166  case 1: shift.SetXYZ(cp, cp, id); break;
167  case 2: shift.SetXYZ(cp, -cp, id); break;
168  case 3: shift.SetXYZ(-cp, -cp, id); break;
169  default: shift.SetXYZ(0, 0, 1); break;
170  };
171  shift.RotateY(temp_pos.Theta());
172  shift.RotateZ(temp_pos.Phi());
173 
174  temp_pos.SetMag(dist);
175 
176  return (temp_pos + shift);
177 }
