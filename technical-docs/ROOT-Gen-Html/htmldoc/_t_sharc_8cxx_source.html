202 TVector3 TSharc::GetPosition(int detector, int frontstrip, int backstrip, double X, double Y, double Z) {
203  int FrontDet = detector;
204  int FrontStr = frontstrip;
205  //int BackDet = detector;
206  int BackStr = backstrip;
207  int nrots = 0; // allows us to rotate into correct position
208 
209  TVector3 position;
210  TVector3 position_offset;
211  position_offset.SetXYZ(X,Y,Z);
212 
213  if(FrontDet>=5 && FrontDet<=8){ //forward box
214  nrots = FrontDet-4; // edited to make box 5 on the ceiling. assuming rotaing ccw around the +z axis!!
215  double x = fXposDB; // ?? x stays the same. first detector is aways defined in the y-z plane.
216  double y = - (fYminDB + (FrontStr+0.5)*fStripFPitch); // [(-36.0) - (+36.0)] // ?? add minus sign, reversve the order of the strips on the ds section.
217  double z = fZminDB + (BackStr+0.5)*fStripBPitch; // [(+9.0) - (+57.0)]
218  position.SetXYZ(x,y,z);
219  }
220  else if(FrontDet>=9 && FrontDet<=12){ //backward box
221  nrots = FrontDet-8; // edited to make box 5 on the ceiling. assuming rotaing ccw around the +z axis!!
222  double x = fXposUB;
223  double y = fYminUB + (FrontStr+0.5)*fStripFPitch; // [(-36.0) - (+36.0)]
224  double z = fZminUB - (BackStr+0.5)*fStripBPitch; // [(-5.0) - (-53.0)]
225  position.SetXYZ(x,y,z);
226  }
227  else if(FrontDet>=13){ // backward (upstream) QQQ
228  nrots = FrontDet-13;
229  double z = fZposUQ;
230  double rho = fRmaxUQ - (FrontStr+0.5)*fRingPitch; // [(+9.0) - (+41.0)]
231  double phi = (fPminUQ + (BackStr+0.5)*fSegmentPitch)*TMath::Pi()/180.0; // [(+2.0) - (+83.6)]
232  position.SetXYZ(rho*TMath::Sin(phi),rho*TMath::Cos(phi),z);
233  }
234  else if(FrontDet<=4){ // forward (downstream) QQQ
235  nrots = FrontDet-1;
236  double z = fZposDQ;
237  double rho = fRmaxDQ - (FrontStr+0.5)*fRingPitch; // [(+9.0) - (+41.0)]
238  double phi = (fPminDQ + (BackStr+0.5)*fSegmentPitch)*TMath::Pi()/180.0; // [(+6.4) - (+88.0)]
239  position.SetXYZ(rho*TMath::Sin(phi),rho*TMath::Cos(phi),z);
240  }
241 
242  position.RotateZ(TMath::Pi()*nrots/2);
243  return (position + position_offset);
244 }

262 double TSharc::GetDetectorThickness(TSharcHit &hit,double dist) {

264  if(dist<0.0)
265  dist = fDetectorThickness[hit.GetDetector()];
