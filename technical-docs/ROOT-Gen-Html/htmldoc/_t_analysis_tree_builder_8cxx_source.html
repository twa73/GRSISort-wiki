333 void TAnalysisTreeBuilder::SortFragmentTreeByTimeStamp() {
334  ///Suprisingly, sorts the fragment tree by times stamps.
335  ///It then takes the sorted fragments and puts them into the eventQ.
336  TFragment* currentFrag = 0;
337 
338  //Find the TFragment branch of the tree
339  fCurrentFragTree->SetBranchAddress("TFragment",&currentFrag);
340 
341  fEntries = fCurrentFragTree->GetEntries();
342 
343  //this vector holds all fragments of one built event and get's passed to the event queue
344  std::vector<TFragment>* event = NULL;
345  //this multiset is used to sort the fragments we're reading sequentially (but not necessarily time-ordered) from the tree
346  std::multiset<TFragment,std::less<TFragment> > sortedFragments;//less is the default ordering option, could be left out
347 
348  //loop over all of the fragments in the tree
349  for(int x = 0; x < fEntries; ++x) {
350  if(fCurrentFragTree->GetEntry(x) == -1 ) {
351  //GetEntry Reads all branches of entry and returns total number of bytes read.
352  //This means that if this if statement passes, we have had an I/O error.
353  printf(DRED "FIRE!!!" RESET_COLOR "\n");
354  continue;
355  }
356  fFragmentsIn++;//Now that we have read a new entry, we need to increment our counter
357 
358  //pull the different pile-up hits apart and put the into the sorted buffer as different fragments
359  //for(size_t hit = 0; hit < currentFrag->Cfd.size(); ++hit) { //fragment inserted twice; pcb.
360  try {
361  //sortedFragments.insert(TFragment(*currentFrag, hit));
362  sortedFragments.insert(*currentFrag); //only insert the fragment once. pcb
363  //sortedFragments.emplace(*currentFrag, hit);
364  } catch (std::bad_alloc& e) {
365  //failed to insert the fragment, check overall size of the multiset
366  if(sortedFragments.size() < TGRSIRunInfo::BufferSize()) {
367  printf(BG_RED WHITE "Ran out of memory trying to insert the %dth fragment into the multiset, minimum requested size of multiset is %d, aborting!" RESET_COLOR "\n", (int) sortedFragments.size(), (int) TGRSIRunInfo::BufferSize());
368  abort();
369  } else {
370  //Wait and try inserting again. If that fails as well we can't do anything else, so we don't catch exceptions here
371  std::this_thread::sleep_for(std::chrono::milliseconds(100));
372  //sortedFragments.insert(TFragment(*currentFrag, hit));
373  sortedFragments.insert(*currentFrag); //only insert the fragment once. pcb
374  }
375  }
376  //}
377 
378  ///We start putting fragments into a build event when either the time
379  ///difference between the first and the last fragment is larger than a
380  ///given value, or if the size of the multiset is above a given threshold
381  ///and the first and last of the fragments are at least BuildWindow apart.
382  long firstTimeStamp = (*(sortedFragments.begin())).GetTimeStamp();
383  long lastTimeStamp = (*(std::prev(sortedFragments.end()))).GetTimeStamp();
384  if(//(lastTimeStamp - firstTimeStamp > TGRSIRunInfo::BufferDuration()) ||
385  ((sortedFragments.size() > TGRSIRunInfo::BufferSize()) && ((lastTimeStamp - firstTimeStamp) > TGRSIRunInfo::BuildWindow()))) {
386  //We are now in a situation where we think that no more fragments will be read that will end up at the beginning of the multiset.
387  //So we can put everything that is in the BuildWindow of the first fragment into our built event.
388  //For this we loop through the fragments in the multiset, put them into the new event vector, and once the time difference is larger than BuildWindow,
389  //we put the event into the queue, erase the fragments from the multiset, and stop the loop.
390  event = new std::vector<TFragment>;
391  for(auto it = sortedFragments.begin(); it != sortedFragments.end(); ++it) {
392  if((it->GetTimeStamp() - firstTimeStamp) <= TGRSIRunInfo::BuildWindow()) {
393  event->push_back(*it);
394  if(TGRSIRunInfo::IsMovingWindow()){
395  firstTimeStamp = it->GetTimeStamp(); //THIS IS FOR MOVING WINDOW
396  }
397  } else {
398  //Add the event to the event Q.
399  TEventQueue::Get()->Add(event);
400  //erase the fragments [first,current), i.e. from the first to the current, but not the current one itself
401  sortedFragments.erase(sortedFragments.begin(),it);
402  //this screws up the iterators, so never remove the break statement after this!
403  break;
404  }
405  }
406  }
407  }//loop over entries
408 
409  printf("Finished reading %d fragments, built %d events, got %d fragments left\t\t\t\t\t\t\t\t\t\t\n", fFragmentsIn, fAnalysisIn, (int) sortedFragments.size());
410 
411  //In case we have fragments left after all of the fragments have been processed, we add them to the queue now.
412  while(!sortedFragments.empty()) {
413  long firstTimeStamp = (*(sortedFragments.begin())).GetTimeStamp();
414  event = new std::vector<TFragment>;
415  auto it = sortedFragments.begin();
416  for(; it != sortedFragments.end(); ++it) {
417  if((it->GetTimeStamp() - firstTimeStamp) <= TGRSIRunInfo::BuildWindow()) {
418  event->push_back(*it);
419  if(TGRSIRunInfo::IsMovingWindow()){
420  firstTimeStamp = it->GetTimeStamp(); //THIS IS FOR MOVING WINDOW
421  }
422  } else {
423  //Add the event to the event Q.
424  TEventQueue::Get()->Add(event);
425  //erase the fragments [first,current), i.e. from the first to the current, but not the current one itself
426  sortedFragments.erase(sortedFragments.begin(),it);
427  //this screws up the iterators, so never remove the break statement after this!
428  break;
429  }
430  }
431  //if this is the last fragments, just fill them
432  if(it == sortedFragments.end()) {
433  //Add the event to the event Q.
434  TEventQueue::Get()->Add(event);
435  //erase the fragments [first,current), i.e. from the first to the current, but not the current one itself
436  sortedFragments.erase(sortedFragments.begin(),it);
437  }
438  }//while sortedFragments isn't empty
439 
440  //Drop the TFragmentBranch from the Cache so we aren't still holding it in memory
441  fCurrentFragTree->DropBranchFromCache(fCurrentFragTree->GetBranch("TFragment"),true);
442  fCurrentFragTree->SetCacheSize(0);
443  return;

744 void TAnalysisTreeBuilder::ProcessEvent() {
745  ///Process the event. We do this by filling a map of the detectors, and put the hits into the appropriate detector class.
746  ///This is important because different detector systems will need different experimental data/conditions to be set.
747  while(TEventQueue::Size() > 0 || !fSortFragmentDone) {
748  if(TEventQueue::Size() == 0) {
749  std::this_thread::sleep_for(std::chrono::milliseconds(100));
750  continue;
751  }
752 
753  //We need to pull the event out of the Event Q
754  std::vector<TFragment>* event = TEventQueue::PopEntry();
755  MNEMONIC mnemonic;
756  std::map<std::string, TDetector*>* detectors = new std::map<std::string, TDetector*>;
757  for(size_t i=0;i<event->size();i++) {
758  TChannel* channel = TChannel::GetChannel(event->at(i).ChannelAddress);
759  if(!channel)
760  continue;
761  ClearMNEMONIC(&mnemonic);
762  ParseMNEMONIC(channel->GetChannelName(),&mnemonic);
763 
764  //We use the MNEMONIC in order to figure out what detector we want to put the set of fragments into
765  if(mnemonic.system.compare("TI")==0) {
766  if(detectors->find("TI") == detectors->end()) {
767  (*detectors)["TI"] = new TTigress;
768  }
769  (*detectors)["TI"]->AddFragment(&(event->at(i)), &mnemonic);
770  } else if(mnemonic.system.compare("SH")==0) {
771  if(detectors->find("SH") == detectors->end()) {
772  (*detectors)["SH"] = new TSharc;
773  }
774  (*detectors)["SH"]->AddFragment(&(event->at(i)), &mnemonic);
775  } else if(mnemonic.system.compare("Tr")==0) {
776  if(detectors->find("Tr") == detectors->end()) {
777  (*detectors)["Tr"] = new TTriFoil;
778  }
779  (*detectors)["Tr"]->AddFragment(&(event->at(i)), &mnemonic);
780  } else if(mnemonic.system.compare("RF")==0) {
781  if(detectors->find("RF") == detectors->end()) {
782  (*detectors)["RF"] = new TRF;
783  }
784  (*detectors)["RF"]->AddFragment(&(event->at(i)), &mnemonic);
785  } else if(mnemonic.system.compare("SP")==0) {
786  if(mnemonic.subsystem.compare("I")==0) {
787  if(detectors->find("SPI") == detectors->end()) {
788  (*detectors)["SPI"] = new TSiLi;
789  }
790  (*detectors)["SPI"]->AddFragment(&(event->at(i)), &mnemonic);
791  } else {
792  if(detectors->find("SPE") == detectors->end()) {
793  (*detectors)["SPE"] = new TS3;
794  }
795  (*detectors)["SPE"]->AddFragment(&(event->at(i)), &mnemonic);
796  }
797  } else if(mnemonic.system.compare("CS")==0) {
798  if(detectors->find("CS") == detectors->end()) {
799  (*detectors)["CS"] = new TCSM;
800  }
801  (*detectors)["CS"]->AddFragment(&(event->at(i)), &mnemonic);
802  } else if(mnemonic.system.compare("GR")==0) {
803  if(detectors->find("GR") == detectors->end()) {
804  (*detectors)["GR"] = new TGriffin;
805  }
806  (*detectors)["GR"]->AddFragment(&(event->at(i)), &mnemonic);
807  } else if(mnemonic.system.compare("SE")==0) {
808  if(detectors->find("SE") == detectors->end()) {
809  (*detectors)["SE"] = new TSceptar;
810  }
811  (*detectors)["SE"]->AddFragment(&(event->at(i)), &mnemonic);
812  } else if(mnemonic.system.compare("PA")==0) {
813  if(detectors->find("PA") == detectors->end()) {
814  (*detectors)["PA"] = new TPaces;
815  }
816  (*detectors)["PA"]->AddFragment(&(event->at(i)), &mnemonic);
817  } else if(mnemonic.system.compare("DS")==0) {
818  if(detectors->find("DS") == detectors->end()) {
819  (*detectors)["DS"] = new TDescant;
820  }
821  (*detectors)["DS"]->AddFragment(&(event->at(i)), &mnemonic);
822  //} else if(mnemonic.system.compare("DA")==0) {
823  // AddFragment(&(event->at(i)), &mnemonic);
824  } else if(mnemonic.system.compare("BA")==0) {
825  if(detectors->find("BA") == detectors->end()) {
826  (*detectors)["BA"] = new TS3;
827  }
828  (*detectors)["BA"]->AddFragment(&(event->at(i)), &mnemonic);
829  } else if(mnemonic.system.compare("ZD")==0) {
830  if(detectors->find("ZD") == detectors->end()) {
831  (*detectors)["ZD"] = new TZeroDegree;
832  }
833  (*detectors)["ZD"]->AddFragment(&(event->at(i)), &mnemonic);
834  } else if(mnemonic.system.compare("TP")==0) {
835  if(detectors->find("TP") == detectors->end()) {
836  (*detectors)["TP"] = new TTip;
837  }
838  (*detectors)["TP"]->AddFragment(&(event->at(i)), &mnemonic);
839  }
840  }
841 
842  if(!detectors->empty()) {
843  BuildActiveAnalysisTreeBranches(detectors);
844  FillWriteQueue(detectors);
845  }
846  delete event;
847  }
