641 bool TGRSILoop::ProcessGRIFFIN(uint32_t* ptr, int& dSize, int bank, TMidasEvent* mEvent, TMidasFile* mFile) {
642  unsigned int mserial=0; if(mEvent) mserial = (unsigned int)(mEvent->GetSerialNumber());
643  unsigned int mtime=0; if(mEvent) mtime = (unsigned int)(mEvent->GetTimeStamp());
644 
645  //loop over words in event to find fragment header
646  for(int index = 0; index < dSize;) {
647  if(((ptr[index])&0xf0000000) == 0x80000000) {
648  //if we found a fragment header we pass the data to the data parser which returns the number of words read
649  int words = TDataParser::GriffinDataToFragment(&ptr[index],dSize-index,bank,mserial,mtime);
650  if(words>0) {
651  //we successfully read one event with <words> words, so we advance the index by words
652  ++fFragsReadFromMidas;
653  index += words;
654  } else {
655  //we failed to read the fragment on word <-words>, so advance the index by -words and we create an error message
656  ++fFragsReadFromMidas; // if the midas bank fails, we assume it only had one frag in it... this is just used for a print statement.
657  index -= words;
658  if(!fSuppressError) {
659  if(!TGRSIOptions::LogErrors()) {
660  printf(DRED "\n//**********************************************//" RESET_COLOR "\n");
661  printf(DRED "\nBad things are happening. Failed on datum %i" RESET_COLOR "\n", index);
662  if(mEvent) mEvent->Print(Form("a%i",index-1));
663  printf(DRED "\n//**********************************************//" RESET_COLOR "\n");
664  } else {
665  std::string errfilename;
666  if(mFile) {
667  if(mFile->GetSubRunNumber() != -1) {
668  errfilename.append(Form("error%05i_%03i.log",mFile->GetRunNumber(),mFile->GetSubRunNumber()));
669  } else {
670  errfilename.append(Form("error%05i.log",mFile->GetRunNumber()));
671  }
672  } else {
673  errfilename.append("error_log.log");
674  }
675  FILE* originalstdout = stdout;
676  FILE* errfileptr = freopen(errfilename.c_str(),"a",stdout);
677  printf("\n//**********************************************//\n");
678  if(mEvent) mEvent->Print("a");
679  printf("\n//**********************************************//\n");
680  fclose(errfileptr);
681  stdout = originalstdout;
682  }
683  }
684  }
685  } else {
686  //this is not a fragment header, so we advance the index
687  ++index;
688  }
689  }
690 
691  return true;
692 }
